// Multi-timeframe table with two rows of states:
//
//  IFVG bias (from confirmed IFVGs)
//    IFVG+ confirmed only        -> bullish  (triangle up)
//    IFVG- confirmed only        -> bearish  (triangle down)
//    both confirmed / none       -> neutral  (circle)
//
//  Delivery (from candle opens and closes)
//    Bullish while up-close candles keep opening higher. When a down-close candle CLOSES below the
//    open of the last up-close candle, delivery turns bearish (and the mirror image for bearish delivery:
//    down-close candles keep opening lower; an up-close candle closing above the open of the last
//    down-close candle turns it bullish).
//    Exception: if the delivery changes state several times in a row without any continuation in between
//    (alternating changes), the range formed by those opens is locked: from then on delivery only
//    changes when an up-close candle closes above the TOP open of the range (bullish) or a down-close
//    candle closes below the BOTTOM open of the range (bearish).
//    Inner candles (open and close within the range of the previous candle, of the last up-close candle for
//    down-close candles, or of the last down-close candle for up-close candles) are ignored. If the candle that
//    closes beyond that open does not take the low (high) of the reference candle, the change is deferred until
//    it is taken.
//
// Timeframes: 1W, 1D, 4H, 1H, 15m, 5m, 3m (each can be enabled/disabled).
// Only those >= the chart timeframe are shown. The chart timeframe is added (in order) when it is not already shown.

// FVG detection (same as the FVGs indicator, including suspension blocks) ===
longFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l1
    lo = h3
    blow1 = math.min(o1,c1)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi3 = math.max(o3,c3)
    fhi = hi
    flo = lo
    if bhi3 < blow2
        lo := bhi3
    if blow1 > bhi2
        hi := blow1
    [hi, lo, fhi, flo]

shortFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l3
    lo = h1
    blow3 = math.min(o3,c3)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi1 = math.max(o1,c1)
    fhi = hi
    flo = lo
    if blow3 > bhi2
        hi := blow3
    if bhi1 < blow2
        lo := bhi1
    [hi, lo, fhi, flo]

// IFVG state on the current series (close-based IFVG logic of the FVGs indicator):
//  - bearish IFVG candidate = last bullish (long) FVG; confirmed when a close goes below it
//    (IFVG-), cancelled if price closes back above it
//  - bullish IFVG candidate = last bearish (short) FVG; confirmed when a close goes above it
//    (IFVG+), cancelled if price closes back below it
// A new FVG of the same kind replaces the candidate and resets its confirmation.
// Returns 0 = none confirmed, 1 = IFVG+ only, 2 = IFVG- only, 3 = both.
ifvgCode() =>
    var float bearHi = na
    var float bearLo = na
    var bool bearConf = false
    var float bullHi = na
    var float bullLo = na
    var bool bullConf = false

    [lhi, llo, lfhi, lflo] = longFVGHL(open[1],close[1],high[1],low[1],
         open[2],close[2],high[2],low[2],
         open[3],close[3],high[3],low[3])
    if lfhi - lflo > minfvg and lfhi - lflo < maxfvg
        bearHi := lhi
        bearLo := llo
        bearConf := false

    [shi, slo, sfhi, sflo] = shortFVGHL(open[1],close[1],high[1],low[1],
         open[2],close[2],high[2],low[2],
         open[3],close[3],high[3],low[3])
    if sfhi - sflo > minfvg and sfhi - sflo < maxfvg
        bullHi := shi
        bullLo := slo
        bullConf := false

    if bullConf and close < bullLo
        bullConf := false
    if not na(bullHi) and not bullConf and close > bullHi
        bullConf := true

    if bearConf and close > bearHi
        bearConf := false
    if not na(bearLo) and not bearConf and close < bearLo
        bearConf := true

    (bullConf ? 1 : 0) + (bearConf ? 2 : 0)

// Delivery state on the current series (same logic as the CISD indicator).
//  st    : 1 bullish, -1 bearish, 0 not yet defined
//  ref   : open of the last up-close candle (while bullish) / last down-close candle (while bearish);
//          refLo / refHi are the low / high of that candle
//  conf  : a continuation happened since the last change of state
//  flips : consecutive changes of state without continuation in between
//  inRng : range locked; rTop / rBot = top / bottom open of the range
//  pend  : deferred CISD (-1 bearish waiting for the low of the last up-close candle to be taken,
//          1 bullish waiting for the high of the last down-close candle to be taken)
// Inner candles are excluded. Returns (st + 1) + (inRng ? 3 : 0): 0 bearish, 1 undefined, 2 bullish, +3 range-locked.
deliveryCode() =>
    var int st = 0
    var float ref = na
    var float refLo = na
    var float refHi = na
    var bool conf = true
    var int flips = 0
    var bool inRng = false
    var float rTop = na
    var float rBot = na
    var float sTop = na
    var float sBot = na
    var int pend = 0
    var float pOpen = na
    var float pLvl = na
    var float pNewOpen = na
    var float pNewHi = na
    var float pNewLo = na
    var float lastUpLo = na
    var float lastUpHi = na
    var float lastDnLo = na
    var float lastDnHi = na

    bool up = close > open
    bool dn = close < open

    // inner candle detection (all ranges are those of the raw previous candles)
    bool useHL = innerRange == "High-Low"
    float xLo = math.min(open, close)
    float xHi = math.max(open, close)
    bool inner = false
    if exclInner and bar_index > 0
        float pLo = useHL ? low[1] : math.min(open[1], close[1])
        float pHi = useHL ? high[1] : math.max(open[1], close[1])
        inner := xLo >= pLo and xHi <= pHi
        if not inner and dn and not na(lastUpLo)
            inner := xLo >= lastUpLo and xHi <= lastUpHi
        if not inner and up and not na(lastDnLo)
            inner := xLo >= lastDnLo and xHi <= lastDnHi

    if inner
        // excluded candle: no reference, no continuation, no change of state
        bool skipped = true
    else if st == 0
        if up or dn
            st := up ? 1 : -1
            ref := open
            refLo := low
            refHi := high
            conf := true
            flips := 0
    else if inRng
        // range locked: only a close beyond the top / bottom open changes the delivery
        if up and close > rTop
            st := 1
            ref := open
            refLo := low
            refHi := high
            inRng := false
            conf := true
            flips := 0
            sTop := na
            sBot := na
            pend := 0
        else if dn and close < rBot
            st := -1
            ref := open
            refLo := low
            refHi := high
            inRng := false
            conf := true
            flips := 0
            sTop := na
            sBot := na
            pend := 0
        else
            rTop := math.max(rTop, open)
            rBot := math.min(rBot, open)
    else
        bool flip = false
        float oRef = ref        // level broken by the change of state
        float nRef = open       // new reference (open of the candle of the new direction)
        float nHi = high
        float nLo = low
        if st == 1
            if pend == -1
                // a down-close candle already closed below the open of the last up-close candle without taking its low
                if dn
                    pNewOpen := open
                    pNewHi := high
                    pNewLo := low
                if close < pOpen and low < pLvl
                    flip := true
                    oRef := pOpen
                    nRef := pNewOpen
                    nHi := pNewHi
                    nLo := pNewLo
                else if close > pOpen or up
                    pend := 0
            if not flip
                if up
                    if open > ref
                        // continuation: up-close candle offering a higher open
                        conf := true
                        flips := 0
                        sTop := na
                        sBot := na
                    ref := open
                    refLo := low
                    refHi := high
                else if dn and close < ref
                    if not deferOn or low < refLo
                        flip := true
                    else
                        pend := -1
                        pOpen := ref
                        pLvl := refLo
                        pNewOpen := open
                        pNewHi := high
                        pNewLo := low
        else
            if pend == 1
                // an up-close candle already closed above the open of the last down-close candle without taking its high
                if up
                    pNewOpen := open
                    pNewHi := high
                    pNewLo := low
                if close > pOpen and high > pLvl
                    flip := true
                    oRef := pOpen
                    nRef := pNewOpen
                    nHi := pNewHi
                    nLo := pNewLo
                else if close < pOpen or dn
                    pend := 0
            if not flip
                if dn
                    if open < ref
                        // continuation: down-close candle offering a lower open
                        conf := true
                        flips := 0
                        sTop := na
                        sBot := na
                    ref := open
                    refLo := low
                    refHi := high
                else if up and close > ref
                    if not deferOn or high > refHi
                        flip := true
                    else
                        pend := 1
                        pOpen := ref
                        pLvl := refHi
                        pNewOpen := open
                        pNewHi := high
                        pNewLo := low
        if flip
            if conf
                flips := 1
                sTop := math.max(oRef, nRef)
                sBot := math.min(oRef, nRef)
            else
                flips += 1
                sTop := math.max(nz(sTop, oRef), math.max(oRef, nRef))
                sBot := math.min(nz(sBot, oRef), math.min(oRef, nRef))
            conf := false
            st := -st
            ref := nRef
            refLo := nLo
            refHi := nHi
            pend := 0
            if flips >= lockN
                inRng := true
                rTop := sTop
                rBot := sBot

    // remember the range of this candle for the next ones
    if up
        lastUpLo := useHL ? low : xLo
        lastUpHi := useHL ? high : xHi
    else if dn
        lastDnLo := useHL ? low : xLo
        lastDnHi := useHL ? high : xHi

    (st + 1) + (inRng ? 3 : 0)

// States at the close of the last COMPLETED candle of timeframe tf (does not repaint).
secPair(string tf) =>
    request.security(syminfo.tickerid, tf, [ifvgCode()[1], deliveryCode()[1]], lookahead = barmerge.lookahead_on)

// label of the chart timeframe, written like the others (1W, 1D, 4H, 1H, 15m ...)
tfName(string s) =>
    float m = str.tonumber(s)
    string r = s
    if not na(m)
        r := m >= 60 and m % 60 == 0 ? str.tostring(m / 60, "#") + "h" : str.tostring(m, "#") + "m"
    else if str.endswith(s, "S")
        r := str.lower(s)
    else if s == "D" or s == "W" or s == "M"
        r := "1" + s
    r
curLabel = tfName(timeframe.period)

// disabled / hidden timeframes request the chart timeframe (cheap, result ignored)

// Table ======================================================
int nvis = (v1 ? 1 : 0) + (v2 ? 1 : 0) + (v3 ? 1 : 0) + (v4 ? 1 : 0) + (v5 ? 1 : 0) + (v6 ? 1 : 0) + (v7 ? 1 : 0) + (vCur ? 1 : 0)
int nRows = (showBias ? 1 : 0) + (showDlv ? 1 : 0)
int off = showHdr ? 1 : 0
int biasRow = showBias ? 0 : -1
int dlvRow = showDlv ? (showBias ? 1 : 0) : -1
int axA = math.max(nvis, 1) + off   // size along the timeframes axis
int axB = 1 + nRows                 // size along the rows axis (timeframe label + one per state row)

var table tbl = table.new(tblpos, horiz ? axA : axB, horiz ? axB : axA,
    frame_color = color.new(color.gray, 50), frame_width = 1,
    border_color = color.new(color.gray, 70), border_width = 1)

// slot: timeframe slot (-1 = row header); r: -1 = timeframe label row, 0.. = state rows
put(int slot, int r, string txt, color tc, string tip) =>
    int a = off + slot
    int b = r + 1
    if horiz
        tbl.cell(a, b, txt, text_color = tc, bgcolor = bgCol, text_size = tblsz, tooltip = tip)
    else
        tbl.cell(b, a, txt, text_color = tc, bgcolor = bgCol, text_size = tblsz, tooltip = tip)

if barstate.islast
    array<int> secs = array.from(s1, s2, s3, s4, s5, s6, s7, chartSecs)
    array<bool> vis = array.from(v1, v2, v3, v4, v5, v6, v7, vCur)
    array<int> bs = array.from(b1, b2, b3, b4, b5, b6, b7, bC)
    array<int> ds = array.from(d1, d2, d3, d4, d5, d6, d7, dC)
    array<string> labels = array.from("1W", "1D", "4H", "1H", "15m", "5m", "3m", curLabel)
    // timeframes ordered from the longest to the shortest, the chart timeframe included
    array<int> idxs = array.sort_indices(secs, order.descending)

    if showHdr
        put(-1, -1, "", txtCol, "")
        if showBias
            put(-1, biasRow, "IFVG", txtCol, "IFVG bias: confirmed IFVG+ (bullish) / IFVG- (bearish)")
        if showDlv
            put(-1, dlvRow, "DLV", txtCol, "Delivery state")

    int slot = 0
    for k = 0 to 7
        int i = idxs.get(k)
        if vis.get(i)
            put(slot, -1, labels.get(i), txtCol, "")

            if showBias
                int code = nz(bs.get(i))
                string sym = "●"
                color col = neutCol
                string tip = "No confirmed IFVG: neutral"
                if code == 1
                    sym := "▲"
                    col := bullCol
                    tip := "IFVG+ confirmed: bullish"
                else if code == 2
                    sym := "▼"
                    col := bearCol
                    tip := "IFVG- confirmed: bearish"
                else if code == 3
                    tip := "IFVG+ and IFVG- confirmed: neutral"
                put(slot, biasRow, sym, col, labels.get(i) + " " + tip)

            if showDlv
                int dc = nz(ds.get(i), 1)
                int dst = dc % 3 - 1
                bool rng = dc >= 3
                string dsym = "●"
                color dcol = neutCol
                string dtip = "Delivery not defined yet"
                if dst == 1
                    dsym := "▲"
                    dcol := bullCol
                    dtip := "Delivery bullish"
                else if dst == -1
                    dsym := "▼"
                    dcol := bearCol
                    dtip := "Delivery bearish"
                if rng
                    dtip += " (range locked)"
                put(slot, dlvRow, dsym, dcol, labels.get(i) + " " + dtip)

            slot += 1
