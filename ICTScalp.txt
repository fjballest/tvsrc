// ICTScalp: enter-long / enter-short signals only, to experiment with scalping conditions.
// The idea: a higher timeframe (HTF, 4H by default) read says where price is likely to go, and the chart
// (lower) timeframe looks for entries in that direction. Both are made of indications. Each indication is
// Off, Consider or Mandatory: a direction is allowed when every Mandatory indication agrees and at least N of
// the Consider ones agree (N capped at the number of Consider ones).
//
// HTF bias indications (computed on closed HTF candles, nothing repaints):
//  - OTE rejection: the HTF major leg runs from its origin to its extreme (it flips on an HTF close beyond the
//    origin). An HTF candle that wicks into the OTE band (0.62-0.79) of that leg and closes back out of it
//    gives the leg direction. Once the retracement reached that band, the retracement itself is the minor leg:
//    a candle that wicks into the OTE band of the minor leg and closes back gives the minor leg direction.
//    The last rejection holds until its leg is broken (a close beyond the major origin, or a new major extreme
//    for a minor leg rejection).
//  - FVG draw: towards the nearest HTF FVG price has not come back to yet (bearish FVG above: long,
//    bullish FVG below: short).
//  - Liquidity draw: after an HTF swing low is taken, long (towards the highs); after a swing high, short.
//  - Bias table: agreement of the rightmost columns of the IFVGCISDbias table.
//
// LTF entry indications. Triggers decide when to enter and which FVG sets the stop; at least one trigger has
// to fire on the entry candle:
//  - IFVG leaving: the candle that confirms the IFVG (close mode of FVGs+IFVG+OTE).
//  - IFVG return: a later candle wicks back into the confirmed IFVG without passing its far edge.
//  - FVG return: a candle comes back into a chart FVG of the trade direction and closes at or beyond its CE.
//  One entry per IFVG confirmation and per FVG.
// Confirmations, checked for the FVG / IFVG of the trigger or within the last candles:
//  - OTE: the IFVG is an OTE IFVG (OTE logic of FVGs+IFVG+OTE), or the FVG matches the OTE band of its leg.
//  - Breaker / Mitigation block: the FVG / IFVG overlaps a breaker / mitigation block of the trade direction
//    that broke and made a swing beyond it (logic of FVGs+IFVG+OTE).
//  - Took liquidity: a short-term high (short) / low (long) was taken shortly before.
//  - HTF level rejection: an ICTcotas level was rejected shortly before (wick through, body on the trade side).
//  - HTF OTE line rejection: same, with the 0.62 / 0.705 / 0.79 lines of the HTF legs above.
//  - Delivery: the delivery of the chart timeframe (IFVGCISDbias) matches the trade.
// Time filter: only inside a time interval and / or the ICT killzones (ICTlevels sessions).
// Everything is evaluated on closed candles only.
//
// Show trades: entry at the close of the signal candle, stop past the FVG / IFVG, its candle 2 or the
// short-term swing, take profit at 1:N, at the next short-term high / low, or at the HTF FVG of the draw
// (the near candle of its three or its near edge). Simulate trades follows each trade until the stop or the
// take profit (a candle reaching both counts as the stop), with optional breakeven and trailing stop. The
// results table has a row per indication in use: the trades where it was true, so the indications can be
// compared.
//
// Keep in sync: the sections below are copies. When FVGs+IFVG+OTE (FVG detection, pivots, OTE,
// close-mode IFVG, breakers, mitigation blocks), ICTcotas (levels, mitigation) or IFVGCISDbias (ifvgCode,
// deliveryCode, secPair) change, the same change has to be made here.

longFVGHL(o1,c1,l1,o2,c2,o3,c3,h3) =>
    hi = l1
    lo = h3
    blow1 = math.min(o1,c1)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi3 = math.max(o3,c3)
    // include suspension blocks
    fhi = hi
    flo = lo
    if bhi3 < blow2
        lo := bhi3
    if blow1 > bhi2
        hi := blow1
    [hi,lo, fhi, flo]

    blow2 = math.min(o2,c2)
    bhi1 = math.max(o1,c1)
    // include suspension blocks
    fhi = hi
    flo = lo
    if blow3 > bhi2
        hi := blow3
    if bhi1 < blow2
        lo := bhi1
    [hi, lo, fhi, flo]

// Returns [legHigh, legLow] of the leg that precedes the FVG starting at bar sbar.
priorLeg(int dir, int sbar, PivSet ps) =>
    float h = na
    float l = na
    if dir == -1
        int n = ps.hb.size()
        if n > 0
            for i = n - 1 to 0
                int hb = ps.hb.get(i)
                if hb < sbar and bar_index - hb < 4000
                    float lo = low[bar_index - hb]
                    for k = hb to sbar
                        lo := math.min(lo, low[bar_index - k])
                    h := ps.hp.get(i)
                    l := lo
                    break
    else
        int n = ps.lb.size()
        if n > 0
            for i = n - 1 to 0
                int lb = ps.lb.get(i)
                if lb < sbar and bar_index - lb < 4000
                    float hi = high[bar_index - lb]
                    for k = lb to sbar
                        hi := math.max(hi, high[bar_index - k])
                    l := ps.lp.get(i)
                    h := hi
                    break
    [h, l]

// OTE tracker (FVGs+IFVG+OTE without the bands)
type OteTrack
    int dir = 0
    float zlo = na      // IFVG low
    float zhi = na      // IFVG high
    float legHi = na
    float legLo = na
    float rext = na     // running retrace extreme since the FVG (high for dir -1, low for dir +1)
    bool armed = false  // IFVG confirmed by a close through it
    bool fired = false  // OTE signal already fired for this confirmation
    int ctime = na      // confirmation time
    int cbar = na       // confirmation bar index
    float pHi = na      // prior leg high (leg before the FVG)
    float pLo = na      // prior leg low
    int sbar = na       // bar index where the FVG starts
    bool rejHit = false // confirming candle rejected the OTE of the prior leg
    string tag = ""     // timeframe label

method oteStart(OteTrack t, int dir, float zlo, float zhi, float origin, int sbar, PivSet ps) =>
    t.rejHit := false
    t.sbar := sbar
    t.tag := ps.tag
    [pgh, pgl] = priorLeg(dir, sbar, ps)
    t.pHi := pgh
    t.pLo := pgl
    float o = origin
    if ps.htf
        float adj = dir == -1 ? ps.lastHigh() : ps.lastLow()
        if not na(adj)
            o := dir == -1 ? math.max(origin, adj) : math.min(origin, adj)
    t.dir := dir
    t.zlo := zlo
    t.zhi := zhi
    t.armed := false
    t.fired := false
    t.ctime := na
    t.rext := origin
    t.legHi := dir == -1 ? o : na
    t.legLo := dir == 1 ? o : na

method oteConfirm(OteTrack t) =>
    if t.dir != 0 and not t.armed
        // rejection of the OTE of the PRIOR leg by the confirming candle
        if useOTE and oteRej and not na(t.pHi) and not na(t.pLo) and t.pHi > t.pLo
            float pr = t.pHi - t.pLo
            float ext = na
            float rLo = na
            float rHi = na
            bool hit = false
            if t.dir == -1
                ext := oteRejAny ? math.max(t.rext, high) : high
                rLo := t.pLo + oteMin * pr
                rHi := t.pLo + oteMax * pr
                hit := ext >= rLo and ext <= (oteRejCap ? rHi : t.pHi) and close <= rHi
            else
                ext := oteRejAny ? math.min(t.rext, low) : low
                rLo := t.pHi - oteMax * pr
                rHi := t.pHi - oteMin * pr
                hit := ext <= rHi and ext >= (oteRejCap ? rLo : t.pLo) and close >= rLo
            if hit
                t.rejHit := true
        t.armed := true
        t.fired := t.rejHit // one OTE signal per IFVG
        t.ctime := time
        t.cbar := bar_index
        if t.dir == -1
            t.legHi := math.max(t.legHi, high)
            t.legLo := low
        else
            t.legLo := math.min(t.legLo, low)
            t.legHi := high

method oteCancel(OteTrack t) =>
    t.rejHit := false
    t.armed := false
    if t.dir == -1
        t.legLo := na
    else
        t.legHi := na

// call once per bar. Returns the direction (-1 / 1) on the bar the OTE is matched, else 0
method oteStep(OteTrack t) =>
    int s = 0
    if useOTE and t.dir != 0
        if not t.armed
            if t.dir == -1
                t.legHi := math.max(t.legHi, high)
                t.rext := math.max(t.rext, high)
            else
                t.legLo := math.min(t.legLo, low)
                t.rext := math.min(t.rext, low)
        else
            if t.rejHit
                s := t.dir
                t.rejHit := false
            float r = t.legHi - t.legLo
            if r > 0
                float bLo = na
                float bHi = na
                if t.dir == -1
                    bLo := t.legLo + oteMin * r
                    bHi := t.legLo + oteMax * r
                else
                    bLo := t.legHi - oteMax * r
                    bHi := t.legHi - oteMin * r
                float mid = math.avg(t.zlo, t.zhi)
                bool m = false
                if oteMode == "Overlap"
                    m := t.zhi >= bLo and t.zlo <= bHi
                else if oteMode == "Fully inside"
                    m := t.zlo >= bLo and t.zhi <= bHi
                else
                    m := mid >= bLo and mid <= bHi
                bool tap = t.dir == -1 ? high >= t.zlo : low <= t.zhi
                if m and tap and not t.fired and time != t.ctime and barstate.isconfirmed
                    t.fired := true
                    s := t.dir
            if t.dir == -1
                t.legLo := math.min(t.legLo, low)
            else
                t.legHi := math.max(t.legHi, high)
    s

        if s2 != 0

        g.tags := tg
    s

        if fvgovl == "Enclosing"
            for o in lvls
                if o.kind == kind and o.tfi == tfi and o.dir == dir and not o.inv and not o.dead
                    if lo <= o.hi and hi >= o.lo and o.lo <= lo and o.hi >= hi

                        lvls.remove(i)
                i -= 1
    ok

addLvl(int kind, int tfi, int dir, float lo, float hi, int t) =>
    Lvl l = na
    if (kind != 0 and kind != 3) or fvgOk(kind, tfi, dir, lo, hi)
        l := Lvl.new(kind, tfi, dir, hi, lo, (hi + lo) / 2, math.max(t, tfirst))
        lvls.push(l)
        int keep = kind == 0 ? nfvg : kind == 1 ? nvi : kind == 3 ? nmf : nwk
        int cnt = 0
        for v in lvls
            if v.kind == kind and v.tfi == tfi and not v.dead
                cnt += 1
        if cnt > keep
            for i = 0 to lvls.size() - 1
                Lvl o = lvls.get(i)
                if o.kind == kind and o.tfi == tfi and not o.dead
                    o.kill()
                    lvls.remove(i)
                    break
    l

wickStep(int tfi, int side, bool q, float sz, float lo, float hi, int t) =>
    int idx = tfi * 2 + side
    Lvl cur = wcur.get(idx)
    if not q
        Lvl none = na
        wcur.set(idx, none)
    else if na(cur) or (cur.dead and sz > cur.wsize)
        Lvl n = addLvl(2, tfi, side == 0 ? -1 : 1, lo, hi, t)
        n.wsize := sz
        wcur.set(idx, n)
    else if sz > cur.wsize
        cur.lo := lo
        cur.hi := hi
        cur.ce := (lo + hi) / 2
        cur.t := math.max(t, tfirst)
        cur.wsize := sz
    true

addFvgs(int kind, int tfi, float minsz, float o1, float h1, float l1, float c1,
     float o2, float h2, float l2, float c2, float h3, float l3, float o3, float c3,
     int t2, int tform) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float top3 = math.max(o3, c3)
    float bot3 = math.min(o3, c3)
    if l1 > h3 and l1 - h3 >= minsz
        float flo = h3
        float fhi = l1
        if fvgvi
            if botP > top3 and botP - top3 >= vimin
                flo := math.min(flo, top3)
                fhi := math.max(fhi, botP)
            if botC > topP and botC - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, botC)
        Lvl nb = addLvl(kind, tfi, 1, flo, fhi, t2)
        if not na(nb)
            nb.tfm := tform
    if h1 < l3 and l3 - h1 >= minsz
        float flo = h1
        float fhi = l3
        if fvgvi
            if topP < bot3 and bot3 - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, bot3)
            if topC < botP and botP - topC >= vimin
                flo := math.min(flo, topC)
                fhi := math.max(fhi, botP)
        Lvl ns = addLvl(kind, tfi, -1, flo, fhi, t2)
        if not na(ns)
            ns.tfm := tform
    true

addWicks(int tfi, float o1, float h1, float l1, float c1, int t1) =>
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float rng = h1 - l1
    float uw = h1 - topC
    float lw = botC - l1
    bool uq = rng > 0 and uw >= wkmin / 100.0 * rng
    bool lq = rng > 0 and lw >= wkmin / 100.0 * rng
    wickStep(tfi, 0, uq, uw, topC, h1, t1)
    wickStep(tfi, 1, lq, lw, l1, botC, t1)
    true

processTf(int tfi, float o1, float h1, float l1, float c1, int t1,
     float o2, float h2, float l2, float c2, int t2, float h3, float l3, float o3, float c3) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    if dofvg
        addFvgs(0, tfi, fvgmin, o1, h1, l1, c1, o2, h2, l2, c2, h3, l3, o3, c3, t2, time)
    if dovi
        if botC > topP and l1 <= h2 and botC - topP >= vimin
            addLvl(1, tfi, 1, topP, botC, t1)
        if topC < botP and h1 >= l2 and botP - topC >= vimin
            addLvl(1, tfi, -1, topC, botP, t1)
    if dowk
        addWicks(tfi, o1, h1, l1, c1, t1)
    true

runMit() =>
    int i = lvls.size() - 1
    while i >= 0
        Lvl l = lvls.get(i)
        if l.dead
            lvls.remove(i)
        else
            bool fvgLive = (l.kind == 0 or l.kind == 3) and doifvg and not l.inv
            bool flipped = false
            if fvgLive and tfNew.get(l.tfi)
                float c = tfClose.get(l.tfi)
                if (l.dir == 1 and c < l.lo) or (l.dir == -1 and c > l.hi)
                    l.inv := true
                    l.dir := -l.dir
                    flipped := true
            if mmode > 0 and not fvgLive and not flipped
                bool m = false
                // close rule: on a close of the level's own timeframe, never on a chart close
                bool tn = tfNew.get(l.tfi)
                float tc = tfClose.get(l.tfi)
                if l.dir == 1
                    m := mmode == 1 ? low <= l.ce : mmode == 2 ? low <= l.lo : tn and tc < l.lo
                else
                    m := mmode == 1 ? high >= l.ce : mmode == 2 ? high >= l.hi : tn and tc > l.hi
                if m
                    l.kill()
                    lvls.remove(i)
        i -= 1
    true

                        bear := true
                    if low <= p and bBot > p
                        bull := true
    [bear, bull]

// Returns 0 = none confirmed, 1 = IFVG+ only, 2 = IFVG- only, 3 = both.
ifvgCode() =>
    var float bearHi = na
    var float bearLo = na
    var bool bearConf = false
    var float bullHi = na
    var float bullLo = na
    var bool bullConf = false

    [lhi, llo, lfhi, lflo] = longFVGHL(open[1], close[1], low[1], open[2],
         close[2], open[3], close[3], high[3])
    if lfhi - lflo > bMinfvg and lfhi - lflo < bMaxfvg
        bearHi := lhi
        bearLo := llo
        bearConf := false

    [shi, slo, sfhi, sflo] = shortFVGHL(open[1], close[1], high[1], open[2],
         close[2], open[3], close[3], low[3])
    if sfhi - sflo > bMinfvg and sfhi - sflo < bMaxfvg
        bullHi := shi
        bullLo := slo
        bullConf := false

    if bullConf and close < bullLo
        bullConf := false
    if not na(bullHi) and not bullConf and close > bullHi
        bullConf := true

    if bearConf and close > bearHi
        bearConf := false
    if not na(bearLo) and not bearConf and close < bearLo
        bearConf := true

    (bullConf ? 1 : 0) + (bearConf ? 2 : 0)

// Returns (st + 1) + (inRng ? 3 : 0): 0 bearish, 1 undefined, 2 bullish, +3 range-locked.
deliveryCode() =>
    var int st = 0
    var float ref = na
    var float refLo = na
    var float refHi = na
    var bool conf = true
    var int flips = 0
    var bool inRng = false
    var float rTop = na
    var float rBot = na
    var float sTop = na
    var float sBot = na
    var int pend = 0
    var float pOpen = na
    var float pLvl = na
    var float pNewOpen = na
    var float pNewHi = na
    var float pNewLo = na
    var float lastUpLo = na
    var float lastUpHi = na
    var float lastDnLo = na
    var float lastDnHi = na

    bool up = close > open
    bool dn = close < open

    // inner candle detection (all ranges are those of the raw previous candles)
    bool useHL = innerRange == "High-Low"
    float xLo = math.min(open, close)
    float xHi = math.max(open, close)
    bool inner = false
    if exclInner and bar_index > 0
        float pLo = useHL ? low[1] : math.min(open[1], close[1])
        float pHi = useHL ? high[1] : math.max(open[1], close[1])
        inner := xLo >= pLo and xHi <= pHi
        if not inner and dn and not na(lastUpLo)
            inner := xLo >= lastUpLo and xHi <= lastUpHi
        if not inner and up and not na(lastDnLo)
            inner := xLo >= lastDnLo and xHi <= lastDnHi

    if inner
        // excluded candle: no reference, no continuation, no change of state
        bool skipped = true
    else if st == 0
        if up or dn
            st := up ? 1 : -1
            ref := open
            refLo := low
            refHi := high
            conf := true
            flips := 0
    else if inRng
        // range locked: only a close beyond the top / bottom open changes the delivery
        if up and close > rTop
            st := 1
            ref := open
            refLo := low
            refHi := high
            inRng := false
            conf := true
            flips := 0
            sTop := na
            sBot := na
            pend := 0
        else if dn and close < rBot
            st := -1
            ref := open
            refLo := low
            refHi := high
            inRng := false
            conf := true
            flips := 0
            sTop := na
            sBot := na
            pend := 0
        else
            rTop := math.max(rTop, open)
            rBot := math.min(rBot, open)
    else
        bool flip = false
        float oRef = ref        // level broken by the change of state
        float nRef = open       // new reference (open of the candle of the new direction)
        float nHi = high
        float nLo = low
        if st == 1
            if pend == -1
                if dn
                    pNewOpen := open
                    pNewHi := high
                    pNewLo := low
                if close < pOpen and low < pLvl
                    flip := true
                    oRef := pOpen
                    nRef := pNewOpen
                    nHi := pNewHi
                    nLo := pNewLo
                else if close > pOpen or up
                    pend := 0
            if not flip
                if up
                    if open > ref
                        // continuation: up-close candle offering a higher open
                        conf := true
                        flips := 0
                        sTop := na
                        sBot := na
                    ref := open
                    refLo := low
                    refHi := high
                else if dn and close < ref
                    if not deferOn or low < refLo
                        flip := true
                    else
                        pend := -1
                        pOpen := ref
                        pLvl := refLo
                        pNewOpen := open
                        pNewHi := high
                        pNewLo := low
        else
            if pend == 1
                if up
                    pNewOpen := open
                    pNewHi := high
                    pNewLo := low
                if close > pOpen and high > pLvl
                    flip := true
                    oRef := pOpen
                    nRef := pNewOpen
                    nHi := pNewHi
                    nLo := pNewLo
                else if close < pOpen or dn
                    pend := 0
            if not flip
                if dn
                    if open < ref
                        // continuation: down-close candle offering a lower open
                        conf := true
                        flips := 0
                        sTop := na
                        sBot := na
                    ref := open
                    refLo := low
                    refHi := high
                else if up and close > ref
                    if not deferOn or high > refHi
                        flip := true
                    else
                        pend := 1
                        pOpen := ref
                        pLvl := refHi
                        pNewOpen := open
                        pNewHi := high
                        pNewLo := low
        if flip
            if conf
                flips := 1
                sTop := math.max(oRef, nRef)
                sBot := math.min(oRef, nRef)
            else
                flips += 1
                sTop := math.max(nz(sTop, oRef), math.max(oRef, nRef))
                sBot := math.min(nz(sBot, oRef), math.min(oRef, nRef))
            conf := false
            st := -st
            ref := nRef
            refLo := nLo
            refHi := nHi
            pend := 0
            if flips >= lockN
                inRng := true
                rTop := sTop
                rBot := sBot

    // remember the range of this candle for the next ones
    if up
        lastUpLo := useHL ? low : xLo
        lastUpHi := useHL ? high : xHi
    else if dn
        lastDnLo := useHL ? low : xLo
        lastDnHi := useHL ? high : xHi

    (st + 1) + (inRng ? 3 : 0)

// ============================================================ HTF bias
// Runs on the HTF candles (request.security); every value is that of the last closed HTF candle.
// Major leg: ld 1 up (org low, ext high), -1 down (org high, ext low); flips on a close beyond org.
// rx: retracement extreme since ext (lowest low of an up leg, highest high of a down leg).
htfCode() =>
    var int ld = 0
    var float org = na
    var float ext = na
    var float rx = na
    var int bOte = 0          // OTE rejection direction
    var int bOteSrc = 0       // 1 major leg, 2 minor leg
    var int bLiq = 0          // liquidity draw
    var float sh = na       // last swing high / low not taken yet
    var float sl = na
    var array<float> uN = array.new<float>()    // bearish FVGs (above): near edge (bottom), newest candle low
    var array<float> uC = array.new<float>()
    var array<float> dN = array.new<float>()    // bullish FVGs (below): near edge (top), newest candle high
    var array<float> dC = array.new<float>()
    float ph = ta.pivothigh(high, oteHtfL, oteHtfR)
    float pl = ta.pivotlow(low, oteHtfL, oteHtfR)
    // --- OTE rejection
    if ld == 0
        ld := 1
        org := low
        ext := high
        rx := high
    else
        float r = math.abs(ext - org)
        float m62 = ext - ld * oteMin * r
        float m79 = ext - ld * oteMax * r
        float mr = math.abs(ext - rx)
        bool minorOk = r > 0 and mr > 0 and (ld == 1 ? rx <= m62 : rx >= m62)
        if ld == 1 ? close < org : close > org
            ld := -ld
            org := ext
            ext := ld == 1 ? high : low
            rx := ext
            bOte := 0
            bOteSrc := 0
        else if ld == 1 ? high > ext : low < ext
            ext := ld == 1 ? high : low
            rx := ext
            if bOteSrc == 2
                bOte := 0
                bOteSrc := 0
        else
            bool maj = r > 0 and (ld == 1 ? low <= m62 and low >= org and close >= m79
                 : high >= m62 and high <= org and close <= m79)
            // minor leg: from ext to rx; price back from rx towards ext into its OTE band
            float n62 = rx + ld * oteMin * mr
            float n79 = rx + ld * oteMax * mr
            bool mnr = minorOk and (ld == 1 ? high >= n62 and high <= ext and close <= n79
                 : low <= n62 and low >= ext and close >= n79)
            if maj
                bOte := ld
                bOteSrc := 1
            else if mnr
                bOte := -ld
                bOteSrc := 2
            rx := ld == 1 ? math.min(rx, low) : math.max(rx, high)
    float rr = math.abs(ext - org)
    float mm = math.abs(ext - rx)
    bool mOk = rr > 0 and mm > 0 and (ld == 1 ? rx <= ext - oteMin * rr : rx >= ext + oteMin * rr)
    float a62 = rr > 0 ? ext - ld * oteMin * rr : na
    float a705 = rr > 0 ? ext - ld * (oteMin + oteMax) / 2 * rr : na
    float a79 = rr > 0 ? ext - ld * oteMax * rr : na
    float i62 = mOk ? rx + ld * oteMin * mm : na
    float i705 = mOk ? rx + ld * (oteMin + oteMax) / 2 * mm : na
    float i79 = mOk ? rx + ld * oteMax * mm : na
    // --- liquidity draw
    bool tH = not na(sh) and high > sh
    bool tL = not na(sl) and low < sl
    if tH and not tL
        bLiq := -1
    else if tL and not tH
        bLiq := 1
    if tH
        sh := na
    if tL
        sl := na
    if not na(ph)
        sh := ph
    if not na(pl)
        sl := pl
    // --- FVG draw: FVGs price has not come back to
    if uN.size() > 0
        for i = uN.size() - 1 to 0
            if high >= uN.get(i)
                uN.remove(i)
                uC.remove(i)
    if dN.size() > 0
        for i = dN.size() - 1 to 0
            if low <= dN.get(i)
                dN.remove(i)
                dC.remove(i)
    if high < low[2] and low[2] - high > minfvg and low[2] - high < maxfvg
        uN.push(high)
        uC.push(low)
        if uN.size() > 10
            uN.shift()
            uC.shift()
    if low > high[2] and low - high[2] > minfvg and low - high[2] < maxfvg
        dN.push(low)
        dC.push(high)
        if dN.size() > 10
            dN.shift()
            dC.shift()
    float upN = na
    float upC = na
    for [i, v] in uN
        if na(upN) or v < upN
            upN := v
            upC := uC.get(i)
    float dnN = na
    float dnC = na
    for [i, v] in dN
        if na(dnN) or v > dnN
            dnN := v
            dnC := dC.get(i)
    int bFvg = na(upN) and na(dnN) ? 0 : na(dnN) ? 1 : na(upN) ? -1 : (upN - close <= close - dnN ? 1 : -1)
    [bOte[1], bFvg[1], bLiq[1], a62[1], a705[1], a79[1], i62[1], i705[1], i79[1], upN[1], upC[1], dnN[1], dnC[1]]

// [pass, bit mask of the true indications] for indications i0..i1 with values v (index 0 is i0) and N
indCheck(int i0, int i1, array<bool> v, int n) =>
    bool ok = true
    int cons = 0
    int cnt = 0
    int mask = 0
    for i = i0 to i1
        int m = indMode.get(i)
        bool t = v.get(i - i0)
        if m > 0 and t
            mask += int(math.pow(2, i))
        if m == 2 and not t
            ok := false
        if m == 1
            cons += 1
            if t
                cnt += 1
    [ok and cnt >= math.min(n, cons), mask]

bitOn(int mask, int i) =>
    int(mask / math.pow(2, i)) % 2 == 1

// ============================================================ breakers and mitigation blocks (FVGs+IFVG+OTE, state only)
type Blk
    int kind        // 0 breaker, 1 mitigation block
    int dir         // 1 bullish (long after the pullback), -1 bearish
    float hi        // zone
    float lo
    bool broken = false
    int bkBar = na  // bar of the close beyond the zone
    bool swung = false  // a swing beyond the zone formed after the break (needed for the pullback)
    bool tagged = false // first pullback with a matching FVG / IFVG seen
    int zb = na     // bar of the last candle of the zone (age)
    int pb = na     // bar of the last pullback into the zone (age once tagged)

// swing sequence and armed levels of one kind (ICTblocks)
type BlkSeq
    array<float> p
    array<int> b
    array<int> t
    float upLvl = na    // swing high waiting for a close above it (bullish), its bar, the last one used
    int upBar = na
    int upDone = na
    float dnLvl = na    // swing low waiting for a close below it (bearish)
    int dnBar = na
    int dnDone = na

var array<Blk> blks = array.new<Blk>()
var BlkSeq bSeq = BlkSeq.new(array.new<float>(), array.new<int>(), array.new<int>())
var BlkSeq mSeq = BlkSeq.new(array.new<float>(), array.new<int>(), array.new<int>())
float bPH = ta.pivothigh(high, brkLen, brkLen)
float bPL = ta.pivotlow(low, brkLen, brkLen)
float mPH = ta.pivothigh(high, mbLen, mbLen)
float mPL = ta.pivotlow(low, mbLen, mbLen)
bool blkOn = modeOf(emBrk) > 0 or modeOf(emMb) > 0

blkPivot(array<float> pr, array<int> br, array<int> ty, float price, int bar, int kind) =>
    bool changed = false
    int n = ty.size()
    if n > 0 and ty.get(n - 1) == kind
        if kind == 1 ? price > pr.get(n - 1) : price < pr.get(n - 1)
            pr.set(n - 1, price)
            br.set(n - 1, bar)
            changed := true
    else
        pr.push(price)
        br.push(bar)
        ty.push(kind)
        changed := true
        while ty.size() > 3
            pr.shift()
            br.shift()
            ty.shift()
    changed

// block zone at the swing k0 bars back: the up-close (dir 1) / down-close (dir -1) candle there or among
// the 3 before it, high to low or its body, plus the consecutive ones before it in "Whole sequence" mode.
// Returns [hi, lo, time of its last candle]
blkZone(int dir, int k0, bool body, bool whole) =>
    float zh = na
    float zl = na
    int zt1 = na
    if k0 + 3 <= 300
        int k = k0
        while k <= k0 + 3 and na(zt1)
            if (close[k] - open[k]) * dir > 0
                zh := body ? math.max(open[k], close[k]) : high[k]
                zl := body ? math.min(open[k], close[k]) : low[k]
                zt1 := time[k]
            k += 1
        if not na(zt1) and whole
            while k <= 300 and (close[k] - open[k]) * dir > 0
                zh := math.max(zh, body ? math.max(open[k], close[k]) : high[k])
                zl := math.min(zl, body ? math.min(open[k], close[k]) : low[k])
                k += 1
    [zh, zl, zt1]

blkAdd(int kind, int dir, float zh, float zl, int zt1) =>
    int zk = 0
    while zk < 300 and time[zk] > zt1
        zk += 1
    blks.push(Blk.new(kind, dir, zh, zl, true, bar_index, zb = bar_index - zk))
    int keep = kind == 0 ? brkKeep : mbKeep
    int cnt = 0
    for b in blks
        if b.kind == kind and b.dir == dir
            cnt += 1
    if cnt > keep
        for i = 0 to blks.size() - 1
            Blk o = blks.get(i)
            if o.kind == kind and o.dir == dir
                blks.remove(i)
                break
    true

// chart FVGs not closed through nor used yet: bullish (long) / bearish (short), for the FVG return trigger
// and the pullback tags of the blocks
var array<FVG> fvL = array.new<FVG>()
var array<FVG> fvS = array.new<FVG>()
if barstate.isconfirmed
    if fvL.size() > 0
        for i = fvL.size() - 1 to 0
            if close < fvL.get(i).lo or fvL.get(i).used
                fvL.remove(i)
    if fvS.size() > 0
        for i = fvS.size() - 1 to 0
            if close > fvS.get(i).hi or fvS.get(i).used
                fvS.remove(i)
    if not na(lastlongfvg)
        fvL.push(lastlongfvg)
        if fvL.size() > 30
            fvL.shift()
    if not na(lastshortfvg)
        fvS.push(lastshortfvg)
        if fvS.size() > 30
            fvS.shift()

ovl(FVG f, Blk b) =>
    not na(f) and f.hi >= b.lo and f.lo <= b.hi

// a live FVG or confirmed IFVG of the block direction overlaps it (the pullback tag of FVGs+IFVG+OTE)
blkMatch(Blk b) =>
    bool r = false
    array<FVG> fs = b.dir == 1 ? fvL : fvS
    for f in fs
        if ovl(f, b)
            r := true
    if b.dir == 1 and not na(bullconfirm) and ovl(bullifvg, b)
        r := true
    if b.dir == -1 and not na(bearconfirm) and ovl(bearifvg, b)
        r := true
    r

// breaker (kind 0): bearish = high, low, higher high; bullish = low, high, lower low.
// mitigation block (kind 1): bearish = high, low, lower high; bullish = low, high, higher low.
// Armed on the swing between the first two; formed on the close through it.
if blkOn and barstate.isconfirmed
    for kind = 0 to 1
        BlkSeq s = kind == 0 ? bSeq : mSeq
        float ph = kind == 0 ? bPH : mPH
        float pl = kind == 0 ? bPL : mPL
        int len = kind == 0 ? brkLen : mbLen
        int age = kind == 0 ? brkAge : mbAge
        bool body = (kind == 0 ? brkPart : mbPart) == "Body"
        bool whole = (kind == 0 ? brkMode : mbMode) == "Whole sequence"
        if not na(ph) and blkPivot(s.p, s.b, s.t, ph, bar_index - len, 1)
            if s.t.size() == 3 and s.t.get(0) == 1 and s.b.get(1) != nz(s.dnDone, -1)
                bool ok = kind == 0 ? s.p.get(2) > s.p.get(0) : s.p.get(2) < s.p.get(0)
                s.dnLvl := ok ? s.p.get(1) : na
                s.dnBar := s.b.get(1)
        if not na(pl) and blkPivot(s.p, s.b, s.t, pl, bar_index - len, -1)
            if s.t.size() == 3 and s.t.get(0) == -1 and s.b.get(1) != nz(s.upDone, -1)
                bool ok = kind == 0 ? s.p.get(2) < s.p.get(0) : s.p.get(2) > s.p.get(0)
                s.upLvl := ok ? s.p.get(1) : na
                s.upBar := s.b.get(1)
        if not na(s.dnLvl) and bar_index - s.dnBar > age
            s.dnLvl := na
        if not na(s.upLvl) and bar_index - s.upBar > age
            s.upLvl := na
        if not na(s.dnLvl) and close < s.dnLvl
            [zh, zl, zt1] = blkZone(-1, bar_index - s.dnBar, body, whole)
            if not na(zt1)
                blkAdd(kind, -1, zh, zl, zt1)
            s.dnDone := s.dnBar
            s.dnLvl := na
        if not na(s.upLvl) and close > s.upLvl
            [zh, zl, zt1] = blkZone(1, bar_index - s.upBar, body, whole)
            if not na(zt1)
                blkAdd(kind, 1, zh, zl, zt1)
            s.upDone := s.upBar
            s.upLvl := na
    if blks.size() > 0
        for i = blks.size() - 1 to 0
            Blk b = blks.get(i)
            int age = b.kind == 0 ? brkAge : mbAge
            float sPH = b.kind == 0 ? bPH : mPH
            float sPL = b.kind == 0 ? bPL : mPL
            int len = b.kind == 0 ? brkLen : mbLen
            bool thru = (b.dir == 1 and close < b.lo) or (b.dir == -1 and close > b.hi)
            bool pull = (b.dir == 1 and low <= b.hi) or (b.dir == -1 and high >= b.lo)
            if b.tagged
                if thru or bar_index - b.pb > age
                    blks.remove(i)
                else if pull
                    b.pb := bar_index
            else if bar_index - b.zb > age
                blks.remove(i)  // no pullback in time
            else if not b.broken
                if (b.dir == 1 and close > b.hi) or (b.dir == -1 and close < b.lo)
                    b.broken := true
                    b.bkBar := bar_index
            else if not b.swung
                if b.dir == 1 and not na(sPH) and bar_index - len >= b.bkBar and sPH > b.hi
                    b.swung := true
                else if b.dir == -1 and not na(sPL) and bar_index - len >= b.bkBar and sPL < b.lo
                    b.swung := true
                else if thru
                    blks.remove(i)
            else if thru
                blks.remove(i)
            else if pull and blkMatch(b)
                b.tagged := true
                b.pb := bar_index

// a block of this kind and direction, past its swing, overlaps f

        for b in blks
            if b.kind == kind and b.dir == dir and b.swung and ovl(f, b)
                r := true
    r

// FVG of direction dir that this candle comes back into, closing at or beyond its CE (newest first)
fvgReturn(int dir) =>
    FVG m = na
    array<FVG> fs = dir == 1 ? fvL : fvS
    if fs.size() > 0
        for i = fs.size() - 1 to 0
            FVG f = fs.get(i)
            if na(m) and not f.used and (dir == 1 ? low <= f.hi and close >= f.ce : high >= f.lo and close <= f.ce)
                m := f
    m

// OTE of a plain FVG: it matches the OTE band of its leg, from the last chart swing before the FVG (low for
// a bullish FVG) to the extreme since the FVG, before this candle
oteFvg(int dir, FVG f) =>
    bool r = false
    int k0 = math.min(bar_index - f.sbar, 4000)
    if k0 >= 1
        [pgh, pgl] = priorLeg(-dir, f.sbar, pv0)
        float o = dir == 1 ? pgl : pgh
        float ex = na
        for k = 1 to k0
            ex := na(ex) ? (dir == 1 ? high[k] : low[k]) : (dir == 1 ? math.max(ex, high[k]) : math.min(ex, low[k]))
        if not na(o) and not na(ex) and (ex - o) * dir > 0
            float rg = math.abs(ex - o)
            float bLo = dir == 1 ? ex - oteMax * rg : ex + oteMin * rg
            float bHi = dir == 1 ? ex - oteMin * rg : ex + oteMax * rg
            if oteMode == "Overlap"
                r := f.hi >= bLo and f.lo <= bHi
            else if oteMode == "Fully inside"
                r := f.lo >= bLo and f.hi <= bHi
            else
                r := f.ce >= bLo and f.ce <= bHi
    r

         not na(sw) and bar_index - sw < liqLook, not na(lr) and bar_index - lr < htfLook,
         not na(orj) and bar_index - orj < htfLook, dlvCur == dir), eN)
    [ok, mask]

// ============================================================ trades
type Trade
    int dir         // 1 long, -1 short
    float entry
    float stop      // current stop (moved by breakeven / trailing)
    float tp
    int bar         // entry bar
    line eln
    line sln
    line tln
    bool done = false
    box rbx = na    // risk box (entry to initial stop)
    box gbx = na    // reward box (entry to take profit)
    float best = na // best price reached after the entry
    bool trail = false
    label lbl = na  // points label above the trade
    string ltxt = ""
    int mask = 0    // indications that were true

// totals of the simulated trades: row 0 all, row 1 + i indication i
var array<int> rsN = array.new<int>(15, 0)
var array<int> rsW = array.new<int>(15, 0)
var array<int> rsL = array.new<int>(15, 0)
var array<float> rsP = array.new<float>(15, 0.0)

var array<Trade> trades = array.new<Trade>()

statAdd(int row, float pts) =>
    rsN.set(row, rsN.get(row) + 1)
    if pts > 0
        rsW.set(row, rsW.get(row) + 1)
    else if pts < 0
        rsL.set(row, rsL.get(row) + 1)
    rsP.set(row, rsP.get(row) + pts)
    true

// stop of an open trade after a closed candle: breakeven once a candle closes in profit by more than
// beMin, trailing to the low (long) / high (short) of each candle once price went trPts in favour (best
// is the best price reached so far). The stop never moves backwards.
manageStop(int dir, float entry, float stp, bool trail, float best) =>
    float st = stp
    bool tr = trail
    if beOn and (close - entry) * dir > beMin
        st := dir == 1 ? math.max(st, entry) : math.min(st, entry)
    if trOn and (best - entry) * dir >= trPts
        tr := true
    if tr
        st := dir == 1 ? math.max(st, low) : math.min(st, high)
    [st, tr]

// follow the open trades (from the candle after the entry): the stop and take profit are tested with
// the levels set at the previous close, then the stop is managed with this candle
if showTrades and simulate and barstate.isconfirmed
    for t in trades
        if not t.done and bar_index > t.bar
            bool hitS = t.dir == -1 ? high >= t.stop : low <= t.stop
            bool hitT = not t.trail and (t.dir == -1 ? low <= t.tp : high >= t.tp)
            t.eln.set_x2(bar_index)
            t.sln.set_x2(bar_index)
            t.tln.set_x2(bar_index)
            if not na(t.rbx)
                t.rbx.set_right(bar_index)
                t.gbx.set_right(bar_index)
            if hitS or hitT
                t.done := true
                float ex = hitS ? t.stop : t.tp
                float pts = (ex - t.entry) * t.dir
                string res = pts > 0 ? "WIN" : pts < 0 ? "LOSS" : "BE"
                if not na(t.gbx) and (ex - t.tp) * t.dir > 0
                    if t.dir == 1
                        t.gbx.set_top(ex)
                    else
                        t.gbx.set_bottom(ex)
                statAdd(0, pts)
                for i = 0 to 13
                    if bitOn(t.mask, i)
                        statAdd(1 + i, pts)
                color rc = pts > 0 ? tpCol : pts < 0 ? stopCol : entryCol
                if not na(t.lbl)
                    t.lbl.set_text(t.ltxt + "\n" + res + " " + (pts > 0 ? "+" : "") + str.tostring(pts, "#.##"))
                    t.lbl.set_color(rc)
                else
                    label.new(bar_index, ex, res + " " + str.tostring(pts, "#.##"),
                         style = label.style_label_left, color = color.new(color.white, 100),
                         textcolor = rc, size = size.small)
            else if beOn or trOn
                t.best := na(t.best) ? (t.dir == 1 ? high : low)
                     : (t.dir == 1 ? math.max(t.best, high) : math.min(t.best, low))
                [st, tr] = manageStop(t.dir, t.entry, t.stop, t.trail, t.best)
                t.stop := st
                t.trail := tr
                t.sln.set_y1(st)
                t.sln.set_y2(st)

// short-term high (dir -1) / low (dir 1) formed from bar b0 to this bar
swingFrom(int dir, int b0) =>
    float r = na
    array<int> bs = dir == -1 ? sthB : stlB
    array<float> ps = dir == -1 ? sthP : stlP
    int n = bs.size()
    if n > 0
        for i = n - 1 to 0
            int b = bs.get(i)
            if b < b0
                break
            float p = ps.get(i)
            r := na(r) ? p : (dir == -1 ? math.max(r, p) : math.min(r, p))
    if na(r)
        int k0 = math.min(bar_index - b0, 4900)
        for k = 0 to k0
            r := na(r) ? (dir == -1 ? high[k] : low[k]) : (dir == -1 ? math.max(r, high[k]) : math.min(r, low[k]))
    r

// safest of candle 2 and the wicks against the trade (dir -1: highs, dir 1: lows) that reached the FVG,
// from candle 2 to this bar
c2Wicks(int dir, FVG f) =>
    float r = dir == -1 ? f.c2hi : f.c2lo
    int k0 = math.min(bar_index - f.sbar - 1, 4900)
    if k0 >= 0
        for k = 0 to k0
            if dir == -1 and high[k] >= f.lo
                r := math.max(r, high[k])
            else if dir == 1 and low[k] <= f.hi
                r := math.min(r, low[k])
    r

// stop and take profit of a trade entered at the close of this candle; ok = false when the stop is
// not on the losing side of the entry
tradeLevels(int dir, FVG f) =>
    float lvl = stopMode == "Candle 2" ? (dir == -1 ? f.c2hi : f.c2lo)
         : stopMode == "Candle 2 and IFVG wicks" ? c2Wicks(dir, f)
         : stopMode == "Short-term high / low" ? swingFrom(dir, f.sbar)
         : (dir == -1 ? f.hi : f.lo)
    float stp = lvl - dir * stopDelta
    float risk = (close - stp) * dir
    float tp = close + dir * rrN * risk
    // nearest short-term high (long) / low (short) not taken yet, at least 1:N away
    if tpMode == "Next short-term high / low"
        float nx = na
        array<float> qs = dir == 1 ? uhP : ulP
        for q in qs
            if (q - tp) * dir >= 0
                nx := na(nx) ? q : (dir == 1 ? math.min(nx, q) : math.max(nx, q))
        if not na(nx)
            tp := nx
    // HTF FVG of the draw in the trade direction
    else if tpMode == "HTF FVG: near candle" or tpMode == "HTF FVG: near edge"
        float edge = dir == 1 ? hUpN : hDnN
        float cnd = dir == 1 ? hUpC : hDnC
        float h = tpMode == "HTF FVG: near candle" and not na(cnd) and (cnd - close) * dir > 0 ? cnd : edge
        if not na(h) and (h - close) * dir > 0
            tp := h
    [stp, tp, risk > 0]

maskText(int mask) =>
    string s = ""
    for i = 0 to 13
        if bitOn(mask, i)
            s += (s == "" ? "" : " ") + indShort.get(i)
    s

newTrade(int dir, FVG f, int mask) =>
    [stp, tp, ok] = tradeLevels(dir, f)
    if ok
        int x2 = simulate ? bar_index : bar_index + segLen
        line e = line.new(bar_index, close, x2, close, color = entryCol, width = 2)
        line sl = line.new(bar_index, stp, x2, stp, color = stopCol, width = 2)
        line tl = line.new(bar_index, tp, x2, tp, color = tpCol, width = 2)
        box rb = na
        box gb = na
        if boxOn
            rb := box.new(bar_index, math.max(close, stp), x2, math.min(close, stp), border_width = 0,
                 bgcolor = color.new(stopCol, boxTr))
            gb := box.new(bar_index, math.max(close, tp), x2, math.min(close, tp), border_width = 0,
                 bgcolor = color.new(tpCol, boxTr))
        label lb = na
        string lt = "SL " + str.tostring(math.abs(close - stp), "#.##") + "   TP "
             + str.tostring(math.abs(tp - close), "#.##") + "\n" + maskText(mask)
        if lblOn
            lb := label.new(bar_index, math.max(close, stp, tp), lt, style = label.style_label_down,
                 color = color.new(entryCol, 10), textcolor = color.white, size = size.normal)
        trades.push(Trade.new(dir, close, stp, tp, bar_index, e, sl, tl, not simulate, rb, gb, na, false, lb, lt, mask))
        // keep the list short; old lines are removed by the 500 line limit
        if trades.size() > 160
            trades.shift()
    true

if showTrades and goShort

if showTrades and goLong
