// [OTE] modified: IFVGs can now be filtered/flagged by ICT OTE (0.62 - 0.79 retracement of the leg)
// [OTEF] plain FVGs flagged OTE too: leg from the last swing low (bullish) / high (bearish) before the FVG to the
//   extreme since; the FVG is tagged OTE when price retraces into it while it matches the OTE band of that leg
// [DE] Delivery Engine features (from Nuno's Delivery Engine V20, updated to V29), all off by default:
//   - displacement filter: FVGs only when the middle candle range exceeds N standard deviations
//   - inversion rules: full-body confirmation, several inversions per zone (IFVG x2...), invalidation
//   - A+ setups: IFVG right after a sweep of the opposite swing liquidity (killzone, HTF structure), and FVGs
//     born from an MSS with displacement after such a sweep; tagged A+ on the boxes
// Label parts: the IFVG labels can show or hide the OTE timeframes, the +/- sign and the A+ tag.
// [BRK] Breakers: bullish = low, high, lower low (zone: the last up-close candle making the high, or the whole
//   sequence of up-close candles there); bearish = high, low, higher high (last down-close candle(s) making the
//   low). Once price closes beyond the zone, makes a swing high (bullish) / low (bearish) beyond it and pulls
//   back to it while a live FVG / IFVG of the trade direction
//   overlaps the zone, the breaker is drawn as a dashed rectangle (with its mid line) from its candle(s) to that
//   pullback. The label of an overlapping IFVG turns bold; an overlapping unlabeled FVG gets B↑ / B↓.
//   The rectangle is removed if the pullback itself trades below the breaker low (bullish) / above its
//   high (bearish) before price closes out of the zone again; otherwise it stays for reference.
// [MB] Mitigation blocks: bullish = low, high, higher low (zone at the high, chosen as for breakers); bearish =
//   high, low, lower high (zone at the low). Same rules as the breakers (close beyond the zone, swing beyond it,
//   pullback); an overlapping IFVG label turns bold with an up / down arrow; an overlapping unlabeled FVG gets
//   M↑ / M↓ (BM↑ / BM↓ when a breaker tagged it too). Own rectangle.

getstyle(s) =>
    switch s
        "solid" => line.style_solid
        "dashed" => line.style_dashed
        "dotted" => line.style_dotted

getsize(s) =>
    switch(s) 
        "auto" => size.auto
        "tiny" => size.tiny
        "small" => size.small
        "normal" => size.normal
        "large" => size.large
        "huge" => size.huge


gethalign(s) =>
    switch s
        "left" => text.align_left
        "right" => text.align_right
        => text.align_center

getvalign(s) =>
    switch s
        "top" => text.align_top
        "bottom" => text.align_bottom
        => text.align_center

    float ce // C.E.

    line loln = na // segments style: low line

    bool own = false // the box exists only because of the IFVG (the FVG itself is not drawn)
    bool shown = false // FVG is in the drawn arrays

    int mbD = 0 // [MB] direction of the mitigation block it was tagged at (1 bullish, -1 bearish), 0 none

// [OTEF] leg of a new FVG (candles [3] to [1]): origin = the lowest low (bullish) / highest high (bearish) from the
// last swing before the FVG, extreme = the high / low of the FVG candles
method oteInit(FVG f) =>
    if oteFvg
        int n = 3
        if f.islong and not na(otePLb) and otePLb <= bar_index - 3
            n := math.min(bar_index - otePLb, 4000)
        else if not f.islong and not na(otePHb) and otePHb <= bar_index - 3
            n := math.min(bar_index - otePHb, 4000)
        float lo = low[1]
        float hi = high[1]
        for k = 1 to n
            if f.islong
                lo := math.min(lo, low[k])
            else
                hi := math.max(hi, high[k])
        if f.islong
            f.oLo := lo
            f.oHi := math.max(high[1], high[2], high[3])
        else
            f.oHi := hi
            f.oLo := math.min(low[1], low[2], low[3])

// [OTEF] once per bar for a live FVG: true on the bar it retraces into the FVG while it matches the OTE band
method oteFvgStep(FVG f) =>
    bool hit = false
    if oteFvg and not f.ote and not na(f.oLo) and not na(f.oHi)
        float r = f.oHi - f.oLo
        if r > 0 and barstate.isconfirmed
            float bLo = f.islong ? f.oHi - oteMax * r : f.oLo + oteMin * r
            float bHi = f.islong ? f.oHi - oteMin * r : f.oLo + oteMax * r
            bool m = oteMode == "Overlap" ? f.hi >= bLo and f.lo <= bHi
                 : oteMode == "Fully inside" ? f.lo >= bLo and f.hi <= bHi
                 : f.ce >= bLo and f.ce <= bHi
            bool tap = f.islong ? low <= f.hi : high >= f.lo
            if m and tap
                f.ote := true
                hit := true
        // keep extending the leg AFTER evaluating this bar
        if f.islong
            f.oHi := math.max(f.oHi, high)
        else
            f.oLo := math.min(f.oLo, low)
    hit

method kill(FVG f) =>
    if not na(f)
        if not na(f.rect)
            f.rect.delete()
        if not na(f.celine)
            f.celine.delete()
        if not na(f.hiln)
            f.hiln.delete()
        if not na(f.loln)
            f.loln.delete()
        f.rect := na
        f.celine := na
        f.hiln := na
        f.loln := na
method alerts(FVG f) =>
    a = false
    if not na(f.hi) and bounces(f.hi)

        a := true
    else if not na(f.lo) and bounces(f.lo)

        a := true
    else if not na(f.ce) and bounces(f.ce)
        alert("bounce on CE", alert.freq_once_per_bar)
        a := true
    if a and debug
        line.new(bar_index,low,bar_index,high,extend = extend.both)
    a

isLongFVG(l1,h3) =>
    l1 > h3
longFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l1
    lo = h3
    blow1 = math.min(o1,c1)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi3 = math.max(o3,c3)
    // include suspension blocks
    fhi = hi
    flo = lo
    if bhi3 < blow2 and not poorLiq
        lo := bhi3
    if blow1 > bhi2 and not poorLiq
        hi := blow1
    [hi,lo, fhi, flo]

isShortFVG(h1,l3) =>
    h1 < l3
shortFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l3
    lo = h1
    blow3 = math.min(o3,c3)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi1 = math.max(o1,c1)
    // include suspension blocks
    fhi = hi
    flo = lo
    if blow3 > bhi2 and not poorLiq
        hi := blow3
    if bhi1 < blow2 and not poorLiq
        lo := bhi1
    [hi, lo, fhi, flo]

// IFVG boxes: the FVG box itself is turned into the IFVG box (no second box on the same place) ====
var array<FVG> keptIf = array.new<FVG>()   // FVGs whose box is currently shown as IFVG

// back to the plain FVG, as it was before becoming an IFVG
method restore(FVG f) =>
    f.kept := false
    f.cand := false
    if not na(f.rect)
        if f.own or stySeg
            f.rect.delete()
            f.rect := na
            f.own := false
        else
            f.rect.set_bgcolor(f.bgColor())
            f.rect.set_border_color(f.lnColor())
            f.rect.set_border_width(f.bw())
            f.rect.set_text(fvgLbl(f))
            f.rect.set_right(f.endsAt())
            f.rect.set_extend((f.isifp or f.isfp) ? extend.right : extend.none)
    true

// the (older) FVG that absorbed a duplicate is the one that is shown
method adopt(FVG fv, int dir) =>
    FVG f = na(fv.dup) ? fv : fv.dup
    color c = dir < 0 ? bearIFVGColor : bullIFVGColor
    if na(f.rect)
        f.rect := box.new(f.start, f.hi, time, f.lo, xloc = xloc.bar_time)
        f.own := not (f.shown or f.isfp or f.isifp)
    if f.isfp or f.isifp
        // also a 1PFVG: IFVG label, but 1PFVG colors and behaviour (extends right)
        color fc = f.isifp ? ifpcol : (f.iswfp ? wfpcol : fpcol)
        color lc = f.isifp ? ilnfpcol : (f.iswfp ? lnwfpcol : lnfpcol)
        f.rect.set_bgcolor(styNoFill ? color(na) : fc)
        f.rect.set_border_color(lc)
        f.rect.set_border_width(1)
        f.rect.set_extend(extend.right)
    else
        f.rect.set_bgcolor(color.new(c, ifvgTrans))
        f.rect.set_border_color(color.new(c, borderTrans))
        f.rect.set_border_width(2)
        f.rect.set_extend(extend.none)
    f.rect.set_right(time)
    // an unconfirmed candidate keeps its B / M / BM label (confirming it sets the IFVG label)
    f.rect.set_text(f.brk or f.mbD != 0 ? fvgLbl(f) : "")
    f.rect.set_text_color(ifvgTxtCol)
    f.rect.set_text_size(ifvgtsz)
    f.rect.set_text_halign(ifvgha)
    f.rect.set_text_valign(ifvgva)
    if not f.kept
        keptIf.push(f)
    f.kept := true
    f.cand := true
    f.kdir := dir
    f.rect

// the IFVG candidate is replaced: keep it if confirmed (and wanted), else restore the FVG
method release(FVG fv, bool confirmed, int ctime) =>
    FVG f = na(fv.dup) ? fv : fv.dup
    f.cand := false
    if f.kept
        if keepIFVG and confirmed
            if na(f.end)
                f.end := ctime
            if keptIf.size() > keepMax + 3
                FVG o = keptIf.shift()
                if o.cand
                    keptIf.push(o)
                else
                    o.restore()
        else
            f.restore()
    true

// extend the IFVG boxes; restore kept ones when invalidated (close back through them)
updKept() =>
    int n = keptIf.size()
    if n > 0
        for i = n - 1 to 0
            FVG f = keptIf.get(i)
            if na(f.rect) or not f.kept
                f.kept := false
                f.cand := false
                keptIf.remove(i)
            else if not f.cand and barstate.isconfirmed and (f.kdir < 0 ? close > f.hi : close < f.lo)
                f.restore()
                keptIf.remove(i)
            else
                f.rect.set_right(time)
    true

// widen g (older FVG) with f (new almost identical one): max hi / min lo, for both, and extend g
method mergeZone(FVG g, FVG f) =>
    if not g.kept
        float nh = math.max(g.hi, f.hi)
        float nl = math.min(g.lo, f.lo)
        g.hi := nh
        g.lo := nl
        g.ce := math.avg(nh, nl)
        f.hi := nh
        f.lo := nl
        f.ce := g.ce
        if not na(g.rect)
            g.rect.set_top(nh)
            g.rect.set_bottom(nl)
        if not na(g.hiln)
            g.hiln.set_y1(nh)
            g.hiln.set_y2(nh)
            g.loln.set_y1(nl)
            g.loln.set_y2(nl)
        if not na(g.celine)
            g.celine.set_y1(g.ce)
            g.celine.set_y2(g.ce)
        g.end := na
        g.ext := true
        bumpTags()
    true

insertSorted(array<FVG> a, FVG f) =>
    int pos = a.size()
    while pos > 0 and a.get(pos - 1).start > f.start
        pos -= 1
    f.shown := true
    a.insert(pos, f)
    if f.brkD != 0 or f.mbD != 0
        bumpTags()
    true

// older drawn FVG (any bias) that is almost exactly the same zone as f
findDup(FVG f) =>
    FVG r = na
    float tol = (f.hi - f.lo) * dupTol / 100.0
    if dupTol > 0
        // drawn FVGs, first presented ones and FVGs whose box is shown as IFVG (they may not be in the other lists)
        for k = 0 to 3
            array<FVG> arr = k == 0 ? lfvgs : (k == 1 ? sfvgs : (k == 2 ? fpfvgs : keptIf))
            int n = arr.size()
            if n > 0
                for i = n - 1 to 0
                    FVG g = arr.get(i)
                    if math.abs(g.hi - f.hi) <= tol and math.abs(g.lo - f.lo) <= tol and (na(r) or g.start > r.start)
                        r := g
                        break
    r

method route(FltState s, FVG f) =>
    FVG d = findDup(f)
    bool shown = false
    if not na(d)
        // extend the old FVG instead of adding the new one; the new FVG keeps feeding the IFVG logic
        f.dup := d
        d.mergeZone(f)
    else if fltOff
        shown := true
    else if s.post < flN
        shown := true
        s.post += 1
    if shown
        insertSorted(f.islong ? lfvgs : sfvgs, f)
    else if na(d)
        pendF.push(f)
        if pendF.size() > 60
            pendF.shift()
        if fltRel
            relF.push(f)
            if relF.size() > 300
                relF.shift()
    true

// show a pending FVG: if a drawn one is almost the same zone, merge into it instead of drawing a second box
reveal(FVG f) =>
    FVG d = f.kept ? na : findDup(f)
    if na(d)
        insertSorted(f.islong ? lfvgs : sfvgs, f)
    else
        f.dup := d
        if f.start < d.start and not d.kept
            // the revealed FVG is the older one: the merged box starts where it started
            d.start := f.start
            if not na(d.rect)
                d.rect.set_left(d.start)
            if not na(d.hiln)
                d.hiln.set_x1(d.start)
                d.loln.set_x1(d.start)
            if not na(d.celine)
                d.celine.set_x1(d.start)
        d.mergeZone(f)
    true

// a sweep happened: reveal the last FVG before it (and, optionally, the most significant one near the middle of the arm)
onSweep(int armStart) =>
    int n = pendF.size()
    if n > 0
        FVG lastF = pendF.get(n - 1)
        // [REL] the last FVG before the sweep waits to become an IFVG
        if fltRel
            lastF.swp := true
            lastF := na
        FVG midF = na
        if fltMid and n > 1 and not na(armStart) and armStart < bar_index
            int mb = math.round((armStart + bar_index) / 2.0)
            int w = math.max(1, math.round((bar_index - armStart) * flWin / 100.0))
            float best = 0.0
            for i = 0 to n - 2
                FVG g = pendF.get(i)
                float sz = g.hi - g.lo
                if g.sbar >= mb - w and g.sbar <= mb + w and sz > best
                    best := sz
                    midF := g
        if not na(lastF)
            reveal(lastF)
        if not na(midF)
            reveal(midF)
        pendF.clear()
    true

// [DE] Delivery Engine =========================================================
// displacement: candle range vs. its standard deviation (TFO criterion of the Delivery Engine)

bool deDispFvg = deRange[1] > deStd[1] and (close[1] > open[1] ? high[2] < low : low[2] > high)
barcolor(deDispBar and deDispFvg ? deDispCol : na, offset = -1)

bool deDispDn = nz(ta.barssince(deRange > deStdMss and close < open), 999) <= deMssWin

// Pivot sets: one for the chart timeframe and one for each of the next two higher timeframes.
// HTF pivots come from request.security (non-repainting: confirmed HTF bars only) and are
// stored with the CHART bar index they map to, so legs can be measured on chart candles.

tfName(string s) =>
    float m = str.tonumber(s)
    string r = s
    if not na(m)
        r := m >= 60 and m % 60 == 0 ? str.tostring(m / 60, "#") + "h" : str.tostring(m, "#") + "m"
    else if s == "1D"
        r := "D"
    r

type PivSet
    array<int> hb       // pivot-high chart bar index
    array<float> hp     // pivot-high price
    array<int> lb       // pivot-low chart bar index
    array<float> lp     // pivot-low price
    string tag = ""     // timeframe label
    bool htf = false    // higher timeframe set?

method addHigh(PivSet ps, int b, float p) =>
    ps.hb.push(b)
    ps.hp.push(p)
    if ps.hb.size() > 60
        ps.hb.shift()
        ps.hp.shift()

method addLow(PivSet ps, int b, float p) =>
    ps.lb.push(b)
    ps.lp.push(p)
    if ps.lb.size() > 60
        ps.lb.shift()
        ps.lp.shift()

method lastHigh(PivSet ps) =>
    ps.hp.size() > 0 ? ps.hp.last() : na

method lastLow(PivSet ps) =>
    ps.lp.size() > 0 ? ps.lp.last() : na

// [REL] small filled FVGs: once a pivot at its candles would be confirmed, hide it if there is none and it
// was not extended by a merged FVG (or became an IFVG box) meanwhile
hasPiv(array<int> bars, int b0, int b1) =>
    bool r = false
    int n = bars.size()
    if n > 0
        for k = n - 1 to 0
            int b = bars.get(k)
            if b < b0
                break
            if b <= b1
                r := true
                break
    r

relHide(FVG f) =>
    array<FVG> arr = f.islong ? lfvgs : sfvgs
    int n = arr.size()
    if n > 0
        for k = n - 1 to 0
            FVG g = arr.get(k)
            if g.start == f.start and g.hi == f.hi and g.lo == f.lo
                arr.remove(k)
                break
    f.shown := false
    f.kill()

if fltRel and relS.size() > 0
    for i = relS.size() - 1 to 0
        FVG f = relS.get(i)
        if bar_index >= math.max(f.dbar, f.sbar + 3 + pivR)
            relS.remove(i)
            bool keep = f.ext or f.kept or f.cand or not f.relSmall()
                 or hasPiv(pv0.hb, f.sbar - 1, f.sbar + 3) or hasPiv(pv0.lb, f.sbar - 1, f.sbar + 3)
            if not keep
                relHide(f)

// chart bar index of the chart candle that opens at (or right before) time t
barOfTime(int t) =>
    int r = na

    for k = 0 to mx

// Returns [legHigh, legLow] of the leg that precedes the FVG starting at bar sbar.
//   dir = -1 (bearish IFVG, from a bullish FVG): prior DOWN leg = last pivot high before sbar -> lowest low up to sbar
//   dir = +1 (bullish IFVG, from a bearish FVG): prior UP leg   = last pivot low before sbar  -> highest high up to sbar
priorLeg(int dir, int sbar, PivSet ps) =>
    float h = na
    float l = na
    if dir == -1
        int n = ps.hb.size()
        if n > 0
            for i = n - 1 to 0
                int hb = ps.hb.get(i)
                if hb < sbar and bar_index - hb < 4000
                    float lo = low[bar_index - hb]
                    for k = hb to sbar
                        lo := math.min(lo, low[bar_index - k])
                    h := ps.hp.get(i)
                    l := lo
                    break
    else
        int n = ps.lb.size()
        if n > 0
            for i = n - 1 to 0
                int lb = ps.lb.get(i)
                if lb < sbar and bar_index - lb < 4000
                    float hi = high[bar_index - lb]
                    for k = lb to sbar
                        hi := math.max(hi, high[bar_index - k])
                    l := ps.lp.get(i)
                    h := hi
                    break
    [h, l]

// [OTE] tracker =================================================
// One tracker follows one IFVG under watch, against the pivots of ONE timeframe (chart, HTF1 or HTF2).
//   dir = -1 : bearish IFVG (bullish FVG closed through to the downside).
//   dir = +1 : bullish IFVG (bearish FVG closed through to the upside).
// Two OTE checks, both done per timeframe:
//   (1) rejection: the confirming candle wicks into the 0.62-0.79 band of the PRIOR leg (before the FVG)
//   (2) retrace  : after the inversion, price retraces into the IFVG while the IFVG matches the OTE band of
//                  the leg swing -> lowest low (bearish) / swing -> highest high (bullish) since confirmation.
//                  On higher timeframes the swing origin is the latest HTF pivot (if beyond the retrace extreme).
type OteTrack
    int dir = 0
    float zlo = na      // IFVG low
    float zhi = na      // IFVG high
    float legHi = na
    float legLo = na
    float rext = na     // running retrace extreme since the FVG (high for dir -1, low for dir +1)
    bool armed = false  // IFVG confirmed by a close through it
    bool fired = false  // OTE signal already fired for this confirmation
    int ctime = na      // confirmation time
    int cbar = na       // confirmation bar index
    box band = na       // OTE band drawn (leg created after the inversion)
    float pHi = na      // prior leg high (leg before the FVG)
    float pLo = na      // prior leg low
    int sbar = na       // bar index where the FVG starts
    bool rejHit = false // confirming candle rejected the OTE of the prior leg
    box rej = na        // prior-leg OTE band drawn
    string tag = ""     // timeframe label
    int col = 0         // margin column of its timeframe (chart, then each higher timeframe)

// first bar of the OTE bands of t: one column per timeframe, so the bands and labels of several
// timeframes do not overlap
oteX0(OteTrack t) =>
    bar_index + oteOffset + t.col * (oteWidth + 1)

// new IFVG candidate under watch (not yet confirmed). origin = retrace extreme since the FVG
method oteStart(OteTrack t, int dir, float zlo, float zhi, float origin, int sbar, PivSet ps) =>
    box.delete(t.band)
    t.band := na
    box.delete(t.rej)
    t.rej := na
    t.rejHit := false
    t.sbar := sbar
    t.tag := ps.tag
    [pgh, pgl] = priorLeg(dir, sbar, ps)
    t.pHi := pgh
    t.pLo := pgl
    float o = origin
    if ps.htf
        float adj = dir == -1 ? ps.lastHigh() : ps.lastLow()
        if not na(adj)
            o := dir == -1 ? math.max(origin, adj) : math.min(origin, adj)
    t.dir := dir
    t.zlo := zlo
    t.zhi := zhi
    t.armed := false
    t.fired := false
    t.ctime := na
    t.rext := origin
    t.legHi := dir == -1 ? o : na
    t.legLo := dir == 1 ? o : na

// close through the FVG: the IFVG is confirmed, the leg starts being measured
method oteConfirm(OteTrack t) =>
    if t.dir != 0 and not t.armed
        // rejection of the OTE of the PRIOR leg by the confirming candle
        if useOTE and oteRej and not na(t.pHi) and not na(t.pLo) and t.pHi > t.pLo
            float pr = t.pHi - t.pLo
            float ext = na
            float rLo = na
            float rHi = na
            bool hit = false
            if t.dir == -1
                // prior down leg: OTE band is above the leg low; wick must reach up into it
                ext := oteRejAny ? math.max(t.rext, high) : high
                rLo := t.pLo + oteMin * pr
                rHi := t.pLo + oteMax * pr
                hit := ext >= rLo and ext <= (oteRejCap ? rHi : t.pHi) and close <= rHi
            else
                // prior up leg: OTE band is below the leg high; wick must reach down into it
                ext := oteRejAny ? math.min(t.rext, low) : low
                rLo := t.pHi - oteMax * pr
                rHi := t.pHi - oteMin * pr
                hit := ext <= rHi and ext >= (oteRejCap ? rLo : t.pLo) and close >= rLo
            if hit
                t.rejHit := true
                if showOTEband
                    t.rej := box.new(oteX0(t), rHi, oteX0(t) + oteWidth, rLo,
                        xloc = xloc.bar_index,
                        bgcolor = oteBandCol,
                        border_color = oteHighlight,
                        border_style = line.style_dashed,
                        border_width = 1,
                        text = t.tag + " OTE" + (t.dir == 1 ? " ↑" : " ↓"),
                        text_size = size.small,
                        text_color = oteTextCol,
                        text_halign = text.align_center)
        t.armed := true
        t.fired := t.rejHit // one OTE signal per IFVG
        t.ctime := time
        t.cbar := bar_index
        if t.dir == -1
            t.legHi := math.max(t.legHi, high)
            t.legLo := low
        else
            t.legLo := math.min(t.legLo, low)
            t.legHi := high

// IFVG invalidated (closed back through it)
method oteCancel(OteTrack t) =>
    box.delete(t.band)
    t.band := na
    box.delete(t.rej)
    t.rej := na
    t.rejHit := false
    t.armed := false
    if t.dir == -1
        t.legLo := na
    else
        t.legHi := na

// call once per bar. Returns the direction (-1 / 1) on the bar the OTE is matched, else 0
method oteStep(OteTrack t) =>
    int s = 0
    if useOTE and t.dir != 0
        if not t.armed
            // before confirmation keep extending the origin of the leg
            if t.dir == -1
                t.legHi := math.max(t.legHi, high)
                t.rext := math.max(t.rext, high)
            else
                t.legLo := math.min(t.legLo, low)
                t.rext := math.min(t.rext, low)
        else
            if not na(t.rej)
                t.rej.set_left(oteX0(t))
                t.rej.set_right(oteX0(t) + oteWidth)
            // rejection of the prior-leg OTE flagged on the confirming candle
            if t.rejHit
                s := t.dir
                t.rejHit := false
            float r = t.legHi - t.legLo
            if r > 0
                float bLo = na
                float bHi = na
                if t.dir == -1
                    bLo := t.legLo + oteMin * r
                    bHi := t.legLo + oteMax * r
                else
                    bLo := t.legHi - oteMax * r
                    bHi := t.legHi - oteMin * r
                float mid = math.avg(t.zlo, t.zhi)
                bool m = false
                if oteMode == "Overlap"
                    m := t.zhi >= bLo and t.zlo <= bHi
                else if oteMode == "Fully inside"
                    m := t.zlo >= bLo and t.zhi <= bHi
                else
                    m := mid >= bLo and mid <= bHi
                if showOTEband
                    color bg = m ? oteBandCol : color.new(color.gray, 92)
                    color bd = m ? oteHighlight : color.new(color.gray, 60)
                    if na(t.band)
                        t.band := box.new(oteX0(t), bHi, oteX0(t) + oteWidth, bLo,
                            xloc = xloc.bar_index,
                            bgcolor = bg,
                            border_color = bd,
                            border_style = line.style_dashed,
                            border_width = 1,
                            text = t.tag + (t.dir == 1 ? " ↑" : " ↓"),
                            text_size = size.tiny,
                            text_color = oteTextCol,
                            text_halign = text.align_left)
                    else
                        t.band.set_top(bHi)
                        t.band.set_bottom(bLo)
                        t.band.set_left(oteX0(t))
                        t.band.set_right(oteX0(t) + oteWidth)
                        t.band.set_bgcolor(bg)
                        t.band.set_border_color(bd)
                        t.band.set_text(t.tag + (t.dir == 1 ? " ↑" : " ↓"))
                // price retraces back into the IFVG while it matches the OTE band
                bool tap = t.dir == -1 ? high >= t.zlo : low <= t.zhi
                if m and tap and not t.fired and time != t.ctime and barstate.isconfirmed
                    t.fired := true
                    s := t.dir
            // keep extending the leg AFTER evaluating this bar
            if t.dir == -1
                t.legLo := math.min(t.legLo, low)
            else
                t.legHi := math.max(t.legHi, high)
    s

// A group runs the three trackers (chart, next TF, second TF) on the same IFVG.
// A confirmed OTE on ANY of them is enough to flag the IFVG as an OTE IFVG (once per IFVG).
type OteGroup
    OteTrack t0
    OteTrack t1
    OteTrack t2
    bool flagged = false
    string tags = ""    // timeframes that matched, e.g. "5m+15m"

method gStart(OteGroup g, int dir, float zlo, float zhi, float origin, int sbar) =>
    g.flagged := false
    g.tags := ""
    g.t0.col := 0
    g.t1.col := 1
    g.t2.col := hasH1 ? 2 : 1
    g.t0.oteStart(dir, zlo, zhi, origin, sbar, pv0)
    if hasH1
        g.t1.oteStart(dir, zlo, zhi, origin, sbar, pv1)
    if hasH2
        g.t2.oteStart(dir, zlo, zhi, origin, sbar, pv2)

method gConfirm(OteGroup g) =>
    g.t0.oteConfirm()
    g.t1.oteConfirm()
    g.t2.oteConfirm()

method gCancel(OteGroup g) =>
    g.flagged := false
    g.tags := ""
    g.t0.oteCancel()
    g.t1.oteCancel()
    g.t2.oteCancel()

method gStep(OteGroup g) =>
    int s0 = g.t0.oteStep()
    int s1 = g.t1.oteStep()
    int s2 = g.t2.oteStep()
    int s = 0
    if not g.flagged and (s0 != 0 or s1 != 0 or s2 != 0)
        g.flagged := true
        s := s0 != 0 ? s0 : (s1 != 0 ? s1 : s2)
        string tg = ""
        if s0 != 0
            tg += g.t0.tag
        if s1 != 0
            tg += (tg == "" ? "" : "+") + g.t1.tag
        if s2 != 0
            tg += (tg == "" ? "" : "+") + g.t2.tag
        g.tags := tg
    s

        sig := -1

// [DE] inversion rules: every close through the zone inverts it again, up to the max; then it is invalidated.
// Bull candidate (bearish FVG): inversions 1, 3, ... are bullish (close above), 2, 4, ... bearish (close below).
if doifvg and not ifvgswings and deInv and barstate.isconfirmed
    if not na(bullifvg) and bullInv >= 0
        bool brk = bullInv % 2 == 0 ? upBreak(bullifvg.hi) : dnBreak(bullifvg.lo)

            if bullInv >= deMaxInv

                bullconfirm := na
                gBull.gCancel() // [OTE]
                deKill(bullifvg)
                ifvgBox := na
            else
                bullInv += 1

                if bullInv == 1

                    gBull.gConfirm() // [OTE]
                else
                    bullconfirm := na
                    gBull.gCancel() // [OTE] only the first inversion
                    deRecolor(ifvgBox, d)

                ifvgBox.set_text(ifvgTxt(d, bullInv, false, "", bullAp, mbOf(bullifvg)))
                sig := d
    if not na(bearifvg) and bearInv >= 0

                gBear.gCancel() // [OTE]

                ifvgBoxb := na
            else

                    gBear.gConfirm() // [OTE]

                    gBear.gCancel() // [OTE] only the first inversion

// [BRK] Breakers =================================================
type Breaker
    int dir         // 1 bullish (long after the pullback), -1 bearish
    float hi        // zone
    float lo
    int t           // time of the first candle of the zone
    bool broken = false
    int bkBar = na  // bar of the close beyond the zone
    bool swung = false  // a swing beyond the zone formed after the break (needed for the pullback)
    bool tagged = false // first pullback with a matching FVG / IFVG seen
    box bx = na     // drawn at that pullback (when the rectangle is on)
    line ml = na    // mid line
    int t1 = na     // time of the last candle of the zone
    array<line> fl = na // flagged candles (wick and body lines)

var array<Breaker> brks = array.new<Breaker>()
float bPH = ta.pivothigh(high, brkLen, brkLen)
float bPL = ta.pivotlow(low, brkLen, brkLen)
var float bLastH = na   // last two pivot highs / lows (price, bar)
var int bLastHb = na
var float bPrevH = na
var int bPrevHb = na
var float bLastL = na
var int bLastLb = na
var float bPrevL = na
var int bPrevLb = na

// zone from candle k bars back: the up-close (dir 1) / down-close (dir -1) candle there or the nearest one
// before it (not before bar 'from'), plus the consecutive ones before it in "Whole sequence" mode
brkZone(int dir, int k0, int from) =>
    float zh = na
    float zl = na
    int zt = na
    int zt1 = na
    int kmax = math.min(bar_index - from, 300)
    int k = k0
    while k <= kmax and na(zt)
        if (close[k] - open[k]) * dir > 0
            zh := high[k]
            zl := low[k]
            zt := time[k]
            zt1 := time[k]
        k += 1
    if not na(zt) and brkMode == "Whole sequence"
        while k <= kmax and (close[k] - open[k]) * dir > 0
            zh := math.max(zh, high[k])
            zl := math.min(zl, low[k])
            zt := time[k]
            k += 1
    [zh, zl, zt, zt1]

brkAdd(int dir, float zh, float zl, int zt, int zt1) =>
    brks.push(Breaker.new(dir, zh, zl, zt, t1 = zt1))
    int cnt = 0
    for b in brks
        if b.dir == dir
            cnt += 1
    if cnt > brkKeep
        for i = 0 to brks.size() - 1
            if brks.get(i).dir == dir
                brks.remove(i)
                break
    true

// FVG shown as the box of a confirmed IFVG (close mode or swing mode), with its IFVG label
sameF(FVG f, FVG g) =>
    FVG h = na(g) ? g : (na(g.dup) ? g : g.dup)
    not na(h) and h.start == f.start and h.hi == f.hi and h.lo == f.lo and h.islong == f.islong

confIfvg(FVG f) =>
    (not na(bullconfirm) and sameF(f, bullifvg)) or (not na(bearconfirm) and sameF(f, bearifvg))
     or (not na(ifsconfirm) and sameF(f, swFvg))

// breaker b drawn up to this candle when FVG / IFVG f overlaps it on the pullback (once); ib: the IFVG box
// when f is the confirmed IFVG (its label turns bold), else na (a plain unlabeled FVG gets B↑ / B↓)
brkTag(FVG f, Breaker b, box ib) =>
    if not na(f) and f.hi >= b.lo and f.lo <= b.hi
        f.brk := true
        f.brkD := b.dir
        bumpTags()
        if not na(ib)
            ib.set_text_formatting(text.format_bold)
        else
            // the label direction is kept for when the box shows the plain FVG label (also after it was an
            // IFVG candidate); an unconfirmed IFVG candidate box shows it too
            f.lblD := b.dir
            if na(f.end) and (not f.kept or (f.cand and not confIfvg(f))) and not (f.aplus and lblAp) and not na(f.rect)
                f.rect.set_text(fvgLbl(f))
                f.rect.set_text_size(ifvgtsz)
                f.rect.set_text_halign(ifvgha)
                f.rect.set_text_valign(ifvgva)
        float mid = (b.hi + b.lo) / 2
        if not b.tagged
            b.tagged := true
            if brkFlag
                b.fl := array.new<line>()
                int kk = 0
                while kk <= 300 and time[kk] >= b.t
                    if time[kk] <= b.t1
                        b.fl.push(line.new(time[kk], high[kk], time[kk], low[kk], xloc = xloc.bar_time,
                             color = brkCol, width = 1))
                        b.fl.push(line.new(time[kk], open[kk], time[kk], close[kk], xloc = xloc.bar_time,
                             color = brkCol, width = 5))
                    kk += 1
            if brkBox or brkShd
                b.bx := box.new(b.t, b.hi, time, b.lo, xloc = xloc.bar_time,
                    border_color = brkShd ? color.new(color.gray, 100) : brkCol, border_style = brkSty,
                    border_width = brkShd ? 0 : brkW, bgcolor = color.new(brkShd ? color.gray : brkCol, brkShd ? 80 : 100))
                b.ml := line.new(b.t, mid, time, mid, xloc = xloc.bar_time, color = brkCol, style = brkSty,
                    width = brkW)
    true

if brkOn and barstate.isconfirmed
    if not na(bPL)
        bPrevL := bLastL
        bPrevLb := bLastLb
        bLastL := bPL
        bLastLb := bar_index - brkLen
        // bullish: low, high, lower low (the high between the two lows)
        if not na(bPrevL) and bLastL < bPrevL and not na(bLastHb) and bLastHb > bPrevLb and bLastHb < bLastLb
            [zh, zl, zt, zt1] = brkZone(1, bar_index - bLastHb, bPrevLb)
            if not na(zt)
                brkAdd(1, zh, zl, zt, zt1)
    if not na(bPH)
        bPrevH := bLastH
        bPrevHb := bLastHb
        bLastH := bPH
        bLastHb := bar_index - brkLen
        // bearish: high, low, higher high (the low between the two highs)
        if not na(bPrevH) and bLastH > bPrevH and not na(bLastLb) and bLastLb > bPrevHb and bLastLb < bLastHb
            [zh, zl, zt, zt1] = brkZone(-1, bar_index - bLastLb, bPrevHb)
            if not na(zt)
                brkAdd(-1, zh, zl, zt, zt1)
    if brks.size() > 0
        for i = brks.size() - 1 to 0
            Breaker b = brks.get(i)
            if b.tagged
                // drawn, pullback still going on: removed when it trades beyond the far side; once price closes
                // out of the zone in the breaker direction the rectangle stays for reference
                if (b.dir == 1 and low < b.lo) or (b.dir == -1 and high > b.hi)
                    if not na(b.bx)
                        b.bx.delete()
                        b.ml.delete()
                    if not na(b.fl)
                        for ln in b.fl
                            ln.delete()
                    brks.remove(i)
                else if (b.dir == 1 and close > b.hi) or (b.dir == -1 and close < b.lo)
                    brks.remove(i)
            else if not b.broken
                if (b.dir == 1 and close > b.hi) or (b.dir == -1 and close < b.lo)
                    b.broken := true
                    b.bkBar := bar_index
            else if not b.swung
                // after the break: wait for a swing high above the zone (bullish) / low below it (bearish)
                if b.dir == 1 and not na(bPH) and bar_index - brkLen >= b.bkBar and bPH > b.hi
                    b.swung := true
                else if b.dir == -1 and not na(bPL) and bar_index - brkLen >= b.bkBar and bPL < b.lo
                    b.swung := true
                else if (b.dir == 1 and close < b.lo) or (b.dir == -1 and close > b.hi)
                    brks.remove(i)  // closed through the other side before the swing: invalidated
            else if (b.dir == 1 and close < b.lo) or (b.dir == -1 and close > b.hi)
                brks.remove(i)  // closed through the other side: invalidated
            else if (b.dir == 1 and low <= b.hi) or (b.dir == -1 and high >= b.lo)
                // pullback into the zone: tag the FVGs / IFVGs of the trade direction that overlap it
                array<FVG> fs0 = b.dir == 1 ? lfvgs : sfvgs
                for f in fs0
                    if na(f.end)
                        brkTag(f, b, na)
                if b.dir == 1
                    if not ifvgswings and not na(bullconfirm)
                        brkTag(bullifvg, b, ifvgBox)
                    if ifvgswings and ifsdir == 1 and not na(ifsconfirm)
                        brkTag(swFvg, b, ifvgBox)
                else
                    if not ifvgswings and not na(bearconfirm)
                        brkTag(bearifvg, b, ifvgBoxb)
                    if ifvgswings and ifsdir == -1 and not na(ifsconfirm)
                        brkTag(swFvg, b, ifvgBox)
                if b.tagged and ((b.dir == 1 and low < b.lo) or (b.dir == -1 and high > b.hi))
                    if not na(b.bx)
                        b.bx.delete()
                        b.ml.delete()
                    if not na(b.fl)
                        for ln in b.fl
                            ln.delete()
                    brks.remove(i)

// [MB] a confirmed IFVG overlapping an FVG of its direction that a breaker / mitigation block tagged (live, or
// mitigated after the IFVG's FVG started) gets the same tag: bold label, plus the arrow for a mitigation block
ifvgInherit(FVG iv, box ib, int which, int dir) =>
    // [PERF] the tags of the FVGs did not change since the last scan for this IFVG: nothing new to inherit
    if not na(iv) and not na(ib) and (na(iv.inhSeq) or iv.inhSeq != tagSeq.get(0))
        iv.inhSeq := tagSeq.get(0)
        array<FVG> fs = dir == 1 ? lfvgs : sfvgs
        bool gb = false
        bool gm = false
        for f in fs
            if (na(f.end) or f.end >= iv.start) and f.hi >= iv.lo and f.lo <= iv.hi
                if f.brkD == dir and brkOn
                    gb := true
                if f.mbD == dir and mbOn
                    gm := true
        if gb and not iv.brk
            iv.brk := true
            ib.set_text_formatting(text.format_bold)
        if gm and iv.mbD != dir
            iv.mbD := dir
            ib.set_text(mbTxt(which, dir))
            ib.set_text_formatting(text.format_bold)
    true

if (brkOn or mbOn) and barstate.isconfirmed
    if not ifvgswings
        if not na(bullconfirm)
            ifvgInherit(bullifvg, ifvgBox, 1, 1)
        if not na(bearconfirm)
            ifvgInherit(bearifvg, ifvgBoxb, -1, -1)
    else if not na(ifsconfirm)
        ifvgInherit(swFvg, ifvgBox, 2, ifsdir)

tfstr() =>
    string tf = timeframe.period
    if timeframe.isseconds
        str.replace(tf, "S", "s")
    else if timeframe.isminutes
        if tf == "60"
            "1h"
        else if tf == "120"
            "2h"
        else if tf == "180"
            "3h"
        else if tf == "240"
            "4h"
        else
            tf + "m"
    else if timeframe.isdaily
        // If it's exactly 1 day ("1D"), UI just shows "D"
        tf == "1D" ? "D" : str.replace(tf, "D", "D") 
    else if timeframe.isweekly
        tf == "1W" ? "W" : str.replace(tf, "W", "W")
    else if timeframe.ismonthly
        tf == "1M" ? "M" : str.replace(tf, "M", "M")
    else
        tf


if doifvgalert and sig != 0 and not (useOTE and oteOnly) // [OTE] plain IFVG alerts can be suppressed
    if sig > 0
        alert("LONG IFVG "+tfstr(), alert.freq_once_per_bar)
    else
        alert("SHORT IFVG "+tfstr(), alert.freq_once_per_bar)
