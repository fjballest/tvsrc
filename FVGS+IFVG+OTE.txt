bounces(p) =>
    (math.abs(p-low) < 2 and close > low+10) or (math.abs(p-high)<2 and close < high-10)

// Wicks ===================================================
wdx() =>
    r = wickdx
    if r == 0 and tfsecs/60 < 15
        r := 4
    else if r == 0
        r := 1
    r
var wicks = array.new<line>()
if dowick
    bodyh = math.max(open[1],close[1])
    bodyl = math.min(open[1],close[1])
    wh = high[1]-bodyh
    wl = bodyl-low[1]
    
    if wh > minwick
        ce = math.avg(high[1],bodyh)
        ln = line.new(bar_index-1, ce, bar_index-1+wdx(), ce, style = wicksty, color = hideReg ? color.new(wickcol, 100) : wickcol)
        wicks.push(ln)
    if wl > minwick
        ce = math.avg(low[1],bodyl)
        ln = line.new(bar_index-1, ce, bar_index-1+wdx(), ce, style = wicksty, color = hideReg ? color.new(wickcol, 100) : wickcol)
        wicks.push(ln)
    if wicks.size() > nwicks
        w = wicks.shift()
        w.delete()
    if dowickalert and barstate.isconfirmed
        for w in wicks
            p = w.get_y1()
            if bounces(p)
                if debug
                    line.new(bar_index,low,bar_index,high,extend = extend.both)
                alert("bounce on wick C.E.", alert.freq_once_per_bar)

    hs = str.substring(s, 0, 2)

// FVG data type, one per handled fvg.
type FVG
    float hi // max value
    float lo // min value
    float ce // C.E.
    bool islong // bullish?
    int start  // time for start
    int end = na// time when mitigated
    box rect = na // box drawn
    line celine = na // CE line
    bool isfp = false // is first presented FVG
    bool iswfp = false // is week first presented FVG
    bool isifp = false // is first presented FVG inverse
    int sbar = na // [OTE] bar index where the FVG starts
    line hiln = na // segments style: high line
    line loln = na // segments style: low line
    bool kept = false // kept confirmed IFVG (box shown as IFVG)
    int kdir = 0 // -1 bearish IFVG, 1 bullish IFVG
    FVG dup = na // older almost identical FVG that absorbed this one
    bool cand = false // current IFVG candidate (its box is shown as IFVG)
    bool own = false // the box exists only because of the IFVG (the FVG itself is not drawn)
    bool shown = false // FVG is in the drawn arrays
    bool aplus = false // [DE] A+ FVG (born from an A+ MSS)
    bool disp = false // [DE] middle candle is a displacement candle


// [DE] border width: thicker for flagged displacement FVGs
method bw(FVG f) =>
    deFlag and f.disp ? deFlagW : 1

method kill(FVG f) =>
    if not na(f)
        if not na(f.rect)
            f.rect.delete()
        if not na(f.celine)
            f.celine.delete()
        if not na(f.hiln)
            f.hiln.delete()
        if not na(f.loln)
            f.loln.delete()
        f.rect := na
        f.celine := na
        f.hiln := na
        f.loln := na
method alerts(FVG f) =>
    a = false
    if not na(f.hi) and bounces(f.hi)
        alert("bounce on FVG hi", alert.freq_once_per_bar)
        a := true
    else if not na(f.lo) and bounces(f.lo)
        alert("bounce on fvg lo", alert.freq_once_per_bar)
        a := true
    else if not na(f.ce) and bounces(f.ce)
        alert("bounce on CE", alert.freq_once_per_bar)
        a := true
    if a and debug
        line.new(bar_index,low,bar_index,high,extend = extend.both)
    a

clearlastfp(array<FVG> fpfvgs) =>
    while fpfvgs.size() > nfp
        f = fpfvgs.shift()
        if not na(f) and f.isfp
            if not na(f.rect)
                f.rect.set_extend(extend.none)
            if not na(f.celine)
                f.celine.set_extend(extend.none)
            if not na(f.hiln)
                f.hiln.set_extend(extend.none)
            if not na(f.loln)
                f.loln.set_extend(extend.none)
            f.isfp := false
            f.kill()

isLongFVG(l1,h3) =>
    l1 > h3
longFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l1
    lo = h3
    blow1 = math.min(o1,c1)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi3 = math.max(o3,c3)
    // include suspension blocks
    fhi = hi
    flo = lo
    if bhi3 < blow2
        lo := bhi3
    if blow1 > bhi2
        hi := blow1
    [hi,lo, fhi, flo]

isShortFVG(h1,l3) =>
    h1 < l3
shortFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l3
    lo = h1
    blow3 = math.min(o3,c3)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi1 = math.max(o1,c1)
    // include suspension blocks
    fhi = hi
    flo = lo
    if blow3 > bhi2
        hi := blow3
    if bhi1 < blow2
        lo := bhi1
    [hi, lo, fhi, flo]

isFPFVG() =>
    r = dofp
    if r
        h = hour(time,tz)
        m = minute(time,tz)
        nmin = 2
        if timeframe.isminutes
            nmin := tfsecs/60 * 2
        r := h == fph and m > fpm+nmin and timeframe.isminutes and tfsecs/60 <=3
    r
if bar_index == 0
    log.info("fpfvg h {0} fpfvgm {1}", fph, fpm)

    f.islong ? (c < f.lo or o < f.lo) : (c>f.hi or o > f.hi)

    if f.isifp
        c := ifpcol
    bool hid = hideReg and (ifvgOnly or (not f.isfp and not f.isifp))
    styNoFill ? color(na) : (hid ? color.new(c, 100) : c)

    if f.isifp

// IFVG boxes: the FVG box itself is turned into the IFVG box (no second box on the same place) ====
var array<FVG> keptIf = array.new<FVG>()   // FVGs whose box is currently shown as IFVG

// back to the plain FVG, as it was before becoming an IFVG
method restore(FVG f) =>
    f.kept := false
    f.cand := false
    if not na(f.rect)
        if f.own or stySeg
            f.rect.delete()
            f.rect := na
            f.own := false
        else
            f.rect.set_bgcolor(f.bgColor())
            f.rect.set_border_color(f.lnColor())
            f.rect.set_border_width(f.bw())
            f.rect.set_text(f.aplus and lblAp and not hideReg ? "A+" : "")
            f.rect.set_right(f.endsAt())
            f.rect.set_extend((f.isifp or f.isfp) ? extend.right : extend.none)
    true

// the (older) FVG that absorbed a duplicate is the one that is shown
method adopt(FVG fv, int dir) =>
    FVG f = na(fv.dup) ? fv : fv.dup
    color c = dir < 0 ? bearIFVGColor : bullIFVGColor
    if na(f.rect)
        f.rect := box.new(f.start, f.hi, time, f.lo, xloc = xloc.bar_time)
        f.own := not (f.shown or f.isfp or f.isifp)
    if f.isfp or f.isifp
        // also a 1PFVG: IFVG label, but 1PFVG colors and behaviour (extends right)
        color fc = f.isifp ? ifpcol : (f.iswfp ? wfpcol : fpcol)
        color lc = f.isifp ? ilnfpcol : (f.iswfp ? lnwfpcol : lnfpcol)
        f.rect.set_bgcolor(styNoFill ? color(na) : fc)
        f.rect.set_border_color(lc)
        f.rect.set_border_width(1)
        f.rect.set_extend(extend.right)
    else
        f.rect.set_bgcolor(color.new(c, ifvgTrans))
        f.rect.set_border_color(color.new(c, borderTrans))
        f.rect.set_border_width(2)
        f.rect.set_extend(extend.none)
    f.rect.set_right(time)
    f.rect.set_text("")
    f.rect.set_text_color(ifvgTxtCol)
    f.rect.set_text_size(ifvgtsz)
    f.rect.set_text_halign(ifvgha)
    f.rect.set_text_valign(ifvgva)
    if not f.kept
        keptIf.push(f)
    f.kept := true
    f.cand := true
    f.kdir := dir
    f.rect

// the IFVG candidate is replaced: keep it if confirmed (and wanted), else restore the FVG
method release(FVG fv, bool confirmed, int ctime) =>
    FVG f = na(fv.dup) ? fv : fv.dup
    f.cand := false
    if f.kept
        if keepIFVG and confirmed
            if na(f.end)
                f.end := ctime
            if keptIf.size() > keepMax + 3
                FVG o = keptIf.shift()
                if o.cand
                    keptIf.push(o)
                else
                    o.restore()
        else
            f.restore()
    true

// extend the IFVG boxes; restore kept ones when invalidated (close back through them)
updKept() =>
    int n = keptIf.size()
    if n > 0
        for i = n - 1 to 0
            FVG f = keptIf.get(i)
            if na(f.rect) or not f.kept
                f.kept := false
                f.cand := false
                keptIf.remove(i)
            else if not f.cand and barstate.isconfirmed and (f.kdir < 0 ? close > f.hi : close < f.lo)
                f.restore()
                keptIf.remove(i)
            else
                f.rect.set_right(time)
    true

// widen g (older FVG) with f (new almost identical one): max hi / min lo, for both, and extend g
method mergeZone(FVG g, FVG f) =>
    if not g.kept
        float nh = math.max(g.hi, f.hi)
        float nl = math.min(g.lo, f.lo)
        g.hi := nh
        g.lo := nl
        g.ce := math.avg(nh, nl)
        f.hi := nh
        f.lo := nl
        f.ce := g.ce
        if not na(g.rect)
            g.rect.set_top(nh)
            g.rect.set_bottom(nl)
        if not na(g.hiln)
            g.hiln.set_y1(nh)
            g.hiln.set_y2(nh)
            g.loln.set_y1(nl)
            g.loln.set_y2(nl)
        if not na(g.celine)
            g.celine.set_y1(g.ce)
            g.celine.set_y2(g.ce)
        g.end := na
    true

// FVG filtering around liquidity sweeps ==============================================
type FltState
    int post = 1000000      // FVGs shown since the last sweep (starts high: nothing shown before the first sweep)
    float stH = na          // last confirmed short-term high (not swept yet)
    float stL = na
    int stHbar = na
    int stLbar = na

var FltState fs = FltState.new()
var array<FVG> pendF = array.new<FVG>()  // FVGs since the last sweep that are not shown (yet)

insertSorted(array<FVG> a, FVG f) =>
    int pos = a.size()
    while pos > 0 and a.get(pos - 1).start > f.start
        pos -= 1
    f.shown := true
    a.insert(pos, f)
    true

// older drawn FVG (any bias) that is almost exactly the same zone as f
findDup(FVG f) =>
    FVG r = na
    float tol = (f.hi - f.lo) * dupTol / 100.0
    if dupTol > 0
        // drawn FVGs, first presented ones and FVGs whose box is shown as IFVG (they may not be in the other lists)
        for k = 0 to 3
            array<FVG> arr = k == 0 ? lfvgs : (k == 1 ? sfvgs : (k == 2 ? fpfvgs : keptIf))
            int n = arr.size()
            if n > 0
                for i = n - 1 to 0
                    FVG g = arr.get(i)
                    if math.abs(g.hi - f.hi) <= tol and math.abs(g.lo - f.lo) <= tol and (na(r) or g.start > r.start)
                        r := g
                        break
    r

method route(FltState s, FVG f) =>
    FVG d = findDup(f)
    bool shown = false
    if not na(d)
        // extend the old FVG instead of adding the new one; the new FVG keeps feeding the IFVG logic
        f.dup := d
        d.mergeZone(f)
    else if fltOff
        shown := true
    else if s.post < flN
        shown := true
        s.post += 1
    if shown
        insertSorted(f.islong ? lfvgs : sfvgs, f)
    else if na(d)
        pendF.push(f)
        if pendF.size() > 60
            pendF.shift()
    true

// show a pending FVG: if a drawn one is almost the same zone, merge into it instead of drawing a second box
reveal(FVG f) =>
    FVG d = f.kept ? na : findDup(f)
    if na(d)
        insertSorted(f.islong ? lfvgs : sfvgs, f)
    else
        f.dup := d
        if f.start < d.start and not d.kept
            // the revealed FVG is the older one: the merged box starts where it started
            d.start := f.start
            if not na(d.rect)
                d.rect.set_left(d.start)
            if not na(d.hiln)
                d.hiln.set_x1(d.start)
                d.loln.set_x1(d.start)
            if not na(d.celine)
                d.celine.set_x1(d.start)
        d.mergeZone(f)
    true

// a sweep happened: reveal the last FVG before it (and, optionally, the most significant one near the middle of the arm)
onSweep(int armStart) =>
    int n = pendF.size()
    if n > 0
        FVG lastF = pendF.get(n - 1)
        FVG midF = na
        if fltMid and n > 1 and not na(armStart) and armStart < bar_index
            int mb = math.round((armStart + bar_index) / 2.0)
            int w = math.max(1, math.round((bar_index - armStart) * flWin / 100.0))
            float best = 0.0
            for i = 0 to n - 2
                FVG g = pendF.get(i)
                float sz = g.hi - g.lo
                if g.sbar >= mb - w and g.sbar <= mb + w and sz > best
                    best := sz
                    midF := g
        reveal(lastF)
        if not na(midF)
            reveal(midF)
        pendF.clear()
    true

// [DE] Delivery Engine =========================================================
// displacement: candle range vs. its standard deviation (TFO criterion of the Delivery Engine)
float deRange = deDispType == "Open to Close" ? math.abs(open - close) : high - low
float deStd = ta.stdev(deRange, deStdLen) * deStdX
bool deDispOK = deRange[2] > deStd[2]   // middle candle of the FVG detected on this bar
// the middle candle of a displacement FVG (detected one candle later, hence the offset)
bool deDispFvg = deRange[1] > deStd[1] and (close[1] > open[1] ? high[2] < low : low[2] > high)
barcolor(deDispBar and deDispFvg ? deDispCol : na, offset = -1)
float deStdMss = ta.stdev(deRange, deStdLen) * deMssX
bool deDispUp = nz(ta.barssince(deRange > deStdMss and close > open), 999) <= deMssWin
bool deDispDn = nz(ta.barssince(deRange > deStdMss and close < open), 999) <= deMssWin

// HTF structure: 1 after a close above the last swing high, -1 after a close below the last swing low
deStruct(int len) =>
    var float sh = na
    var float sl = na
    var int d = 0
    float ph = ta.pivothigh(high, len, len)
    float pl = ta.pivotlow(low, len, len)
    if not na(sh) and close > sh
        d := 1
        sh := na
    if not na(sl) and close < sl
        d := -1
        sl := na
    if not na(ph)
        sh := ph
    if not na(pl)
        sl := pl
    d

// [OTE] prior leg detection =====================================
// Pivot sets: one for the chart timeframe and one for each of the next two higher timeframes.
// HTF pivots come from request.security (non-repainting: confirmed HTF bars only) and are
// stored with the CHART bar index they map to, so legs can be measured on chart candles.

tfName(string s) =>
    float m = str.tonumber(s)
    string r = s
    if not na(m)
        r := m >= 60 and m % 60 == 0 ? str.tostring(m / 60, "#") + "h" : str.tostring(m, "#") + "m"
    else if s == "1D"
        r := "D"
    r

type PivSet
    array<int> hb       // pivot-high chart bar index
    array<float> hp     // pivot-high price
    array<int> lb       // pivot-low chart bar index
    array<float> lp     // pivot-low price
    string tag = ""     // timeframe label
    bool htf = false    // higher timeframe set?

method addHigh(PivSet ps, int b, float p) =>
    ps.hb.push(b)
    ps.hp.push(p)
    if ps.hb.size() > 60
        ps.hb.shift()
        ps.hp.shift()

method addLow(PivSet ps, int b, float p) =>
    ps.lb.push(b)
    ps.lp.push(p)
    if ps.lb.size() > 60
        ps.lb.shift()
        ps.lp.shift()

method lastHigh(PivSet ps) =>
    ps.hp.size() > 0 ? ps.hp.last() : na

method lastLow(PivSet ps) =>
    ps.lp.size() > 0 ? ps.lp.last() : na

// Returns [legHigh, legLow] of the leg that precedes the FVG starting at bar sbar.
//   dir = -1 (bearish IFVG, from a bullish FVG): prior DOWN leg = last pivot high before sbar -> lowest low up to sbar
//   dir = +1 (bullish IFVG, from a bearish FVG): prior UP leg   = last pivot low before sbar  -> highest high up to sbar
priorLeg(int dir, int sbar, PivSet ps) =>
    float h = na
    float l = na
    if dir == -1
        int n = ps.hb.size()
        if n > 0
            for i = n - 1 to 0
                int hb = ps.hb.get(i)
                if hb < sbar and bar_index - hb < 4000
                    float lo = low[bar_index - hb]
                    for k = hb to sbar
                        lo := math.min(lo, low[bar_index - k])
                    h := ps.hp.get(i)
                    l := lo
                    break
    else
        int n = ps.lb.size()
        if n > 0
            for i = n - 1 to 0
                int lb = ps.lb.get(i)
                if lb < sbar and bar_index - lb < 4000
                    float hi = high[bar_index - lb]
                    for k = lb to sbar
                        hi := math.max(hi, high[bar_index - k])
                    l := ps.lp.get(i)
                    h := hi
                    break
    [h, l]

// [OTE] tracker =================================================
// One tracker follows one IFVG under watch, against the pivots of ONE timeframe (chart, HTF1 or HTF2).
//   dir = -1 : bearish IFVG (bullish FVG closed through to the downside).
//   dir = +1 : bullish IFVG (bearish FVG closed through to the upside).
// Two OTE checks, both done per timeframe:
//   (1) rejection: the confirming candle wicks into the 0.62-0.79 band of the PRIOR leg (before the FVG)
//   (2) retrace  : after the inversion, price retraces into the IFVG while the IFVG matches the OTE band of
//                  the leg swing -> lowest low (bearish) / swing -> highest high (bullish) since confirmation.
//                  On higher timeframes the swing origin is the latest HTF pivot (if beyond the retrace extreme).
type OteTrack
    int dir = 0
    float zlo = na      // IFVG low
    float zhi = na      // IFVG high
    float legHi = na
    float legLo = na
    float rext = na     // running retrace extreme since the FVG (high for dir -1, low for dir +1)
    bool armed = false  // IFVG confirmed by a close through it
    bool fired = false  // OTE signal already fired for this confirmation
    int ctime = na      // confirmation time
    int cbar = na       // confirmation bar index
    box band = na       // OTE band drawn (leg created after the inversion)
    float pHi = na      // prior leg high (leg before the FVG)
    float pLo = na      // prior leg low
    int sbar = na       // bar index where the FVG starts
    bool rejHit = false // confirming candle rejected the OTE of the prior leg
    box rej = na        // prior-leg OTE band drawn
    string tag = ""     // timeframe label

// new IFVG candidate under watch (not yet confirmed). origin = retrace extreme since the FVG
method oteStart(OteTrack t, int dir, float zlo, float zhi, float origin, int sbar, PivSet ps) =>
    box.delete(t.band)
    t.band := na
    box.delete(t.rej)
    t.rej := na
    t.rejHit := false
    t.sbar := sbar
    t.tag := ps.tag
    [pgh, pgl] = priorLeg(dir, sbar, ps)
    t.pHi := pgh
    t.pLo := pgl
    float o = origin
    if ps.htf
        float adj = dir == -1 ? ps.lastHigh() : ps.lastLow()
        if not na(adj)
            o := dir == -1 ? math.max(origin, adj) : math.min(origin, adj)
    t.dir := dir
    t.zlo := zlo
    t.zhi := zhi
    t.armed := false
    t.fired := false
    t.ctime := na
    t.rext := origin
    t.legHi := dir == -1 ? o : na
    t.legLo := dir == 1 ? o : na

// close through the FVG: the IFVG is confirmed, the leg starts being measured
method oteConfirm(OteTrack t) =>
    if t.dir != 0 and not t.armed
        // rejection of the OTE of the PRIOR leg by the confirming candle
        if useOTE and oteRej and not na(t.pHi) and not na(t.pLo) and t.pHi > t.pLo
            float pr = t.pHi - t.pLo
            float ext = na
            float rLo = na
            float rHi = na
            bool hit = false
            if t.dir == -1
                // prior down leg: OTE band is above the leg low; wick must reach up into it
                ext := oteRejAny ? math.max(t.rext, high) : high
                rLo := t.pLo + oteMin * pr
                rHi := t.pLo + oteMax * pr
                hit := ext >= rLo and ext <= (oteRejCap ? rHi : t.pHi) and close <= rHi
            else
                // prior up leg: OTE band is below the leg high; wick must reach down into it
                ext := oteRejAny ? math.min(t.rext, low) : low
                rLo := t.pHi - oteMax * pr
                rHi := t.pHi - oteMin * pr
                hit := ext <= rHi and ext >= (oteRejCap ? rLo : t.pLo) and close >= rLo
            if hit
                t.rejHit := true
                if showOTEband
                    t.rej := box.new(bar_index + oteOffset, rHi, bar_index + oteOffset + oteWidth, rLo,
                        xloc = xloc.bar_index,
                        bgcolor = oteBandCol,
                        border_color = oteHighlight,
                        border_style = line.style_dashed,
                        border_width = 1,
                        text = t.tag + " OTE",
                        text_size = size.small,
                        text_color = oteTextCol,
                        text_halign = text.align_center)
        t.armed := true
        t.fired := t.rejHit // one OTE signal per IFVG
        t.ctime := time
        t.cbar := bar_index
        if t.dir == -1
            t.legHi := math.max(t.legHi, high)
            t.legLo := low
        else
            t.legLo := math.min(t.legLo, low)
            t.legHi := high

// IFVG invalidated (closed back through it)
method oteCancel(OteTrack t) =>
    box.delete(t.band)
    t.band := na
    box.delete(t.rej)
    t.rej := na
    t.rejHit := false
    t.armed := false
    if t.dir == -1
        t.legLo := na
    else
        t.legHi := na

// call once per bar. Returns the direction (-1 / 1) on the bar the OTE is matched, else 0
method oteStep(OteTrack t) =>
    int s = 0
    if useOTE and t.dir != 0
        if not t.armed
            // before confirmation keep extending the origin of the leg
            if t.dir == -1
                t.legHi := math.max(t.legHi, high)
                t.rext := math.max(t.rext, high)
            else
                t.legLo := math.min(t.legLo, low)
                t.rext := math.min(t.rext, low)
        else
            if not na(t.rej)
                t.rej.set_left(bar_index + oteOffset)
                t.rej.set_right(bar_index + oteOffset + oteWidth)
            // rejection of the prior-leg OTE flagged on the confirming candle
            if t.rejHit
                s := t.dir
                t.rejHit := false
            float r = t.legHi - t.legLo
            if r > 0
                float bLo = na
                float bHi = na
                if t.dir == -1
                    bLo := t.legLo + oteMin * r
                    bHi := t.legLo + oteMax * r
                else
                    bLo := t.legHi - oteMax * r
                    bHi := t.legHi - oteMin * r
                float mid = math.avg(t.zlo, t.zhi)
                bool m = false
                if oteMode == "Overlap"
                    m := t.zhi >= bLo and t.zlo <= bHi
                else if oteMode == "Fully inside"
                    m := t.zlo >= bLo and t.zhi <= bHi
                else
                    m := mid >= bLo and mid <= bHi
                if showOTEband
                    color bg = m ? oteBandCol : color.new(color.gray, 92)
                    color bd = m ? oteHighlight : color.new(color.gray, 60)
                    if na(t.band)
                        t.band := box.new(bar_index + oteOffset, bHi, bar_index + oteOffset + oteWidth, bLo,
                            xloc = xloc.bar_index,
                            bgcolor = bg,
                            border_color = bd,
                            border_style = line.style_dashed,
                            border_width = 1,
                            text = t.tag,
                            text_size = size.tiny,
                            text_color = oteTextCol,
                            text_halign = text.align_left)
                    else
                        t.band.set_top(bHi)
                        t.band.set_bottom(bLo)
                        t.band.set_left(bar_index + oteOffset)
                        t.band.set_right(bar_index + oteOffset + oteWidth)
                        t.band.set_bgcolor(bg)
                        t.band.set_border_color(bd)
                // price retraces back into the IFVG while it matches the OTE band
                bool tap = t.dir == -1 ? high >= t.zlo : low <= t.zhi
                if m and tap and not t.fired and time != t.ctime and barstate.isconfirmed
                    t.fired := true
                    s := t.dir
            // keep extending the leg AFTER evaluating this bar
            if t.dir == -1
                t.legLo := math.min(t.legLo, low)
            else
                t.legHi := math.max(t.legHi, high)
    s

// A group runs the three trackers (chart, next TF, second TF) on the same IFVG.
// A confirmed OTE on ANY of them is enough to flag the IFVG as an OTE IFVG (once per IFVG).
type OteGroup
    OteTrack t0
    OteTrack t1
    OteTrack t2
    bool flagged = false
    string tags = ""    // timeframes that matched, e.g. "5m+15m"

method gStart(OteGroup g, int dir, float zlo, float zhi, float origin, int sbar) =>
    g.flagged := false
    g.tags := ""
    g.t0.oteStart(dir, zlo, zhi, origin, sbar, pv0)
    if hasH1
        g.t1.oteStart(dir, zlo, zhi, origin, sbar, pv1)
    if hasH2
        g.t2.oteStart(dir, zlo, zhi, origin, sbar, pv2)

method gConfirm(OteGroup g) =>
    g.t0.oteConfirm()
    g.t1.oteConfirm()
    g.t2.oteConfirm()

method gCancel(OteGroup g) =>
    g.flagged := false
    g.tags := ""
    g.t0.oteCancel()
    g.t1.oteCancel()
    g.t2.oteCancel()

method gStep(OteGroup g) =>
    int s0 = g.t0.oteStep()
    int s1 = g.t1.oteStep()
    int s2 = g.t2.oteStep()
    int s = 0
    if not g.flagged and (s0 != 0 or s1 != 0 or s2 != 0)
        g.flagged := true
        s := s0 != 0 ? s0 : (s1 != 0 ? s1 : s2)
        string tg = ""
        if s0 != 0
            tg += g.t0.tag
        if s1 != 0
            tg += (tg == "" ? "" : "+") + g.t1.tag
        if s2 != 0
            tg += (tg == "" ? "" : "+") + g.t2.tag
        g.tags := tg
    s

deRecolor(box b, int d) =>

        color c = d < 0 ? bearIFVGColor : bullIFVGColor
        b.set_bgcolor(color.new(c, ifvgTrans))
        b.set_border_color(color.new(c, borderTrans))

// [DE] inversion rules: every close through the zone inverts it again, up to the max; then it is invalidated.
// Bull candidate (bearish FVG): inversions 1, 3, ... are bullish (close above), 2, 4, ... bearish (close below).
if doifvg and not ifvgswings and deInv and barstate.isconfirmed
    if not na(bullifvg) and bullInv >= 0
        bool brk = bullInv % 2 == 0 ? upBreak(bullifvg.hi) : dnBreak(bullifvg.lo)
        if brk
            if bullInv >= deMaxInv
                bullInv := -1
                bullconfirm := na
                gBull.gCancel() // [OTE]
                deKill(bullifvg)
                ifvgBox := na
            else
                bullInv += 1
                int d = bullInv % 2 == 1 ? 1 : -1
                if bullInv == 1
                    bullconfirm := time
                    gBull.gConfirm() // [OTE]
                else
                    bullconfirm := na
                    gBull.gCancel() // [OTE] only the first inversion
                    deRecolor(ifvgBox, d)
                bullAp := deAp(d)
                ifvgBox.set_text(ifvgTxt(d, bullInv, false, "", bullAp))
                sig := d
    if not na(bearifvg) and bearInv >= 0
        bool brk = bearInv % 2 == 0 ? dnBreak(bearifvg.lo) : upBreak(bearifvg.hi)
        if brk
            if bearInv >= deMaxInv
                bearInv := -1
                bearconfirm := na
                gBear.gCancel() // [OTE]
                deKill(bearifvg)
                ifvgBoxb := na
            else
                bearInv += 1
                int d = bearInv % 2 == 1 ? -1 : 1
                if bearInv == 1
                    bearconfirm := time
                    gBear.gConfirm() // [OTE]
                else
                    bearconfirm := na
                    gBear.gCancel() // [OTE] only the first inversion
                    deRecolor(ifvgBoxb, d)
                bearAp := deAp(d)
                ifvgBoxb.set_text(ifvgTxt(d, bearInv, false, "", bearAp))
                sig := sig == 0 ? d : sig

tfstr() =>
    string tf = timeframe.period
    if timeframe.isseconds
        str.replace(tf, "S", "s")
    else if timeframe.isminutes
        if tf == "60"
            "1h"
        else if tf == "120"
            "2h"
        else if tf == "180"
            "3h"
        else if tf == "240"
            "4h"
        else
            tf + "m"
    else if timeframe.isdaily
        // If it's exactly 1 day ("1D"), UI just shows "D"
        tf == "1D" ? "D" : str.replace(tf, "D", "D") 
    else if timeframe.isweekly
        tf == "1W" ? "W" : str.replace(tf, "W", "W")
    else if timeframe.ismonthly
        tf == "1M" ? "M" : str.replace(tf, "M", "M")
    else
        tf
