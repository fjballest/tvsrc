// ICTlevels: day/week/month levels, session ranges, NDOG/NWOG/RTHOG gaps, EQH/EQL.
// Platform timezone: New York for NQ, Madrid for DAX.
//
// Changes in this version
// - Time-of-day logic uses the script's tz (no timestamp()/exchange-tz mismatch), overnight
// sessions work (Asia label fixed)
//  - Month lines roll over (one security call, lines updated on the last bar)
//  - RTHOG: hint from 15 min before the open, adjusted each bar, fixed at the open bar
//  - All unit mismatches fixed (ms everywhere), dead code and typos removed
//  - Unbroken session max/min get tiny text labels
// - Object budget: day count is capped so lines/labels never exceed the 500 limits (warning in the
// top-right table)
//  - EQH / EQL levels (equal pivot highs/lows within an ATR tolerance, removed when swept), also a
//    pivot that matches an earlier unswept 3-candle swing (equal, or the swing a few points beyond it)
//  - Premarket high / low drawn as red lines whenever its quadrants or stddevs are drawn
//  - Automatic stddevs (optional): the latest RTHOG and premarket range get more stddev lines while the
//    day runs (until the market close), so that every stddev price reached is drawn plus the next one
//  - TGIF: 20% / 30% retracement levels of a weekly displacement candle without its final wick, on
//    Friday, or on Monday when price did not get there on Friday; shown only while they apply
//  - Performance: only active zones are updated per bar, ended zones are only monitored for sweeps,
// right-edge extension and table work happen only on the last bar, history is limited to a time
// window

//@version=6
indicator("ICTlevels", overlay = true, max_lines_count = 500, max_boxes_count = 500,
     max_labels_count = 500, max_bars_back = 5000)

const color transparent = #ffffff00

getstyle(string s) =>
    switch s
        "dashed" => line.style_dashed
        "dotted" => line.style_dotted
        => line.style_solid

getsize(string s) =>
    switch s
        "tiny" => size.tiny
        "small" => size.small
        "large" => size.large
        "huge" => size.huge
        => size.normal

type Sty
    string sty
    int wid
    color col

     or (dow == dayofweek.monday and dowPrev == dayofweek.friday)

// ============================================================ object budget (500 lines / labels /
// boxes)
int nsesEff = math.min(nses, 40)
int ngapsEff = math.min(ngaps, 15)
int gapKinds = (dondog ? 1 : 0) + (donwog ? 1 : 0)
int lnMonths = domonth ? 9 : 0
int lnGaps = dogaps ? gapKinds * 2 * ngapsEff : 0
int lnRth = dogaps and dorthog ? 4 * ngapsEff + 2 * rthogpo3num + 8 + (rthogauto ? 12 : 0)
     + (dorthogoct ? 4 : 0) : 0
int lnSes = doses ? 2 * nsesEff : 0
int lnPre = preOn ? 4 * nsesEff : 0
int lnPmx = preOn ? pmkeep * ((dopmq ? 3 : 0) + (dopmpo3 ? 2 * pmpo3num : 0) + (dopmq or dopmpo3 ? 2 : 0))
     + (pmauto and dopmpo3 ? 12 : 0) : 0
int lnEq = doeqhl ? (doeqd ? 4 : 2) * eqkeep : 0
int lbSes = doses and doseslbl ? 2 * nsesEff : 0
int lbPre = preOn and doseslbl ? 2 * nsesEff : 0
int lnTg = dotgif ? 2 : 0
int lnReserve = lnMonths + lnGaps + lnRth + lnSes + lnPre + lnPmx + lnEq + lnTg + 10
int lbReserve = lnMonths + lnGaps + lnRth + lbSes + lbPre + lnPmx + lnEq + lnTg + 10
int lnPerDay = (dolines ? 1 : 0) + (dodmaxmin ? 2 : 0) + (doeq ? 1 : 0) + (doopen ? 1 : 0)
int lbPerDay = (doeq ? 1 : 0) + (doopen ? 1 : 0) + (dodaylbl and dodmaxmin ? 2 : 0)
int daysByLn = lnPerDay > 0 ? math.floor((500 - lnReserve) / lnPerDay) : ndays
int daysByLb = lbPerDay > 0 ? math.floor((500 - lbReserve) / lbPerDay) : ndays
int ndaysEff = math.max(1, math.min(ndays, math.min(daysByLn, daysByLb)))

// short date of t for labels: day of the month and the month, one letter when no other month starts with it
// (F, S, O, N, D), else three letters (Jan, Mar, Apr, May, Jun, Jul, Aug): 23O, 23Jul
sdate(int t) =>
    int m = month(t, tz)
    string mm = m == 2 ? "F" : m == 9 ? "S" : m == 10 ? "O" : m == 11 ? "N" : m == 12 ? "D"
         : str.format_time(t, "MMM", tz)
    str.tostring(dayofmonth(t, tz)) + mm

    sdate(time)

    label b = label.new(x2, p, txt, xloc = xloc.bar_time, style = label.style_label_left,

        b.set_textcolor(c)

tinyLbl(label cur, bool show, int x, float y, string txt, color c, string sz) =>
    label r = cur
    if show
        if na(r)
            r := label.new(x, y, txt, xloc = xloc.bar_time, style = label.style_label_left,
                 color = transparent, textcolor = c, size = sz)
        else
            r.set_xy(x, y)
            r.set_text(txt)
    else if not na(r)
        r.delete()
        r := na
    r

parseSecs(string s) =>
    60 * (60 * int(str.tonumber(str.substring(s, 0, 2))) + int(str.tonumber(str.substring(s, 2,
         4))))

    ZoneInfo zi = na

        zi := ZoneInfo.new(name, parseSecs(str.substring(ival, 0, 4)),
             parseSecs(str.substring(ival, 5, 9)), sty)
    zi

method inside(ZoneInfo z, int t) =>
    z.ss < z.es ? (t >= z.ss and t < z.es) : (t >= z.ss or t < z.es)

// the bar contains the start instant
method starts(ZoneInfo z) =>
    (z.ss - tsec + 86400) % 86400 < tfsecs

// the bar contains the end instant, or the session was left (data gaps)
method ends(ZoneInfo z) =>
    (z.es - tsec + 86400) % 86400 < tfsecs or (z.inside(tsecPrev) and not z.inside(tsec))

// ============================================================ automatic stddevs
// stddevs to draw on one side of a range: at least n; with auto, also every stddev reached by the extreme
// ext (base: the range edge on that side, u: the step, sgn 1 above, -1 below) plus the next one
sdCount(int n, bool auto, float base, float u, float ext, int sgn) =>
    int r = n
    if auto and not na(ext) and u > 0
        int reached = math.max(0, int(math.floor((ext - base) * sgn / u)))
        r := math.max(n, math.min(reached + 1, 40))
    r

// stddev name: distance from the range edge in whole ranges (step i of u, range r), for example "+0.25", "-1.5"
sdName(int i, float u, float r, int sgn) =>
    (sgn > 0 ? "+" : "-") + str.tostring(i * u / r, "#.###")

// premarket quadrant and stddev objects
method pmclear(Zone z) =>
    if not na(z.qlines)
        for l in z.qlines
            rmLine(l)
        z.qlines.clear()
    if not na(z.qlabels)
        for b in z.qlabels
            rmLabel(b)
        z.qlabels.clear()
    if not na(z.sdlines)
        for l in z.sdlines
            rmLine(l)
        z.sdlines.clear()
    if not na(z.sdlabels)
        for b in z.sdlabels
            rmLabel(b)
        z.sdlabels.clear()
    true

// quadrants (25%, CE, 75%) and stddevs of the range, like the RTHOG ones. Drawn from a
// couple of bars before the end of the range (or the last bar while it is forming)
method pmdraw(Zone z, bool auto = false) =>
    z.pmclear()
    float r = z.hi - z.lo
    int x1 = na(z.tclose) ? tright(xoff) : math.min(tright(xoff), z.tclose)
    int x0 = (na(z.tend) ? time : z.tend) - tfms * rdx
    string nm = sdate(z.tstart)
    if r > 0
        if dopmq and r / 4.0 > pmqmin
            if na(z.qlines)
                z.qlines := array.new<line>()
                z.qlabels := array.new<label>()
            for k = 1 to 3
                string pt = k == 1 ? "dq" : k == 2 ? "ce" : "uq"
                [l1, b1] = mkLL(x0, x1, z.lo + k * r / 4.0, pmcolor, k == 2 ? pmcesty : pmqsty,
                     pmwid, "PMKT " + pt + " " + nm, pmcolor, pmlsz)
                z.qlines.push(l1)
                z.qlabels.push(b1)
        if dopmpo3
            float u = 0.0
            if r / 8 > pmpo3min
                u := r / 8
            else if r / 4 > pmpo3min
                u := r / 4
            else if r / 2 > pmpo3min
                u := r / 2
            else if r > pmpo3min
                u := r
            if u > 0
                if na(z.sdlines)
                    z.sdlines := array.new<line>()
                    z.sdlabels := array.new<label>()
                    z.sdnames := array.new<string>()
                z.sdnames.clear()
                int nu = sdCount(pmpo3num, auto, z.hi, u, z.xhi, 1)
                int nd = sdCount(pmpo3num, auto, z.lo, u, z.xlo, -1)
                for i = 1 to math.max(nu, nd)
                    if i <= nu
                        [l2, b2] = mkLL(x0, x1, z.hi + i * u, pmcolor, pmpo3sty, pmwid,
                             "PMKT" + sdName(i, u, r, 1), pmcolor, pmlsz)
                        z.sdlines.push(l2)
                        z.sdlabels.push(b2)
                        z.sdnames.push(sdName(i, u, r, 1))
                    if i <= nd
                        [l3, b3] = mkLL(x0, x1, z.lo - i * u, pmcolor, pmpo3sty, pmwid,
                             "PMKT" + sdName(i, u, r, -1), pmcolor, pmlsz)
                        z.sdlines.push(l3)
                        z.sdlabels.push(b3)
                        z.sdnames.push(sdName(i, u, r, -1))
        // high and low of the range, red solid lines, whenever quadrants or stddevs are drawn
        bool anyq = not na(z.qlines) and z.qlines.size() > 0
        bool anysd = not na(z.sdlines) and z.sdlines.size() > 0
        if anyq or anysd
            if na(z.qlines)
                z.qlines := array.new<line>()
                z.qlabels := array.new<label>()
            [lh, bh] = mkLL(x0, x1, z.hi, pmcolor, line.style_solid, pmwid, "PMKT hi " + nm, pmcolor, pmlsz)
            z.qlines.push(lh)
            z.qlabels.push(bh)
            [ll, bl] = mkLL(x0, x1, z.lo, pmcolor, line.style_solid, pmwid, "PMKT lo " + nm, pmcolor, pmlsz)
            z.qlines.push(ll)
            z.qlabels.push(bl)
    true

method kill(Zone z) =>
    z.dead := true
    z.pmclear()
    rmBox(z.rect)
    rmLine(z.vline)
    rmLine(z.vline2)
    rmLine(z.maxln)
    rmLine(z.minln)
    rmLine(z.eqln)
    rmLine(z.opln)
    rmLabel(z.maxlbl)
    rmLabel(z.minlbl)
    rmLabel(z.eqlabel)
    rmLabel(z.oplabel)

// create or update every object of the zone from its current state
method refresh(Zone z) =>
    int re = tright(2)
    if not na(z.tclose)
        re := math.min(re, z.tclose)
    int xe = na(z.tend) ? re : z.tend
    if z.drawvlines or (z.isday and dolines)
        if na(z.vline)
            z.vline := line.new(z.tstart, z.lo, z.tstart, z.hi + syminfo.mintick,
                 extend = extend.both, xloc = xloc.bar_time,
                 color = z.lnsty.col, style = z.isweek ? wdaylnsty : z.lnsty.sty, width = z.isweek
                 ? wdaywid : z.lnsty.wid)
        if z.drawvlines and not na(z.tend) and na(z.vline2)
            int x2 = z.tend + tfms
            z.vline2 := line.new(x2, z.lo, x2, z.hi + syminfo.mintick, extend = extend.both,
                 xloc = xloc.bar_time,
                 color = z.lnsty.col, style = z.lnsty.sty, width = z.lnsty.wid)
    if z.drawmaxmin
        int hx = na(z.hidonet) ? re : math.min(z.hidonet, re)
        int lx = na(z.lodonet) ? re : math.min(z.lodonet, re)
        if na(z.maxln)
            z.maxln := line.new(z.hit, z.hi, hx, z.hi, xloc = xloc.bar_time, color = z.maxsty.col,
                 width = z.maxsty.wid, style = z.maxsty.sty)
        else
            z.maxln.set_xy1(z.hit, z.hi)
            z.maxln.set_xy2(hx, z.hi)
        if na(z.minln)
            z.minln := line.new(z.lot, z.lo, lx, z.lo, xloc = xloc.bar_time, color = z.minsty.col,
                 width = z.minsty.wid, style = z.minsty.sty)
        else
            z.minln.set_xy1(z.lot, z.lo)
            z.minln.set_xy2(lx, z.lo)
        if z.drawlbl
            string lsz = z.isday ? dlblsz : slblsz
            string hn = z.name + " hi"
            string ln = z.name + " lo"
            if z.isday
                hn := na(z.pwhi) or z.hi >= z.pwhi ? "week high" : "day high"
                ln := na(z.pwlo) or z.lo <= z.pwlo ? "week low" : "day low"
                if z.hmode == 1
                    hn := "day REHs"
                if z.lmode == 1
                    ln := "day RELs"
            z.maxlbl := tinyLbl(z.maxlbl, na(z.hidonet) and z.hmode != 2, re, z.hi, hn,
                 z.maxsty.col, lsz)
            z.minlbl := tinyLbl(z.minlbl, na(z.lodonet) and z.lmode != 2, re, z.lo, ln,
                 z.minsty.col, lsz)
    if z.isday and doeq
        float eq = math.avg(z.hi, z.lo)
        int ex = na(z.extendt) ? re : z.extendt
        int x0 = math.min(z.hit, z.lot)
        if na(z.eqln)
            z.eqln := line.new(x0, eq, ex, eq, xloc = xloc.bar_time, color = eqcolor, width = eqwid,
                 style = eqlnsty)
            z.eqlabel := label.new(ex, eq, "eq", xloc = xloc.bar_time,
                 style = label.style_label_left, color = transparent, textcolor = eqcolor)
        else
            z.eqln.set_xy1(x0, eq)
            z.eqln.set_xy2(ex, eq)
            z.eqlabel.set_xy(ex, eq)
    if z.isday and doopen
        if na(z.opln)
            z.opln := line.new(z.tstart, z.op, xe, z.op, xloc = xloc.bar_time, color = opcolor,
                 width = opwid, style = oplnsty)
            z.oplabel := label.new(xe, z.op, "M.O.", xloc = xloc.bar_time,
                 style = label.style_label_left, color = transparent, textcolor = opcolor)
        else
            z.opln.set_x2(xe)
            z.oplabel.set_x(xe)
    if z.drawrect
        if na(z.rect)
            z.rect := box.new(z.tstart, z.hi, xe, z.lo, xloc = xloc.bar_time,
                 border_color = rectbord ? color.new(z.lnsty.col, rtransp / 2) : transparent,
                 border_width = rectbord ? 2 : 1, bgcolor = rectbord ? transparent : z.lnsty.col)
        else
            z.rect.set_right(xe)
            z.rect.set_top(z.hi)
            z.rect.set_bottom(z.lo)
    z

// needs right-edge updates on the last bar
method live(Zone z) =>
    na(z.tend) or (z.drawmaxmin and (na(z.hidonet) or na(z.lodonet))) or (z.isday and na(z.extendt))

// only state, no drawing: cheap per bar
method upd(Zone z) =>
    if high >= z.hi
        z.hit := time
        z.hi := high
    if low <= z.lo
        z.lot := time
        z.lo := low

endZone(Zone z) =>
    if na(z.tend)
        z.tend := time[1]
        z.refresh()
        if z.drawmaxmin
            monz.push(z)

runActive() =>
    int i = actz.size() - 1
    while i >= 0
        Zone z = actz.get(i)
        if z.dead or not na(z.tend)
            actz.remove(i)
        else if not na(z.zi) and z.zi.ends()
            endZone(z)
            actz.remove(i)
        else
            z.upd()
        i -= 1

// unbroken day highs (lows) close to each other: the lines stay, but only the higher high
// (lower low) keeps a label, which reads REHs (RELs)
markRel(array<Zone> arr, float tol) =>
    int n = arr.size()
    if n > 0
        for i = 0 to n - 1
            Zone a = arr.get(i)
            a.hmode := 0
            a.lmode := 0
            for j = 0 to n - 1
                if j != i
                    Zone b = arr.get(j)
                    if na(a.hidonet) and na(b.hidonet) and math.abs(a.hi - b.hi) <= tol
                        if b.hi > a.hi or (b.hi == a.hi and j > i)
                            a.hmode := 2
                        else if a.hmode == 0
                            a.hmode := 1
                    if na(a.lodonet) and na(b.lodonet) and math.abs(a.lo - b.lo) <= tol
                        if b.lo < a.lo or (b.lo == a.lo and j > i)
                            a.lmode := 2
                        else if a.lmode == 0
                            a.lmode := 1

    true

// ended zones: only watch for the max/min being taken
runMon() =>
    int i = monz.size() - 1
    while i >= 0
        Zone z = monz.get(i)
        if z.dead
            monz.remove(i)
        else
            bool ch = false
            if na(z.hidonet) and high > z.hi
                z.hidonet := time
                ch := true
            if na(z.lodonet) and low < z.lo
                z.lodonet := time
                ch := true
            if ch
                z.refresh()
            if not na(z.hidonet) and not na(z.lodonet)
                monz.remove(i)
        i -= 1

if inWin and tfok
    if newDay
        Zone d = mkZone("D")
        d.isday := true
        d.isweek := newWeek
        d.lnsty := daysty
        d.maxsty := daymaxsty
        d.minsty := dayminsty
        d.drawmaxmin := dodmaxmin
        d.drawlbl := dodaylbl
        if days.size() > 0
            Zone ld = days.last()
            if not d.isweek

    if doses

    if preOn and premarket.starts()

        p.drawmaxmin := dosmaxmin

    runMon()

// ============================================================ TGIF
// Week candle (from newWeek) tracked here. On the first bar of Friday the week so far is tested: a
// displacement (price within tgnear x range of the week high / low, whatever the body) arms the
// levels, measured from the week high (bullish, retracement down) or low
// (bearish, retracement up). On Friday they follow new week extremes until price reaches the first
// level, then stay fixed. When the week closes displaced and the retracement did not happen on Friday,
// the levels of the final week apply on Monday (until Tuesday starts). Otherwise they are removed when
// the day ends.
tgEval(float h, float l, float o, float c) =>
    float r = h - l
    int d = 0
    if r > 0
        if h - c <= tgnear * r
            d := 1
        else if c - l <= tgnear * r
            d := -1
    d

var float twO = na      // week open, high, low
var float twH = na
var float twL = na
var int tgDir = 0       // 1: bullish week, retracement down; -1: bearish week, retracement up; 0: none
var bool tgMon = false  // levels carried to Monday
var bool tgHit = false  // the first level was reached (the levels stay fixed)
var float tgX = na      // extreme and range the levels are measured from
var float tgR = na
var int tgT0 = na       // start of the lines
var line tgL1 = na
var line tgL2 = na
var label tgB1 = na
var label tgB2 = na

if inWin and tfok and dotgif
    bool clr = false
    if newWeek
        int d = na(twO) ? 0 : tgEval(twH, twL, twO, close[1])
        bool carry = d != 0 and not (tgDir == d and tgHit)
        clr := true
        tgDir := 0
        tgMon := false
        tgHit := false
        if carry
            tgDir := d
            tgMon := true
            tgX := d == 1 ? twH : twL
            tgR := twH - twL
            tgT0 := time
        twO := open
        twH := high
        twL := low
    else if not na(twO)
        if newDay and dow == dayofweek.friday
            clr := true
            tgDir := tgEval(twH, twL, twO, close[1])
            tgMon := false
            tgHit := false
            tgT0 := time
        else if newDay and tgDir != 0 and not (tgMon and dow == dayofweek.monday)
            // Friday or Monday is over (Monday levels start with the week, before Monday's midnight)
            clr := true
            tgDir := 0
            tgMon := false
        twH := math.max(twH, high)
        twL := math.min(twL, low)
    if clr
        rmLine(tgL1)
        rmLine(tgL2)
        rmLabel(tgB1)
        rmLabel(tgB2)
        tgL1 := na
        tgL2 := na
        tgB1 := na
        tgB2 := na
    if tgDir != 0
        if not tgMon and not tgHit
            tgX := tgDir == 1 ? twH : twL
            tgR := twH - twL
        float v1 = tgX - tgDir * tgp1 * tgR
        float v2 = tgX - tgDir * tgp2 * tgR
        color c = tgDir == 1 ? tgdncol : tgupcol
        int x1 = tright(xoff)
        if na(tgL1)
            [l1, b1] = mkLL(tgT0, x1, v1, c, tgsty, tgwid, "TGIF" + str.tostring(tgp1 * 100, "#.#"), c,
                 tglsz)
            [l2, b2] = mkLL(tgT0, x1, v2, c, tgsty, tgwid, "TGIF" + str.tostring(tgp2 * 100, "#.#"), c,
                 tglsz)
            tgL1 := l1
            tgB1 := b1
            tgL2 := l2
            tgB2 := b2
        else
            tgL1.set_y1(v1)
            tgL1.set_y2(v1)
            tgL1.set_x2(x1)
            tgB1.set_xy(x1, v1)
            tgL2.set_y1(v2)
            tgL2.set_y2(v2)
            tgL2.set_x2(x1)
            tgB2.set_xy(x1, v2)
        if not tgHit and (tgDir == 1 ? low <= v1 : high >= v1)
            tgHit := true

// text of the gap without the date: plain, or combined (NDOGx3, NWOGx2, NWOG+NDOG)

    else if g.n > 1

    b

// labels of the levels h and l: two labels "NDOGx3 hi 23O" / "NDOGx3 lo 23O", or a single one in the middle
// ("NDOGx3 23O") when the levels are within tol of each other. The date is the one of the gap that draws them.
method glabels(Gap g, float h, float l, float tol, int x) =>

    if small
        rmLabel(g.lolbl)
        g.lolbl := na
    else
        if na(g.lolbl)
            g.lolbl := label.new(x, l, b + " lo" + dt, xloc = xloc.bar_time,
                 style = label.style_label_left, color = transparent, textcolor = gs.col)
        g.lolbl.set_xy(x, l)
        g.lolbl.set_text(b + " lo" + dt)
    true

// ---- combining gaps at similar levels (derived state, recomputed on the last bar)
method gcomb(Gap g, float tol) =>
    if not g.show
        g.kill()
    else
        if na(g.hiln)
            g.gmake()
        int x = tright(xoff)
        g.hiln.set_xy1(g.ut, g.uhi)
        g.hiln.set_xy2(x, g.uhi)
        g.loln.set_xy1(g.ut, g.ulo)
        g.loln.set_xy2(x, g.ulo)
        g.glabels(g.uhi, g.ulo, tol, x)
        if not na(g.rect)
            g.rect.set_lefttop(g.ut, g.uhi)
            g.rect.set_rightbottom(x, g.ulo)
    true

// gaps whose ranges overlap or are within tol are one group: the oldest draws the
// outer levels of the group, the others are hidden
gcluster(array<Gap> arr, float tol) =>
    int n = arr.size()
    // a for loop from 0 to n - 1 runs backwards when n = 0, so an empty list is skipped
    if n > 0
        array<int> cid = array.new<int>()
        for i = 0 to n - 1
            Gap g = arr.get(i)
            g.n := 1
            g.nd := 0
            g.show := true
            g.uhi := g.hi
            g.ulo := g.lo
            g.ut := g.tstart
            cid.push(i)
        bool ch = n > 1
        while ch
            ch := false
            for i = 0 to n - 2
                for j = i + 1 to n - 1
                    if cid.get(i) != cid.get(j)
                        Gap a = arr.get(i)
                        Gap b = arr.get(j)
                        if a.hi + tol >= b.lo and b.hi + tol >= a.lo
                            int from = cid.get(j)
                            int xto = cid.get(i)
                            for k = 0 to n - 1
                                if cid.get(k) == from
                                    cid.set(k, xto)
                            ch := true
        array<int> first = array.new<int>(n, -1)
        for i = 0 to n - 1
            int c = cid.get(i)
            int r = first.get(c)
            if r < 0
                first.set(c, i)
            else
                Gap rep = arr.get(r)
                Gap g = arr.get(i)
                rep.n += 1
                rep.uhi := math.max(rep.uhi, g.hi)
                rep.ulo := math.min(rep.ulo, g.lo)
                g.show := false

    true

combineGaps(array<Gap> dg, array<Gap> wg, float tol) =>
    gcluster(dg, tol)
    gcluster(wg, tol)
    for w in wg
        if w.show
            for d in dg
                if d.show and math.abs(d.uhi - w.uhi) <= tol and math.abs(d.ulo - w.ulo) <= tol
                    w.nd += d.n
                    d.show := false
    for g in dg
        g.gcomb(tol)
    for g in wg
        g.gcomb(tol)
    true

// octants: 1/8, 3/8, 5/8, 7/8 of the gap, from the open to noon (or the right edge before noon)
method roct(RTHOG g) =>
    g.killoct()
    float ody = math.abs(g.op - g.cl) / 8.0
    g.noon := timestamp(tz, year(g.tend, tz), month(g.tend, tz), dayofmonth(g.tend, tz), 12, 0)
    int x0 = g.tend - tfms * rdx
    int x1 = math.min(tright(xoff), g.noon)
    if ody > rthogqmin and x1 > x0
        if na(g.oclines)
            g.oclines := array.new<line>()
            g.oclabels := array.new<label>()
        float p = math.min(g.cl, g.op)
        string nm = timestr()
        for k in array.from(1, 3, 5, 7)
            [l, b] = mkLL(x0, x1, p + k * ody, lrgapcolor, rgapqlnsty, 1, "RTHOG " + str.tostring(k) + "/8 "
                 + nm, lrgapcolor)
            g.oclines.push(l)
            g.oclabels.push(b)

// last bar: the octants follow the right edge until noon

method rpo3(RTHOG g) =>
    g.killpo3()
    float gap = math.abs(g.op - g.cl)
    float u = 0.0
    if gap / 8 > rthogpo3min
        u := gap / 8
    else if gap / 4 > rthogpo3min
        u := gap / 4
    else if gap / 2 > rthogpo3min
        u := gap / 2
    else if gap > rthogpo3min
        u := gap
    if u > 0
        if na(g.po3lines)
            g.po3lines := array.new<line>()
            g.po3labels := array.new<label>()
            g.po3names := array.new<string>()
        g.po3names.clear()
        int x0 = g.tend - tfms * rdx
        int x1 = tright(xoff)
        float pu0 = math.max(g.cl, g.op)
        float pd0 = math.min(g.cl, g.op)
        int nu = sdCount(rthogpo3num, rthogauto, pu0, u, g.xhi, 1)
        int nd = sdCount(rthogpo3num, rthogauto, pd0, u, g.xlo, -1)
        for i = 1 to math.max(nu, nd)
            if i <= nu
                [l1, b1] = mkLL(x0, x1, pu0 + i * u, lrgapcolor, rgappo3lnsty, rgapwid, "RTHOG"
                     + sdName(i, u, gap, 1), lrgapcolor)
                g.po3lines.push(l1)
                g.po3labels.push(b1)
                g.po3names.push(sdName(i, u, gap, 1))
            if i <= nd
                [l2, b2] = mkLL(x0, x1, pd0 - i * u, lrgapcolor, rgappo3lnsty, rgapwid, "RTHOG"
                     + sdName(i, u, gap, -1), lrgapcolor)
                g.po3lines.push(l2)
                g.po3labels.push(b2)
                g.po3names.push(sdName(i, u, gap, -1))

    // automatic stddevs: extremes of the day from the open (open bar included) until the close
    if rthogauto and not mktClose and rthogs.size() > 0
        RTHOG ag = rthogs.last()
        if not na(ag.tend) and time > ag.tend
            ag.xhi := math.max(ag.xhi, high)
            ag.xlo := math.min(ag.xlo, low)
// automatic stddevs: extremes of the latest premarket range from its end until the market close
if preOn and pmauto and premarkets.size() > 0
    Zone az = premarkets.last()
    if not na(az.tend) and time > az.tend and (na(az.tclose) or time < az.tclose)
        az.xhi := na(az.xhi) ? high : math.max(az.xhi, high)
        az.xlo := na(az.xlo) ? low : math.min(az.xlo, low)

// ============================================================ EQH / EQL
float atrv = ta.atr(14)
float ph = ta.pivothigh(high, eqlen, eqlen)
float pl = ta.pivotlow(low, eqlen, eqlen)
int pivT = time[eqlen]

dropFirst(array<Piv> arr, bool wantEq) =>
    for i = 0 to arr.size() - 1
        Piv v = arr.get(i)
        if v.eq == wantEq
            rmLine(v.ln)
            rmLabel(v.lb)
            arr.remove(i)
            break

trimPivs(array<Piv> arr) =>
    int ne = 0
    for v in arr
        if v.eq
            ne += 1
    while ne > eqkeep
        dropFirst(arr, true)
        ne -= 1
    while arr.size() - ne > 30
        dropFirst(arr, false)

sweepPivs(array<Piv> arr, bool isHigh, float tol) =>
    int i = arr.size() - 1
    while i >= 0
        Piv v = arr.get(i)
        float slack = v.eq ? 0.0 : tol
        bool swept = isHigh ? high > v.p + slack : low < v.p - slack
        if swept
            rmLine(v.ln)
            rmLabel(v.lb)
            arr.remove(i)
        i -= 1

addPiv(array<Piv> arr, float p, int t, bool isHigh, float tol) =>
    Piv m = na
    for v in arr
        if math.abs(v.p - p) <= tol
            m := v
            break
    if not na(m)
        m.eq := true
        if isHigh ? p > m.p : p < m.p
            m.p := p
    else
        arr.push(Piv.new(p, t))
    trimPivs(arr)

drawEq(array<Piv> arr, string txt, color c) =>
    int x = tright(xoff)
    for v in arr
        if v.eq
            if na(v.ln)
                v.ln := line.new(v.t, v.p, x, v.p, xloc = xloc.bar_time, color = c, style = eqhlsty,
                     width = eqhlwid)
                v.lb := label.new(x, v.p, txt, xloc = xloc.bar_time, style = label.style_label_left,
                     color = transparent, textcolor = c, size = lblsz,
                     text_formatting = text.format_bold)
            else
                v.ln.set_xy1(v.t, v.p)
                v.ln.set_xy2(x, v.p)
                v.lb.set_xy(x, v.p)
    true
// 3-candle swings (one candle on each side), kept until price takes them
float ph3 = ta.pivothigh(high, 1, 1)
float pl3 = ta.pivotlow(low, 1, 1)
var array<float> sh3p = array.new<float>()
var array<int> sh3t = array.new<int>()
var array<float> sl3p = array.new<float>()
var array<int> sl3t = array.new<int>()

sweep3(array<float> ps, array<int> ts, bool isHigh) =>
    int i = ps.size() - 1
    while i >= 0
        if isHigh ? high > ps.get(i) : low < ps.get(i)
            ps.remove(i)
            ts.remove(i)
        i -= 1
    true

add3(array<float> ps, array<int> ts, float p, int t) =>
    ps.push(p)
    ts.push(t)
    if ps.size() > 60
        ps.shift()
        ts.shift()
    true

// the pivot just added at time t (price p) becomes an EQH / EQL when an earlier unswept 3-candle swing
// has the same price or is at most eq3pts beyond it (above for highs, below for lows). The level is the
// swing price, drawn from the swing.
eq3(array<Piv> arr, array<float> ps, array<int> ts, float p, int t, bool isHigh) =>
    if arr.size() > 0
        Piv v = arr.last()
        if v.t == t and not v.eq
            int n = ps.size()
            if n > 0
                for i = n - 1 to 0
                    float q = ps.get(i)
                    int qt = ts.get(i)
                    float d = isHigh ? q - p : p - q
                    if qt < t and d >= 0 and d <= eq3pts
                        v.eq := true
                        v.p := q
                        v.t := qt
                        trimPivs(arr)
                        break
    true

if inWin and tfok and doeqhl

    sweepPivs(lpv, false, tol)
    sweep3(sh3p, sh3t, true)
    sweep3(sl3p, sl3t, false)
    if not na(ph)
        addPiv(hpv, ph, pivT, true, tol)
        if doeq3
            eq3(hpv, sh3p, sh3t, ph, pivT, true)
    if not na(pl)
        addPiv(lpv, pl, pivT, false, tol)
        if doeq3
            eq3(lpv, sl3p, sl3t, pl, pivT, false)
    if not na(ph3)
        add3(sh3p, sh3t, ph3, time[1])
    if not na(pl3)
        add3(sl3p, sl3t, pl3, time[1])

// ============================================================ RTHOG / premarket stddev labels
// stddev lines of the RTHOG and the premarket range at almost the same price share one
// label: "RTHOG+0.5 PMKT+1". The other label is emptied.
combinePo3(RTHOG g, Zone z, float tol) =>
    if not na(g.po3labels) and not na(z.sdlabels) and not na(g.po3names) and not na(z.sdnames)
        for [k, l] in g.po3labels
            l.set_text("RTHOG" + g.po3names.get(k))
        for [k, l] in z.sdlabels
            l.set_text("PMKT" + z.sdnames.get(k))
        array<bool> used = array.new<bool>(z.sdlabels.size(), false)
        for [k, l] in g.po3labels
            for [m, q] in z.sdlabels
                if not used.get(m) and math.abs(l.get_y() - q.get_y()) <= tol
                    l.set_text("RTHOG" + g.po3names.get(k) + " PMKT" + z.sdnames.get(m))
                    q.set_text("")
                    used.set(m, true)
                    break
    true
