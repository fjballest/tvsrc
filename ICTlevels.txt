monthName(n) =>
    switch n
        1  => "Jan"
        2  => "Feb"
        3  => "Mar"
        4  => "Apr"
        5  => "May"
        6  => "Jun"
        7  => "Jul"
        8  => "Aug"
        9  => "Sep"
        10 => "Oct"
        11 => "Nov"
        12 => "Dec"

timestr(t) =>
    str.tostring(dayofmonth(t)) + " " + monthName(month(t))

toff(n) =>
    (time-time[1])*n
timeafter(n) =>
    time + toff(n)
timeafterlast(n) =>
    time+toff(n)

method kill(RTHOG g) =>
    if not na(g.cline)
        line.delete(g.cline)
        g.cline := na
    if not na(g.mline)
        line.delete(g.mline)
        g.mline := na
    if not na(g.cline2)
        line.delete(g.cline2)
        g.cline2 := na
    if not na(g.opline)
        line.delete(g.opline)
        g.opline := na
    if not na(g.clabel)
        label.delete(g.clabel)
        g.clabel := na
    if not na(g.mlabel)
        label.delete(g.mlabel)
        g.mlabel := na
    if not na(g.clabel2)
        label.delete(g.clabel2)
        g.clabel2 := na
    if not na(g.oplabel)
        label.delete(g.oplabel)
        g.oplabel := na
    if not na(g.rect)
        g.rect.delete()
        g.rect := na

method rstarts(RTHOG g) =>
    g.isold := false
    dx = (timeframe.period == "1")  ? 4 : 2
    g.cl := close[1]
    g.sty := rgapsty
    g.tstart := time[1] // it is actually time_close, but use time to draw the line from the end bar
    log.info("market close at {0} {1}", g.cl, g.tstart)
    g.tend := na
    g.op := na
    g.kill()
    g.cline := line.new(g.tstart, g.cl, timeafterlast(xoff), g.cl, xloc = xloc.bar_time, color = rgapcolor, width=g.sty.wid, style=g.sty.sty)
    g.clabel := label.new(timeafterlast(xoff), g.cl, "new RTHOG settlement", 
        xloc = xloc.bar_time, style = label.style_label_left, color=transparent, textcolor = rgapcolor)

method drawOldEnded(RTHOG g) =>
    g.sty := rgapsty
    dx = (timeframe.period == "1")  ? 4 : 2
    if not na(g.cline)
        g.cline.delete()
        g.clabel.delete()
        g.cline := line.new(g.tstart-toff(dx), g.cl, g.tstart+toff(dx), g.cl, 
            xloc = xloc.bar_time, color = g.sty.col, width=g.sty.wid, style=g.sty.sty)
        g.clabel := label.new(g.tstart+toff(dx), g.cl, "rthog.cl", 
            xloc = xloc.bar_time, style = label.style_label_left, color=transparent, textcolor = color.black)

    if not na(g.cline2)
        g.cline2.delete()
    if not na(g.clabel2)
        g.clabel2.delete()
    te0 = g.tend-toff(dx)
    te1 = g.tend+toff(dx)
    g.cline2 := line.new(te0, g.cl, te1, g.cl, 
        xloc = xloc.bar_time, style=g.sty.sty, color = g.sty.col, width = g.sty.wid)
    g.clabel2 := label.new(te1, g.cl, "rthog.cl", 
        xloc = xloc.bar_time, style = label.style_label_left, color=transparent, textcolor = color.black)   
    mid = math.avg(g.cl, g.op)
    if not na(g.mline)
        g.mline.delete()
    if not na(g.mlabel)
        g.mlabel.delete()
    g.mline := line.new(te0, mid, te1, mid, xloc = xloc.bar_time, color = g.sty.col, style = g.sty.sty, width = g.sty.wid)
    g.mlabel := label.new(te1, mid, "rthog.ce", 
        xloc = xloc.bar_time, style = label.style_label_left, color=transparent, textcolor = color.black)
    if not na(g.opline)
        g.opline.delete()
    if not na(g.oplabel)
        g.oplabel.delete()
    g.opline := line.new(te0, g.op, te1, g.op, xloc = xloc.bar_time, color=g.sty.col, style = g.sty.sty, width = g.sty.wid)
    g.oplabel := label.new(te1, g.op, "rtog.op", 
        xloc = xloc.bar_time, style = label.style_label_left, color = transparent, textcolor = color.black)
    if (g.extendgaps) 
        g.cline2.set_extend(extend.right)
        g.mline.set_extend(extend.right)
        g.opline.set_extend(extend.right)
    else
        g.cline2.set_extend(extend.none)
        g.mline.set_extend(extend.none)
        g.opline.set_extend(extend.none)
    if not na(g.rect)
        g.rect.delete()
        g.rect := na

method drawEnded(RTHOG g) =>
    dx = (timeframe.period == "1")  ? 4 : 2

    mid = math.avg(g.cl, g.op)
    if not na(g.mline)
        g.mline.delete()
    if not na(g.mlabel)
        g.mlabel.delete()
    if not na(g.opline)
        g.opline.delete()
    if not na(g.oplabel)
        g.oplabel.delete()
    if not na(g.rect)
        g.rect.delete()
    nm = timestr(time)
    te1 = timeafterlast(xoff)
    g.mline := line.new(g.tend-toff(dx), mid, te1, mid, 
        xloc = xloc.bar_time, color = rgapcolor, style = g.sty.sty, width = g.sty.wid)
    g.mlabel := label.new(te1, mid, "RTHOG ce " +nm, 
        xloc = xloc.bar_time, style = label.style_label_left, color=transparent, textcolor = rgapcolor)
    g.opline := line.new(g.tend, g.op, te1, g.op, 
        xloc = xloc.bar_time, color=rgapcolor, style = g.sty.sty, width = g.sty.wid)
    g.oplabel := label.new(te1, g.op, "RTHOG op", 
        xloc = xloc.bar_time, style = label.style_label_left, color = transparent, textcolor = rgapcolor)
    if g.op > g.cl
        g.oplabel.set_text("RTHOG hi " + nm)
        g.clabel.set_text("RTHOG lo " +nm)
    else
        g.oplabel.set_text("RTHOG lo "+nm)
        g.clabel.set_text("RTHOG hi "+ nm)
    if dogrect 
        g.rect := box.new(g.tend-toff(dx), g.op, te1, g.cl, xloc = xloc.bar_time, border_color = transparent,
        bgcolor = color.new(rgapcolor, grtransp))
method rends(RTHOG g) =>
    g.op := open
    g.tend := time
    g.drawEnded()
       

method setOld(RTHOG g) =>
    g.isold := true
    g.drawOldEnded()
 
method upd(RTHOG g) =>
    if not g.isold
        te1 = timeafterlast(xoff)
        if not na(g.clabel)
            g.clabel.set_x(te1)
            g.cline.set_x2(te1)
        if not na(g.oplabel)
            g.opline.set_x2(te1)
            g.oplabel.set_x(te1)
        if not na(g.rect)
            g.rect.set_right(te1)
        if not na(g.mlabel)
            g.mlabel.set_x(te1)
        if not na(g.mline)
            g.mline.set_x2(te1)
        

hmParse(string s) =>
    hs = str.substring(s, 0, 2)
    ms = str.substring(s, 2, 4)
    [int(str.tonumber(hs)), int(str.tonumber(ms))]

sesParse(string s) =>
    vals = str.split(s, '-')
    st = vals.size() > 0 ? vals.get(0) : "0000"
    end = vals.size() > 1 ? vals.get(1) : st
    [sth, stm] = hmParse(st)

    if zn.name == "D"  and doeq

            zn.eqln := line.new(math.min(zn.hit,zn.lot), eq, e, eq, xloc = xloc.bar_time,
                color = zn.eqsty.col, width = zn.eqsty.wid, style=zn.eqsty.sty)
            zn.eqlabel := label.new(e, eq, "eq", style = label.style_label_left, xloc = xloc.bar_time,

        else

            zn.eqln.set_y2(eq)
            zn.eqlabel.set_y(eq)

            zn.eqlabel.set_x(e)
            zn.eqln.set_x2(e)
    true

method update(Zone zn, Zone last=na) =>
    if na(zn.tend)
        if high >= zn.hi 
            zn.hit := time
            zn.hi := high
            if not na(zn.maxln)
                zn.maxln.set_x1(time)
                zn.maxln.set_y1(zn.hi)
                zn.maxln.set_y2(zn.hi)
        if low <= zn.lo
            zn.lot := time
            zn.lo := low
            if not na(zn.minln)
                zn.minln.set_x1(time)
                zn.minln.set_y1(zn.lo)
                zn.minln.set_y2(zn.lo)
    else
        if high > zn.hi and na(zn.hidonet)
            zn.hidonet := time
        if low < zn.lo and na(zn.lodonet)
            zn.lodonet := time
    if not na(last) and not na(zn.tend) and na(last.extendt)
        last.extendt := zn.tend

    xe = e

        xe := open

    g.hi := math.max(s, xe)

        g.name := "NWOG " + timestr(time)
    else
        g.name := "NDOG " + timestr(time)

            bgcolor = color.new(g.sty.col, grtransp))
