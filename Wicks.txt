//@version=6
indicator("Wicks", overlay=true, max_lines_count=500)

// Redraws the candle wicks as thick lines, as OTEFibWick does, and nothing else.

GW = "Wicks"
wick_width = input.int(2, "Wick Width", group = GW, minval = 1, maxval = 10, inline = "w")
wick_color = input.color(color.gray, "color", group = GW, inline = "w")
dAlt = input.bool(false, "Data from another symbol", group = GW, inline = "dsym",
     tooltip = "Draw the wicks of this symbol's candles, at the chart timeframe, instead of the chart's own")
dSym = input.symbol("CME_MINI:MNQ1!", "", group = GW, inline = "dsym")

string dataSym = dAlt ? dSym : syminfo.tickerid
[dO, dH, dL, dC] = request.security(dataSym, timeframe.period, [open, high, low, close], gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off)

// one line per wick; TradingView removes the oldest lines past max_lines_count (500), so the last wicks that fit
// are kept. On the live bar each tick is rolled back with its lines, so a new line per tick is right.
if not na(dO)
    float bodyTop = math.max(dO, dC)
    float bodyBot = math.min(dO, dC)
    if dH > bodyTop
        line.new(bar_index, dH, bar_index, bodyTop, color = wick_color, width = wick_width)
    if dL < bodyBot
        line.new(bar_index, dL, bar_index, bodyBot, color = wick_color, width = wick_width)
