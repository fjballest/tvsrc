eqcol(color color1, color color2) =>
    r1 = color.r(color1)
    g1 = color.g(color1)
    b1 = color.b(color1)
    a1 = color.t(color1) 
    
    r2 = color.r(color2)
    g2 = color.g(color2)
    b2 = color.b(color2)
    a2 = color.t(color2)
    
    r1 == r2 and g1 == g2 and b1 == b2 and math.abs(a1-a2) <= 10

istransp(color col) =>
    color.t(col) >= 100

// given a name like H4, return a timeframe str
tfstr(string tf) =>
    s = tf 
    if tf == "m15"
        s := "15"
    else if tf == "H1"
        s := "60"
    else if tf == "H4"
        s := "240"
    else if tf == "D" or tf == "W" or tf == "M"
        s := tf
    s




tfOK(string tf) =>
    tfsecs < timeframe.in_seconds(tfstr(tf))

sLongFVG(l1,h3) =>
    l1 > h3
longFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l1
    lo = h3
    blow1 = math.min(o1,c1)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi3 = math.max(o3,c3)
    // include suspension blocks
    fhi = hi
    flo = lo
    if bhi3 < blow2
        lo := bhi3
    if blow1 > bhi2
        hi := blow1
    [hi,lo, fhi, flo]

isShortFVG(h1,l3) =>
    h1 < l3
shortFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l3
    lo = h1
    blow3 = math.min(o3,c3)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi1 = math.max(o1,c1)
    // include suspension blocks
    fhi = hi
    flo = lo
    if blow3 > bhi2
        hi := blow3
    if bhi1 < blow2
        lo := bhi1
    [hi, lo, fhi, flo]

method kill(FVG f) =>
    if not na(f) and not na(f.rect)
        f.rect.delete()
        f.rect := na
    if not na(f) and not na(f.celine)
        f.celine.delete()
        f.celine := na

method setup(FVG f) =>
    f.col := f.islong ? lcolfn(f.tf) : scolfn(f.tf)
    f.bcol := f.islong ? lcolbfn(f.tf) : scolbfn(f.tf)
    f.cecol := f.bcol
    f.bwid := borderfn(f.tf)
    f.txcol := f.bcol
    if eqcol(f.bcol, f.col) or istransp(f.bcol)
        f.txcol := labelcol
    if f.bwid == 0
        f.bcol := transparent


timeoff(n) =>
    time_close + tfsecs * n

method update(FVG f) =>
    if f.islong and high < f.lo 
        f.done := true
        f.end := timeoff(0)
    else if not f.islong and low > f.hi 
        f.done := true
        f.end := timeoff(0)
    if na(f.rect)
        f.rect := box.new(f.start, f.hi, timeoff(2), f.lo, f.bcol, f.bwid, bordersty,
            xloc = xloc.bar_time,
            bgcolor = nobox?transparent:f.col, text = dolabel ?f.tf:"", text_size = tsz,
            text_color = f.txcol,
            text_halign = txalgn)
    else if not f.done
        f.rect.set_right(timeoff(2))
    if na(f.celine) and doce
        f.celine := line.new(f.start, f.ce, timeoff(2), f.ce, 
            xloc = xloc.bar_time, color = f.cecol, style=cesty,width =cewidth<0?f.bwid:cewidth)
    else if doce and not f.done
        f.celine.set_x2(timeoff(2))
    f.done

getbars(string tf) => 
    x = tfstr(tf)
    request.security(syminfo.tickerid, x,
        [open[1],close[1],high[1],low[1],open[2],close[2],high[2],low[2],open[3],close[3],high[3],low[3], time[2]],
        lookahead = barmerge.lookahead_on)


isdup(FVG[] fvgs, float hi, float lo) =>
    d = false
    for f in fvgs
        if f.hi == hi and f.lo == lo
            d := true
            break
    d


mkFVGS(string tf, FVG[] fvgs) =>
    [o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3,t2] = getbars(tf)
    [h, l, fh, fl] = longFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3)
    FVG f = na
    if fh > fl+5 and not isdup(fvgs, h, l)
        f := FVG.new(tf, true, t2, na, h, l, math.avg(h,l))
    if na(f)
        [sh,sl, fsh, fsl] = shortFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3)
        if fsh > fsl+5 and not isdup(fvgs, sh, sl)
            f := FVG.new(tf, false, t2, na, sh, sl, math.avg(sh,sl))
        // short FVG found
    if not na(f)
        f.setup()
        fvgs.push(f)
    if fvgs.size() > nfvgs
        x = fvgs.shift()
        x.kill()
        x := na
