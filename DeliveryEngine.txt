deleteLiq(Liq l) =>
    l.ln.delete()
    if not na(l.lb)
        l.lb.delete()

ageLevel(string oldName, string newName) =>
    if liqs.size() > 0
        for l in liqs
            parts = str.split(l.kind, " · ")
            bool ch = false
            for i = 0 to parts.size() - 1
                if parts.get(i) == oldName
                    parts.set(i, newName)
                    ch := true
            if ch
                l.kind := parts.join(" · ")
                if not na(l.lb)
                    l.lb.set_text(l.kind)
    true

liqPrio(string k) =>
    str.startswith(k, "PW") ? 7 : str.startswith(k, "PD") ? 6 : str.startswith(k, "ERL") ? 5 :
     k == "EQH" or k == "EQL" ? 3 : k == "REH" or k == "REL" ? 2 : k == "BSL" or k == "SSL" ? 1 : 4

newLiq(int x1, float price, bool isHigh, string kind, color col, string sty, int wid) =>
    xs  = math.max(x1, bar_index - 4999)
    ln  = line.new(xs, price, bar_index + 2, price, color = col, style = sty, width = wid)
    label lb = liq_txt ? label.new(bar_index + 2, price, kind, style = label.style_label_left, color = color.new(color.black, 100), textcolor = col, size = size.tiny) : na
    liqs.push(Liq.new(ln, lb, price, isHigh, x1, kind))
    if liqs.size() > max_liq
        int worst = 0
        int wPrio = liqPrio(liqs.get(0).kind)
        for i = 1 to liqs.size() - 2
            pr = liqPrio(liqs.get(i).kind)
            if pr < wPrio
                wPrio := pr
                worst := i
        deleteLiq(liqs.get(worst))
        liqs.remove(worst)
    true

addLiq(int x1, float price, bool isHigh, string kind, color col, string sty, int wid) =>
    tol = nz(gAtr) * merge_tol
    int idx = -1
    if liqs.size() > 0
        for i = liqs.size() - 1 to 0
            o = liqs.get(i)
            if o.isHigh == isHigh and math.abs(o.price - price) <= tol
                idx := i
                break
    if idx < 0
        newLiq(x1, price, isHigh, kind, col, sty, wid)
    else
        o  = liqs.get(idx)
        pn = liqPrio(kind)
        po = liqPrio(o.kind)
        if pn == 1 or po == 1 or (pn <= 3 and po <= 3)
            if pn > po
                deleteLiq(o)
                liqs.remove(idx)
                newLiq(x1, price, isHigh, kind, col, sty, wid)
        else if not str.contains(o.kind, kind)
            txt = pn > po ? kind + " · " + o.kind : o.kind + " · " + kind
            ext = isHigh ? math.max(o.price, price) : math.min(o.price, price)
            o.kind  := txt
            o.price := ext
            o.ln.set_y1(ext)
            o.ln.set_y2(ext)
            if pn > po
                o.ln.set_color(col)
                o.ln.set_style(sty)
                o.ln.set_width(wid)
            if not na(o.lb)
                o.lb.set_text(txt)
                o.lb.set_y(ext)
                if pn > po
                    o.lb.set_textcolor(col)
    idx

findEqual(array<Piv> arr, float p, int x, bool isHigh, float atrv) =>
    string kind = ""
    int    qx   = na
    float  qp   = na
    if arr.size() > 0
        for j = arr.size() - 1 to 0
            q = arr.get(j)
            if x - q.x > eqBars
                break
            diff = math.abs(p - q.price)
            lvl  = isHigh ? math.max(p, q.price) : math.min(p, q.price)
            clean = true
            k0 = bar_index - x + 1
            k1 = bar_index - q.x - 1
            if k1 >= k0
                for k = k0 to k1
                    if isHigh ? high[k] > lvl : low[k] < lvl
                        clean := false
                        break
            if clean
                if diff <= eq_tol * atrv
                    if (isHigh ? show_eqh : show_eql)
                        kind := isHigh ? "EQH" : "EQL"
                else if diff <= re_tol * atrv and (isHigh ? show_reh : show_rel)
                    kind := isHigh ? "REH" : "REL"
            if kind != ""
                qx := q.x
                qp := q.price
                break
    [kind, qx, qp]

killzone(HL k, bool show, string sess, string name, color col) =>
    inS = kzActive and not na(time("", sess, kz_tz))
    st  = inS and not inS[1]
    en  = not inS and inS[1]
    if show
        if st
            sfx = k.dt != "" ? k.dt : "-1"
            ageLevel(name + " H", name + " " + sfx + " H")
            ageLevel(name + " L", name + " " + sfx + " L")
            k.dt := dateTxt()
            k.h  := high
            k.l  := low
            k.hx := bar_index
            k.lx := bar_index
            k.bx := na
            if histOK
                k.bx := box.new(bar_index, high, bar_index, low, border_color = color.new(col, 60), bgcolor = col, text = kz_txt ? name : "", text_color = color.new(col, 20), text_size = size.tiny, text_halign = text.align_left, text_valign = text.align_top)
                kzBoxes.push(k.bx)
                if kzBoxes.size() > kz_max
                    kzBoxes.shift().delete()
        else if inS and not na(k.h)
            if high > k.h
                k.h  := high
                k.hx := bar_index
            if low < k.l
                k.l  := low
                k.lx := bar_index
            if not na(k.bx)
                k.bx.set_top(k.h)
                k.bx.set_bottom(k.l)
                k.bx.set_right(bar_index)
        if en and kz_liq and not na(k.h)
            lc = color.new(col, 0)
            addLiq(k.hx, k.h, true,  name + " H", lc, line.style_solid, 1)
            addLiq(k.lx, k.l, false, name + " L", lc, line.style_solid, 1)
    show and inS

structDol(int len) =>
    var float sh  = na
    var float sl  = na
    var bool  shB = true
    var bool  slB = true
    var int   d   = 0
    var int   evt = 0
    var int   ed  = 0
    var array<float> hiLv = array.new<float>()
    var array<float> loLv = array.new<float>()
    var array<float> eqHi = array.new<float>()
    var array<float> eqLo = array.new<float>()
    var int   shI = na
    var int   slI = na
    var float oS  = na
    var float oE  = na
    var int   oD  = 0
    var bool  oL  = false
    var int   oBn = na
    bool mUp = false
    bool mDn = false
    var float upO = na
    var int   upB = na
    var float dnO = na
    var int   dnB = na
    var float rU  = na
    var int   rUB = na
    var float rUH = na
    var float rD  = na
    var int   rDB = na
    var float rDL = na
    atrT = ta.atr(14)
    cr  = disp_type == "Open to Close" ? math.abs(close - open) : high - low
    sdT = ta.stdev(cr, std_len)
    sdm = sdT * mss_sx
    upD = nz(ta.barssince(cr > sdm and close > open), 999) <= mss_win
    dnD = nz(ta.barssince(cr > sdm and close < open), 999) <= mss_win
    lHi = ta.highest(high, cisd_look)
    lLo = ta.lowest(low, cisd_look)
    if not shB and not na(sh) and close > sh
        shB := true
        evt := d != 1 ? (upD or not mss_disp ? 2 : 3) : 1
        d   := 1
        ed  := 1
        mUp := evt == 2
    if not slB and not na(sl) and close < sl
        slB := true
        evt := d != -1 ? (dnD or not mss_disp ? 2 : 3) : 1
        d   := -1
        ed  := -1
        mDn := evt == 2
    if mUp or mDn
        int nL = math.min(bar_index - nz(mUp ? shI : slI, bar_index), 200)
        float stL = mUp ? low : high
        for i = 0 to nL
            stL := mUp ? math.min(stL, low[i]) : math.max(stL, high[i])
        oS  := stL
        oE  := mUp ? high : low
        oD  := mUp ? 1 : -1
        oL  := false
        oBn := bar_index
    else if oD != 0
        if (oD == 1 and close < oS) or (oD == -1 and close > oS)
            oD := 0
        else if not oL and bar_index > oBn
            if (oD == 1 and high > oE) or (oD == -1 and low < oE)
                oE := oD == 1 ? high : low
            l62 = oD == 1 ? oE - 0.62 * (oE - oS) : oE + 0.62 * (oS - oE)
            if (oD == 1 and low <= l62) or (oD == -1 and high >= l62)
                oL := true
    bool hiSw = false
    bool loSw = false
    if hiLv.size() > 0
        for i = hiLv.size() - 1 to 0
            if high > hiLv.get(i)
                hiLv.remove(i)
                hiSw := true
    if loLv.size() > 0
        for i = loLv.size() - 1 to 0
            if low < loLv.get(i)
                loLv.remove(i)
                loSw := true
    if eqHi.size() > 0
        for i = eqHi.size() - 1 to 0
            if high > eqHi.get(i)
                eqHi.remove(i)
    if eqLo.size() > 0
        for i = eqLo.size() - 1 to 0
            if low < eqLo.get(i)
                eqLo.remove(i)
    isUp = close > open
    isDn = close < open
    if isUp and not (close[1] > open[1])
        upO := open
    if isDn and not (close[1] < open[1])
        dnO := open
    if isUp
        upB := bar_index
    if isDn
        dnB := bar_index
    tU = cisd_sweep ? hiSw : high >= lHi
    tD = cisd_sweep ? loSw : low <= lLo
    if tU and not na(upB) and bar_index - upB <= 2
        rU  := upO
        rUB := bar_index
        rUH := high
    if tD and not na(dnB) and bar_index - dnB <= 2
        rD  := dnO
        rDB := bar_index
        rDL := low
    if not na(rU)
        rUH := math.max(rUH, high)
        if bar_index - rUB > cisdBars
            rU := na
    if not na(rD)
        rDL := math.min(rDL, low)
        if bar_index - rDB > cisdBars
            rD := na
    if not na(rU) and close < rU
        okB = switch cisd_str
            "Cualquier cierre" => true
            "Cuerpo ≥ 50%"     => isDn and (open - close) >= 0.5 * (high - low)
            => isDn and cr > sdT * std_x
        rngU = rUH - rU >= cisd_rng * atrT
        if rngU and okB
            evt := 4
            ed  := -1
            rU  := na
        else if not rngU
            rU := na
    if not na(rD) and close > rD
        okB = switch cisd_str
            "Cualquier cierre" => true
            "Cuerpo ≥ 50%"     => isUp and (close - open) >= 0.5 * (high - low)
            => isUp and cr > sdT * std_x
        rngD = rD - rDL >= cisd_rng * atrT
        if rngD and okB
            evt := 4
            ed  := 1
            rD  := na
        else if not rngD
            rD := na
    ph = ta.pivothigh(high, len, len)
    pl = ta.pivotlow(low, len, len)
    if not na(ph)
        sh  := ph
        shI := bar_index - len
        shB := false
        hiLv.push(ph)
        if hiLv.size() > 40
            hiLv.shift()
        if hiLv.size() > 1
            for i = 0 to hiLv.size() - 2
                if math.abs(hiLv.get(i) - ph) <= eq_tol * atrT
                    float qv = math.max(hiLv.get(i), ph)
                    bool dupQ = false
                    for v in eqHi
                        if math.abs(v - qv) <= eq_tol * atrT
                            dupQ := true
                    if not dupQ
                        eqHi.push(qv)
                        if eqHi.size() > 20
                            eqHi.shift()
    if not na(pl)
        sl  := pl
        slI := bar_index - len
        slB := false
        loLv.push(pl)
        if loLv.size() > 40
            loLv.shift()
        if loLv.size() > 1
            for i = 0 to loLv.size() - 2
                if math.abs(loLv.get(i) - pl) <= eq_tol * atrT
                    float qv = math.min(loLv.get(i), pl)
                    bool dupQ = false
                    for v in eqLo
                        if math.abs(v - qv) <= eq_tol * atrT
                            dupQ := true
                    if not dupQ
                        eqLo.push(qv)
                        if eqLo.size() > 20
                            eqLo.shift()
    dd = dol_dir == "Estructura" ? d : (ed != 0 ? ed : d)
    float dol = na
    if dd == 1
        for v in hiLv
            if v > close and (na(dol) or v < dol)
                dol := v
    else if dd == -1
        for v in loLv
            if v < close and (na(dol) or v > dol)
                dol := v
    float eqUp = na
    for v in eqHi
        if v > close and (na(eqUp) or v < eqUp)
            eqUp := v
    float eqDn = na
    for v in eqLo
        if v < close and (na(eqDn) or v > eqDn)
            eqDn := v
    [d, evt, ed, dol, eqUp, eqDn, oS, oE, oD, oL]

dolPick(array<float> outPx, array<int> outDir, array<string> outTxt, array<string> outKey, array<bool> outIrl) =>
    wantDir = dol_dir == "Entrega" ? delivDir : dol_dir == "Estructura" ? dir : 0
    useCE   = dol_fvg_lvl == "CE (50%)"
    useIrl  = dol_irl or dol_sel == "Ciclo ERL ↔ IRL"
    array<Cand> cands = array.new<Cand>()
    for l in liqs
        cands.push(Cand.new(l.price, l.isHigh ? 1 : -1, l.kind, false))
    if useIrl
        for f in zones
            if f.inv == 0 and not f.touched and (f.btm > close or f.top < close)
                up = f.btm > close
                cands.push(Cand.new(useCE ? (f.top + f.btm) / 2 : up ? f.btm : f.top, up ? 1 : -1, "FVG★", true))
        for h in hfvgs
            if dol_hfvg and not h.reached and (h.btm > close or h.top < close)
                up = h.btm > close
                cands.push(Cand.new(useCE ? (h.top + h.btm) / 2 : up ? h.btm : h.top, up ? 1 : -1, (h.inv ? "IFVG " : "FVG ") + h.tf, true))
        for g in gaps
            if not g.reached and (g.btm > close or g.top < close)
                up = g.btm > close
                cands.push(Cand.new(useCE ? (g.top + g.btm) / 2 : up ? g.btm : g.top, up ? 1 : -1, g.kind, true))

    if dol_hteq
        addEq(cands, htf1_on, htf1, eh1, el1)
        addEq(cands, htf2_on, htf2, eh2, el2)
        addEq(cands, htf3_on, htf3, eh3, el3)
        addEq(cands, htf4_on, htf4, eh4, el4)
        addEq(cands, htf5_on, htf5, eh5, el5)

    array<Cand> pool = array.new<Cand>()
    for c in cands
        dA = math.abs(c.price - close) / atr
        if dA >= dol_min and (dol_max == 0 or dA <= dol_max) and (dol_dir == "Libre" or c.d == wantDir)
            pool.push(c)

    outPx.clear()
    outDir.clear()
    outTxt.clear()
    outKey.clear()
    outIrl.clear()
    nOut = 1 + dol_ladder

    if pool.size() > 0
        if dol_sel == "Más cercano"
            array<float> dists = array.new<float>()
            for c in pool
                dists.push(math.abs(c.price - close))
            idxN = dists.sort_indices(order.ascending)
            for j in idxN
                if outPx.size() >= nOut
                    break
                c = pool.get(j)
                outPx.push(c.price)
                outDir.push(c.d)
                outTxt.push(c.kind)
                outIrl.push(c.irl)
                outKey.push(dolKey(c.d, c.price))

        else
            lastErl = nz(lastExtSweep, -1)
            wantIrl = lastErl >= 0 and lastErl > nz(lastIrl, -1)
            array<float> dists = array.new<float>()
            for c in pool
                dists.push(math.abs(c.price - close))
            idx = dists.sort_indices(order.ascending)
            for pass = 0 to 1
                fam = pass == 0 ? wantIrl : not wantIrl
                for j in idx
                    if outPx.size() >= nOut
                        break
                    c = pool.get(j)
