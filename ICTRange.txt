// Definition of a range
type  RangeInfo
    string tz // eg UTC-4
    int starth  // start hour
    int startm  // min
    int endh    // end hour
    int endm    // min
    Sty sty = na // style


// Instance of a range
type Range
    int startidx = na //  barindex when starts
    int endidx = na // end done
    float hi 
    float lo
    int hiidx = na
    int loidx = na
    int extendidx = na // end for lines that extend another zone
    box rect = na // rectangle drawn if any
    array<line> qs =  na // array of cuadrants and PO3s
    bool drawrect = false
    RangeInfo ri = na // for  range instances
    Sty sty = na// line style
    float minprj = na   // price for min stddev
    float maxprj = na   // price for max stddev
    float wid = na // sep between lines
    line vline0 = na
    line vline1 = na
     



hmParse(string s) =>
    hs = str.substring(s, 0, 2)
    ms = str.substring(s, 2, 4)
    [int(str.tonumber(hs)), int(str.tonumber(ms))]

sesParse(string s) =>
    vals = str.split(s, '-')
    st = vals.size() > 0 ? vals.get(0) : "0000"
    end = vals.size() > 1 ? vals.get(1) : st
    [sth, stm] = hmParse(st)
    [endh, endm] = hmParse(end)
    [sth, stm, endh, endm]

mkRangeInfo(string ival, string tz, Sty sty = na) =>
    RangeInfo zi = na
    [sth, stm, eh, em] = sesParse(ival)
    zi := RangeInfo.new(tz, sth, stm, eh, em)
    zi.sty := sty
    zi

method starts(RangeInfo zn) =>
    hour(time,zn.tz) == zn.starth and minute(time,zn.tz) == zn.startm
 
method starting(RangeInfo zn) =>
    h = hour(time,zn.tz)
    m = minute(time,zn.tz)
    hm = h*60+m
    sthm = zn.starth*60 + zn.startm
    hm + 15 >= sthm and hm <= sthm

method ends(RangeInfo zn) =>
    hour(time,zn.tz) == zn.endh and minute(time,zn.tz) == zn.endm


mkRange(RangeInfo zi = na) =>
    zz = Range.new(bar_index, na, high, low, bar_index, bar_index)
    zz.ri := zi
    zz

method mkline(Range zn, float p, bool isx = false) =>
    i = (isx?zn.endidx:zn.startidx)
    ln = line.new(i, p, last_bar_index, p, color = zn.ri.sty.col,
        extend = extend.right,
        width = zn.ri.sty.wid, style=zn.ri.sty.sty)
    zn.qs.push(ln)
   
method mklines(Range zn) =>
    zn.mkline(zn.hi)
    zn.mkline(zn.lo)
    rg = math.abs(zn.hi-zn.lo)
    zn.wid := rg/nq
    xnq = nq
    while zn.wid < 30 and xnq > 1
        xnq -= 1
        zn.wid := rg/xnq
    p = zn.lo
    zn.minprj := zn.lo
    zn.maxprj := zn.hi
    for i = 0 to nq-1
        p += zn.wid 
        zn.mkline(p)
    p := zn.hi
    for i = 0 to po3up-1
        if po3up <= 0
            break
        p += zn.wid 
        zn.mkline(p, isx=true)
        zn.maxprj := p
    for i = 1 to po3down-1
        if po3down <= 0
            break
        p -= zn.wid 
        zn.mkline(p, isx = true)
        zn.minprj := p



    
method update(Range zn, bool islast = false) =>
    if na(zn.endidx)
        if high >= zn.hi 
            zn.hiidx := bar_index
            zn.hi := high
        if low <= zn.lo
            zn.loidx := bar_index
            zn.lo := low
    if not na(zn.extendidx) and not na(zn.qs)
        for q in zn.qs
            q.set_extend(extend.none)
            q.set_x2(zn.extendidx)
    if not na(zn.endidx) and na(zn.extendidx) and auto
        if low < zn.minprj
            zn.minprj := zn.minprj - zn.wid
            zn.mkline(zn.minprj, isx = true)
        if high > zn.maxprj
            zn.maxprj := zn.maxprj + zn.wid
            zn.mkline(zn.maxprj, isx = true)
    true
method ended(Range zn) =>
    if na(zn.endidx)
        zn.endidx := bar_index-1
    if last_bar_index - zn.startidx < 5000
        if na(zn.rect) and dorect
            zn.rect := box.new(zn.startidx, zn.lo, zn.endidx, zn.hi,  border_color = zn.ri.sty.col, bgcolor = zn.ri.sty.col2)
        if na(zn.qs)
            zn.qs := array.new<line>()
        if zn.qs.size() == 0
            zn.mklines()
        if na(zn.vline0)
            zn.vline0 := line.new(zn.startidx, high, zn.startidx, low, color = vlnsty.col,
                extend = extend.both,
                width = vlnsty.wid, style=vlnsty.sty)
        if na(zn.vline1)
            zn.vline1 := line.new(zn.endidx+1, high, zn.endidx+1, low, color = vlnsty.col,
                extend = extend.both,
                width = vlnsty.wid, style=vlnsty.sty)
 
method limit(Range zn, int limit) => 
    if na(zn.extendidx)
        zn.extendidx := limit
        if not na(zn.qs)
            for q in zn.qs
                q.set_extend(extend.none)
                q.set_x2(zn.extendidx)




method kill(Range zn) =>
    if not na(zn.rect)
        zn.rect.delete()
        zn.rect := na
    if not na(zn.qs)
        while zn.qs.size() > 0
            line.delete(zn.qs.shift())
        zn.qs.clear()
        zn.qs := na
    if not na(zn.vline0)
        zn.vline0.delete()
        zn.vline0 := na
    if not na(zn.vline1)
        zn.vline1.delete()
        zn.vline1 := na

limitolds(array<Range> ranges) =>
    Range rg = ranges.size() > 1 ? ranges.get(ranges.size()-1) : na
    if not na(rg) and not na(rg.endidx)
        for i = 0 to ranges.size() - 2
            x = ranges.get(i)
            x.limit(bar_index-1)
