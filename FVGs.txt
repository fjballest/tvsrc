// FVG data type, one per handled fvg.
type FVG
    float hi // max value
    float lo // min value
    float ce // C.E.
    bool islong // bullish?
    int start  // time for start
    int end = na// time when mitigated
    box rect = na // box drawn
    line celine = na // CE line
    bool isfp = false // is first presented FVG
    bool iswfp = false // is week first presented FVG
    bool isifp = false // is first presented FVG inverse


method kill(FVG f) =>
    if not na(f)
        if not na(f.rect)
            f.rect.delete()
        if not na(f.celine)
            f.celine.delete()
        f.rect := na
        f.celine := na
method alerts(FVG f) =>
    a = false
    if not na(f.hi) and bounces(f.hi)
        alert("bounce on FVG hi", alert.freq_once_per_bar)
        a := true
    else if not na(f.lo) and bounces(f.lo)
        alert("bounce on fvg lo", alert.freq_once_per_bar)
        a := true
    else if not na(f.ce) and bounces(f.ce)
        alert("bounce on CE", alert.freq_once_per_bar)
        a := true
    if a and debug
        line.new(bar_index,low,bar_index,high,extend = extend.both)
    a

clearlastfp(array<FVG> fpfvgs) =>
    while fpfvgs.size() > nfp
        f = fpfvgs.shift()
        if not na(f) and f.isfp
            if not na(f.rect)
                f.rect.set_extend(extend.none)
            if not na(f.celine)
                f.celine.set_extend(extend.none)
            f.isfp := false
            f.kill()

isLongFVG(l1,h3) =>
    l1 > h3
longFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l1
    lo = h3
    blow1 = math.min(o1,c1)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi3 = math.max(o3,c3)
    // include suspension blocks
    fhi = hi
    flo = lo
    if bhi3 < blow2
        lo := bhi3
    if blow1 > bhi2
        hi := blow1
    [hi,lo, fhi, flo]

isShortFVG(h1,l3) =>
    h1 < l3
shortFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l3
    lo = h1
    blow3 = math.min(o3,c3)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi1 = math.max(o1,c1)
    // include suspension blocks
    fhi = hi
    flo = lo
    if blow3 > bhi2
        hi := blow3
    if bhi1 < blow2
        lo := bhi1
    [hi, lo, fhi, flo]

isFPFVG() =>
    r = dofp
    if r
        h = hour(time,tz)
        m = minute(time,tz)
        nmin = 2
        if timeframe.isminutes
            nmin := tfsecs/60 * 2
        r := h == fph and m > fpm+nmin and timeframe.isminutes and tfsecs/60 <=3
    r
if bar_index == 0
    log.info("fpfvg h {0} fpfvgm {1}", fph, fpm)

eqFVG(a,b) =>
    a.hi == b.hi and a.lo == b.lo and a.idx == b.idx

method endsAt(FVG f) =>
    na(f.end) ? time : f.end

method isDone(FVG f, o, c) =>
    f.islong ? (c < f.lo or o < f.lo) : (c>f.hi or o > f.hi)

method bgColor(FVG f) =>
    c = f.islong ? lcol : scol
    if f.isfp
        c := fpcol
        if f.iswfp
            c := wfpcol
    if f.isifp
        c := ifpcol
    c
    

method lnColor(FVG f) =>
    c = f.islong ? lnlcol : lnscol
    if f.isfp
        c := lnfpcol
        if f.iswfp
            c := lnwfpcol
    if f.isifp
        c := ilnfpcol
    c

timeat(int n) =>
    time[-n]
FVG lastlongfvg = na
FVG lastshortfvg = na
if true
    [hi, lo, fhi, flo] = longFVGHL(open[1],close[1],high[1],low[1],
         open[2],close[2],high[2],low[2],
         open[3],close[3],high[3],low[3])
    if fhi - flo > minfvg and fhi-flo < maxfvg
        f = FVG.new(hi,lo,math.avg(hi,lo),true,timeat(-3))
        lastlongfvg := f
        if not hasifp and hasfp and doifp
            fp = fpfvgs.last()
            if fp.islong != f.islong
                f.isifp := true
                hasifp := true
                fpfvgs.push(f)
        if not hasfp and isFPFVG() and not f.isifp
            f.isfp := true
            hasfp := f.isfp
            fpfvgs.push(f)
            clearlastfp(fpfvgs)
        else if not fponly and not f.isifp
            lfvgs.push(f)
        else
            f.kill()
if true
    [hi, lo, fhi, flo] = shortFVGHL(open[1],close[1],high[1],low[1],
         open[2],close[2],high[2],low[2],
         open[3],close[3],high[3],low[3])
    if fhi - flo > minfvg and fhi - flo < maxfvg
        f = FVG.new(hi,lo,math.avg(hi,lo),false,timeat(-3))
        lastshortfvg := f
        if not hasifp and hasfp and doifp
            fp = fpfvgs.last()
            if fp.islong != f.islong
                f.isifp := true
                hasifp := true
                fpfvgs.push(f)
        if not hasfp and isFPFVG() and not f.isifp
            f.isfp := true
            f.iswfp := dayofweek == dayofweek.monday
            hasfp := f.isfp
            fpfvgs.push(f)
            clearlastfp(fpfvgs)
        else if not fponly and not  f.isifp
            sfvgs.push(f)
        else 
            f.kill()

updfvgs(array<FVG> fvgs) =>
    if fvgs.size() > 0
        for i = 0 to fvgs.size()-1
            f = fvgs.get(i)
            isdone = not na(f.end)
            // check if filled
            if not isdone and f.isDone(open,close)
                f.end := f.endsAt()
            et = f.endsAt()
            if na(f.rect)
                f.rect := box.new(f.start, f.hi, et, f.lo,
                    bgcolor = f.bgColor(),
                    border_color = f.lnColor(),
                    border_width = 1,
                    xloc = xloc.bar_time,
                    extend = (f.isifp or f.isfp) ? extend.right : extend.none,
                    border_style = line.style_solid)
            if na(f.celine) and f.hi-f.lo > mince
                f.celine := line.new(f.start,f.ce,et,f.ce,
                    color = f.lnColor(),
                    style = cesty,
                    xloc = xloc.bar_time,
                    width = celnwid)
                if f.isfp
                    f.celine.set_extend(extend.right)
            if na(f.end)
                f.rect.set_right(et)
                if not na(f.celine)
                    f.celine.set_x2(et)
            if dofvgalert and na(f.end)
                f.alerts()
        if fvgs.size() >= maxfvgs  and fvgs.size() > 0
            df = fvgs.shift()
            df.kill()

if not visible and barstate.isnew

findFVG(int sTime, array<FVG> fvgs) =>
    var FVG rf = na
    int sz = array.size(fvgs)
    if sz > 0
        for i = sz - 1 to 0
            rf := fvgs.get(i)
            if rf.start < sTime
                break
    rf

tfstr() =>
    string tf = timeframe.period
    if timeframe.isseconds
        str.replace(tf, "S", "s")
    else if timeframe.isminutes
        if tf == "60"
            "1h"
        else if tf == "120"
            "2h"
        else if tf == "180"
            "3h"
        else if tf == "240"
            "4h"
        else
            tf + "m"
    else if timeframe.isdaily
        // If it's exactly 1 day ("1D"), UI just shows "D"
        tf == "1D" ? "D" : str.replace(tf, "D", "D") 
    else if timeframe.isweekly
        tf == "1W" ? "W" : str.replace(tf, "W", "W")
    else if timeframe.ismonthly
        tf == "1M" ? "M" : str.replace(tf, "M", "M")
    else
        tf

    if sig > 0
        alert("LONG IFVG "+tfstr(), alert.freq_once_per_bar)
    else
        alert("SHORT IFVG "+tfstr(), alert.freq_once_per_bar)
