
// ============================================================ OB mean threshold (15m, experimental)
// Added to ICTblocks by tools/mkfvgderiv.py from tools/ictblocks-ob.txt (not part of FVGS+IFVG+OTE.txt).
// [OB] a run of 15m candles closing the same way is an order block zone (bodies or full candles); a candle of the
// other color whose body stays inside the zone does not end the run, but a second one in a row does (the move away
// starts there, and the next run starts with both). A candle that would make the zone taller than obMaxH 15m ATRs
// also ends it. Up-close runs are resistance OBs, armed (dir -1) by a close below the zone and disarmed by a close
// above its MT; down-close runs are support OBs, armed (dir 1) by a close above the zone and disarmed by a close
// below its MT. A disarmed OB is armed again by the next close out of its far side; once armed, a close beyond its
// far end (above a resistance OB, below a support OB) discards it, and so does age (obLook). disp: how far price
// went away from it since it was armed (its significance). Only on charts of 15m or less; 15m candles come from
// request.security (completed candles only, so nothing repaints).
const string G_OB = "OB mean threshold (15m, experimental)"
bool doOb = input.bool(true, "OB MT levels", group = G_OB, inline = "o1",
     tooltip = "15m order blocks (charts of 15m or less only). A run of 15m candles closing the same way is an OB; "
     + "one candle of the other color whose body stays inside the zone does not end it, two in a row do. Up-close "
     + "OBs are resistance: once price closes below one, its MT (the middle) is drawn above price for the retrace "
     + "back up, until a close above the MT. Down-close OBs are support the same way. A close beyond the far end "
     + "of a used OB discards it, and so does being older than the lookback. Of each side, the OB covering price "
     + "is drawn (the deepest move away from it, if several), else the nearest one.")
string obPart = input.string("Body", "zone", options = ["Body", "Full candle"], group = G_OB, inline = "o1")
int obLook = input.int(300, "lookback (15m candles)", minval = 20, maxval = 2000, group = G_OB, inline = "o2")
bool obZone = input.bool(false, "draw the zone", group = G_OB, inline = "o2")
float obMaxH = input.float(4.0, "max height (15m ATRs)", minval = 0.5, step = 0.25, group = G_OB, inline = "o4",
     tooltip = "Height cap: a candle that would make the zone taller than this many 15m ATRs ends the run (and starts the next one), so a rally of same-color candles is not one huge OB; the consolidation before it stays its own OB.")
int obXoff = input.int(2, "label offset (bars)", minval = 0, group = G_OB, inline = "o4")
color obResCol = input.color(color.red, "resistance", group = G_OB, inline = "o3")
color obSupCol = input.color(color.green, "support", group = G_OB, inline = "o3")
string obSty = getstyle(input.string("dashed", "", options = ["solid", "dashed", "dotted"], group = G_OB,
     inline = "o3"))
int obWid = input.int(1, "width", minval = 1, maxval = 4, group = G_OB, inline = "o3")

type Ob
    float hi
    float lo
    int t
    int k        // 15m candle count at its end
    int sg = 0   // 1 up-close candles (resistance OB), -1 down-close candles (support OB)
    int dir = 0  // -1 armed resistance, 1 armed support, 0 not armed (not left yet, or closed beyond its MT)
    float ext = na
    float disp = 0.0
    bool used = false  // armed at least once

type ObRun
    int sg = 0   // 1 up-close run, -1 down-close run, 0 none
    float hi = na
    float lo = na
    int t = na
    int n = 0
    int k = 0    // 15m candles seen
    float atr = na   // 15m ATR(14) (RMA of the true range), for the height cap
    float pc = na    // previous 15m close
    int op = 0       // 1: the last candle was an inside candle of the other color
    float oh = na    // its zone and time, for the next run when a second one follows
    float ol = na
    int ot = na

var array<Ob> obs = array.new<Ob>()
var ObRun obRun = ObRun.new()
var int obLastT = na
var line obResLn = na
var line obSupLn = na
var label obResLb = na
var label obSupLb = na
var box obResBx = na
var box obSupBx = na

obStep(float o, float h, float l, float c, int t) =>
    ObRun r = obRun
    r.k += 1
    int obk = r.k
    float zh = obPart == "Body" ? math.max(o, c) : h
    float zl = obPart == "Body" ? math.min(o, c) : l
    int sg = c > o ? 1 : c < o ? -1 : 0
    // one candle of the other color inside the zone stays in the run; a second one in a row is the move away
    bool inside = r.sg != 0 and r.op == 0 and math.max(o, c) <= r.hi and math.min(o, c) >= r.lo and r.n < 40
    // height cap: a candle that would make the zone taller than obMaxH 15m ATRs starts a new run
    bool fits = na(r.atr) or math.max(r.hi, zh) - math.min(r.lo, zl) <= obMaxH * r.atr
    float tr = na(r.pc) ? h - l : math.max(h - l, math.abs(h - r.pc), math.abs(l - r.pc))
    r.atr := na(r.atr) ? tr : (r.atr * 13 + tr) / 14
    r.pc := c
    if r.sg != 0 and (sg == r.sg or sg == 0 or inside) and fits
        if sg == -r.sg
            r.op := 1
            r.oh := zh
            r.ol := zl
            r.ot := t
        else
            r.hi := math.max(r.hi, zh)
            r.lo := math.min(r.lo, zl)
            if sg == r.sg
                r.op := 0
        r.n += 1
    else
        if r.sg != 0
            obs.push(Ob.new(r.hi, r.lo, r.t, obk - 1, r.sg))
        bool two = r.op == 1 and sg != 0 and sg == -r.sg
        r.hi := two ? math.max(r.oh, zh) : zh
        r.lo := two ? math.min(r.ol, zl) : zl
        r.t := two ? r.ot : t
        r.n := two ? 2 : 1
        r.sg := sg
        r.op := 0
    int i = obs.size() - 1
    while i >= 0
        Ob b = obs.get(i)
        float mt = (b.hi + b.lo) / 2
        // once used as a level, a close beyond its far end (back through the whole OB) discards it
        bool gone = b.used and (b.sg == 1 ? c > b.hi : c < b.lo)
        if b.sg == 1
            if c < b.lo and b.dir != -1
                b.dir := -1
                b.used := true
                b.ext := l
            else if b.dir == -1 and c > mt
                b.dir := 0
            else if b.dir == -1
                b.ext := math.min(b.ext, l)
        else
            if c > b.hi and b.dir != 1
                b.dir := 1
                b.used := true
                b.ext := h
            else if b.dir == 1 and c < mt
                b.dir := 0
            else if b.dir == 1
                b.ext := math.max(b.ext, h)
        b.disp := b.dir == -1 ? b.lo - b.ext : b.dir == 1 ? b.ext - b.hi : 0.0
        if gone or obk - b.k > obLook
            obs.remove(i)
        i -= 1
    if obs.size() > 200
        obs.shift()
    true

bool obOk = doOb and timeframe.in_seconds() <= 900
[obT0, obO1, obH1, obL1, obC1, obT1] = request.security(syminfo.tickerid, obOk ? "15" : timeframe.period,
     [time, open[1], high[1], low[1], close[1], time[1]], lookahead = barmerge.lookahead_on)
if obOk and not na(obT0)
    if not na(obLastT) and obT0 != obLastT and not na(obC1)
        obStep(obO1, obH1, obL1, obC1, obT1)
    obLastT := obT0

// [OB] the armed resistance OB covering price (the most significant one, if several), else the nearest one above
// it; mirrored for support below price: MT line (and zone), label
obDraw(int dir, line ln0, label lb0, box bx0) =>
    Ob best = na
    for b in obs
        if b.dir == dir and (dir == -1 ? b.hi > close : b.lo < close)
            if na(best)
                best := b
            else
                bool cb = b.lo <= close and close <= b.hi
                bool cbest = best.lo <= close and close <= best.hi
                float db = dir == -1 ? b.lo - close : close - b.hi
                float dbest = dir == -1 ? best.lo - close : close - best.hi
                if cb != cbest ? cb : cb ? b.disp > best.disp : db < dbest or (db == dbest and b.disp > best.disp)
                    best := b
    if not na(ln0)
        ln0.delete()
    if not na(lb0)
        lb0.delete()
    if not na(bx0)
        bx0.delete()
    line ln = na
    label lb = na
    box bx = na
    if not na(best)
        color cc = dir == -1 ? obResCol : obSupCol
        float mt = (best.hi + best.lo) / 2
        int x = time + obXoff * timeframe.in_seconds() * 1000
        // the MT line runs up to the label (its left end, half a bar before the text)
        ln := line.new(best.t, mt, math.max(time, x - timeframe.in_seconds() * 500), mt, xloc = xloc.bar_time, color = cc, style = obSty, width = obWid)
        lb := label.new(x, mt, "OBMT", xloc = xloc.bar_time, style = label.style_label_left,
             color = color.new(color.black, 100), textcolor = cc, size = size.small)
        bx := obZone ? box.new(best.t, best.hi, time, best.lo, xloc = xloc.bar_time, border_color = color.new(cc, 70),
             bgcolor = color.new(cc, 90)) : na
    [ln, lb, bx]

if barstate.islast
    [obRl, obRb, obRx] = obDraw(-1, obResLn, obResLb, obResBx)
    obResLn := obRl
    obResLb := obRb
    obResBx := obRx
    [obSl, obSb, obSx] = obDraw(1, obSupLn, obSupLb, obSupBx)
    obSupLn := obSl
    obSupLb := obSb
    obSupBx := obSx
