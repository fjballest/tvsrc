// [ATT] dimming steps of a rank (0 important, 1 normal, 2 minor) at the attenuation level
attE(int rank) =>
    attLvl == 0 ? 0 : rank == 0 ? (attLvl >= 4 ? 2 : attLvl >= 2 ? 1 : 0) : rank == 1 ? attLvl : attLvl + 1
// line / text color, width and fill transparency of a rank
attL(color c, int rank) =>
    int e = attE(rank)
    e == 0 ? c : color.new(c, color.t(c) + (100 - color.t(c)) * math.min(e * 0.18, 0.9))
attW(int w, int rank) =>
    math.max(1, w - (attE(rank) >= 2 ? 1 : 0))
attT(int t, int rank) =>
    int(math.min(100, t + (100 - t) * attE(rank) * 0.2))

// only timeframes above the chart are used, never the chart's own
mfOk(int s) =>
    s > tfsecs

mfEnabled(int i) =>
    int s = mfSecs.get(i)
    bool on = domf and mfOk(s) and mfSel.get(i)
    if domf and s > tfsecs and mfnext > 0
        int rank = 0
        for j = 0 to i - 1
            if mfSecs.get(j) > tfsecs
                rank += 1
        if rank < mfnext
            on := true
    on

wkEnabled(int i) =>
    int s = mfSecs.get(i)
    dowk and mfOk(s) and ((i == 1 and wk1h) or (i == 2 and wk4h))

mfcol(int i, int d) =>
    i == 0 ? (d == 1 ? mf15b : mf15s) : i == 1 ? (d == 1 ? mf1hb : mf1hs) : i == 2
         ? (d == 1 ? mf4hb : mf4hs) : i == 3 ? (d == 1 ? mfdb : mfds) : i == 4 ? (d == 1 ? mfwb : mfws)
         : (d == 1 ? mfmb : mfms)

// ============================================================ helpers
rmLine(line l) =>
    if not na(l)
        l.delete()
    true

rmLabel(label l) =>
    if not na(l)
        l.delete()
    true

rmBox(box b) =>
    if not na(b)
        b.delete()
    true

tright(int n) =>
    time + n * tfms

method kill(Lvl l) =>
    l.dead := true
    rmLine(l.hiln)
    rmLine(l.loln)
    rmLine(l.celn)
    rmLabel(l.hilb)
    rmLabel(l.lolb)
    rmLabel(l.celb)
    rmBox(l.bx)
    rmBox(l.cebx)
    rmLine(l.vln)
    rmLine(l.hiln2)
    rmLine(l.loln2)
    rmLine(l.celn2)
    true

tcol(int tfi, int d) =>
    tfi >= 3 ? mfcol(tfi - 3, d) : tfi == 0 ? (d == 1 ? dbull : dbear) : tfi == 1
         ? (d == 1 ? wbull : wbear) : (d == 1 ? mbull : mbear)

ksty(int k) =>
    k == 0 ? fvgsty : k == 1 ? visty : wksty

kcesty(int k) =>
    k == 0 ? fvgcesty : k == 1 ? vicesty : wksty

kwid(int k, int tfi) =>
    (k == 0 ? fvgwid : k == 1 ? viwid : wkwid) + (tfi < 3 ? tfi * wextra : 0)

// part: 0 high, 1 low, 2 CE. Wick levels only show the CE
kshow(int k, int part) =>
    k == 0 ? (part == 0 ? fvgshi : part == 1 ? fvgslo : fvgsce) : k == 1
         ? (part == 0 ? vishi : part == 1 ? vislo : visce) : part == 2

kfill(int k) =>
    k == 0 ? not fvgnofill : k == 1 ? not vinofill : false

lbltxt(Lvl l, string part) =>
    string nm = l.kind == 0 ? (l.inv ? txtifvg : txtfvg) : l.kind == 1 ? txtvi : l.kind == 4 ? txtulw
         : (l.dir == -1 ? txtwku : txtwkl)
    string sg = lbldir and l.kind < 2 ? (l.dir == 1 ? "+" : "-") : ""
    string tf = l.tfi >= 3 ? mfNames.get(l.tfi - 3) : l.tfi == 0 ? txtd : l.tfi == 1 ? txtw : txtm
    tf + " " + nm + sg + " " + part

mkLine(line cur, int x1, float p, int x2, color c, string sty, int w) =>
    line r = cur
    if na(r)
        r := line.new(x1, p, x2, p, xloc = xloc.bar_time, color = c, style = sty, width = w)
    else
        r.set_xy1(x1, p)
        r.set_xy2(x2, p)
        r.set_color(c)
    r

mkLbl(label cur, int x, float p, string txt, color c) =>
    label r = cur
    if na(r)
        r := label.new(x, p, txt, xloc = xloc.bar_time, style = label.style_label_left,
             color = transparent, textcolor = c, size = lblsz, text_formatting = lblbold
             ? text.format_bold : text.format_none)
    else
        r.set_xy(x, p)
        r.set_text(txt)
        r.set_textcolor(c)
    r

// [ATT] rank of a level: weekly / monthly important, daily / 4H normal, 15m / 1H minor;
// volume imbalances and wick levels one rank lower than FVGs
method rank(Lvl l) =>
    int b = l.tfi == 1 or l.tfi == 2 or l.tfi == 7 or l.tfi == 8 ? 0 : l.tfi == 0 or l.tfi == 5 or l.tfi == 6
         ? 1 : 2
    math.min(2, b + (l.kind == 0 or l.kind == 3 ? 0 : 1))

// [ATT] fill transparency of a D / W / M FVG or VI rectangle (100: no fill). Levels 0 and 1 as attT, level 2
// fades faster, levels 3 and 4 draw no fill except for VIs when they are few and narrow (vinar)
var bool vinar = false
fillT(int t, int rk, int kind, float h) =>
    attLvl <= 1 ? attT(t, rk) : attLvl == 2 ? int(math.min(100, t + (100 - t) * math.min(0.9, attE(rk) * 0.35)))
         : kind == 1 and vinar and not na(dAtr) and h <= 0.1 * dAtr ? int(t + (100 - t) * 0.7) : 100

rmLbl(label lb) =>
    if not na(lb)
        lb.delete()
    label(na)

// create or update the objects of a level, right edge at the last bar
method draw(Lvl l) =>
    int x = tright(xoff)
    int rk = l.rank()
    color c = attL(tcol(l.tfi, l.dir), rk)
    int w = attW(kwid(l.kind, l.tfi), rk)
    string s1 = ksty(l.kind)
    string s2 = kcesty(l.kind)
    // one label for the level when its hi / lo / CE labels would overlap
    bool shi = kshow(l.kind, 0)
    bool slo = kshow(l.kind, 1)
    bool sce = kshow(l.kind, 2)
    float gap = sce and (shi or slo) ? (l.hi - l.lo) / 2 : l.hi - l.lo
    int nl = (shi ? 1 : 0) + (slo ? 1 : 0) + (sce ? 1 : 0)
    bool one = dolbl and lblmrg > 0 and not na(dAtr) and nl >= 2 and gap < lblmrg * dAtr
    int ft = fillT(l.kind == 0 ? fvgtr : vitr, rk, l.kind, l.hi - l.lo)
    if kfill(l.kind) and ft >= 100
        rmBox(l.bx)
        l.bx := na
    else if kfill(l.kind)
        color bg = color.new(tcol(l.tfi, l.dir), ft)
        if na(l.bx)
            l.bx := box.new(l.t, l.hi, x, l.lo, xloc = xloc.bar_time, border_color = transparent,
                 bgcolor = bg)
        else
            l.bx.set_lefttop(l.t, l.hi)
            l.bx.set_rightbottom(x, l.lo)
            l.bx.set_bgcolor(bg)
    if shi
        l.hiln := mkLine(l.hiln, l.t, l.hi, x, c, s1, w)
        l.hiln.set_width(w)
        if dolbl and not one
            lbls.push(LblE.new(l, 0, l.hi, lbltxt(l, ptxthi), c, rk))
    if slo
        l.loln := mkLine(l.loln, l.t, l.lo, x, c, s1, w)
        l.loln.set_width(w)
        if dolbl and not one
            lbls.push(LblE.new(l, 1, l.lo, lbltxt(l, ptxtlo), c, rk))
    if sce
        l.celn := mkLine(l.celn, l.t, l.ce, x, c, s2, w)
        l.celn.set_width(w)
    if one
        lbls.push(LblE.new(l, 2, sce ? l.ce : (l.hi + l.lo) / 2, str.trim(lbltxt(l, "")), c, rk))
    else if sce and dolbl
        lbls.push(LblE.new(l, 2, l.ce, lbltxt(l, ptxtce), c, rk))
    true

// labels of the last bar: those of different levels closer than lblmrg x daily ATR become one label,
// "D VI / D FVG lo", at the price and in the color of the most important one (lowest rank, then first)
lblApply() =>
    int n = lbls.size()
    if n > 0
        array<float> ys = array.new<float>()
        for e in lbls
            ys.push(e.y)
        array<int> idx = ys.sort_indices()
        float tol = lblmrg > 0 and not na(dAtr) ? lblmrg * dAtr : 0.0
        int x = tright(xoff)
        int i = 0
        while i < n
            LblE e0 = lbls.get(idx.get(i))
            int j = i + 1
            while j < n and lbls.get(idx.get(j)).y - e0.y < tol
                j += 1
            // representative: lowest rank, then first
            int best = i
            for k = i to j - 1
                if lbls.get(idx.get(k)).rk < lbls.get(idx.get(best)).rk
                    best := k
            LblE b = lbls.get(idx.get(best))
            string txt = b.txt
            for k = i to j - 1
                if k != best
                    txt += " / " + lbls.get(idx.get(k)).txt
            Lvl l = b.l
            if b.part == 0
                l.hilb := mkLbl(l.hilb, x, b.y, txt, b.c)
                l.lshi := true
            else if b.part == 1
                l.lolb := mkLbl(l.lolb, x, b.y, txt, b.c)
                l.lslo := true
            else
                l.celb := mkLbl(l.celb, x, b.y, txt, b.c)
                l.lsce := true
            i := j
    true

// margin column of timeframe i: its rank among the enabled timeframes
mfCol(int i) =>
    int c = 0
    if i > 0
        for j = 0 to i - 1
            if mfEnabled(j)
                c += 1
    c

// horizontal line by bar index (margin only, a few bars around the last one)
mkLineI(line cur, int x1, float p, int x2, color c, string sty, int w) =>
    line r = cur
    if na(r)
        r := line.new(x1, p, x2, p, xloc = xloc.bar_index, color = c, style = sty, width = w)
    else
        r.set_xy1(x1, p)
        r.set_xy2(x2, p)
        r.set_color(c)
    r

// FVG by timeframe in the right margin: box in its timeframe column, CE box, arrow label, extended levels.
// The margin objects are placed by bar index (bar times to the right of the last bar collapse while the
// market is closed); the extended levels use bar time up to the last bar, then a bar-index link to the box.
method drawMM(Lvl l) =>
    int rk = l.rank()
    color c = attL(mfcol(l.tfi - 3, l.dir), rk)
    int mw = attW(mfwid, rk)
    // [ATT] the margin (boxes, labels, links to the boxes) is dimmed as an important level;
    // only the extended levels over the candles use the level's own rank
    color cm = attL(mfcol(l.tfi - 3, l.dir), 0)
    int mwm = attW(mfwid, 0)
    color bg = mffill == "never" ? transparent : color.new(mfcol(l.tfi - 3, l.dir), attT(mftr, 0))
    int x0 = bar_index + mfmoff + mfCol(l.tfi - 3) * (mfmw + 1)
    int x1 = x0 + mfmw
    if na(l.bx)
        l.bx := box.new(x0, l.hi, x1, l.lo, xloc = xloc.bar_index, border_color = cm,
             border_style = mfsty, border_width = mwm, bgcolor = bg)
    else
        l.bx.set_lefttop(x0, l.hi)
        l.bx.set_rightbottom(x1, l.lo)
        l.bx.set_border_color(cm)
        l.bx.set_bgcolor(bg)
    if mfshowce
        if na(l.cebx)
            l.cebx := box.new(x0, l.ce, x1, l.ce, xloc = xloc.bar_index, border_color = cm,
                 border_style = mfcesty, border_width = mwm, bgcolor = transparent)
        else
            l.cebx.set_lefttop(x0, l.ce)
            l.cebx.set_rightbottom(x1, l.ce)
            l.cebx.set_border_color(cm)
    if mfext
        int xs = mfextto == "to start" ? l.t : mfextto == "to creation" ? nz(l.tfm, l.t)
             : time[math.min(mfextn, bar_index)]
        l.hiln := mkLine(l.hiln, xs, l.hi, time, c, mfsty, mw)
        l.loln := mkLine(l.loln, xs, l.lo, time, c, mfsty, mw)
        l.hiln2 := mkLineI(l.hiln2, bar_index, l.hi, x0, cm, mfsty, mwm)
        l.loln2 := mkLineI(l.loln2, bar_index, l.lo, x0, cm, mfsty, mwm)
        if mfshowce
            l.celn := mkLine(l.celn, xs, l.ce, time, c, mfcesty, mw)
            l.celn2 := mkLineI(l.celn2, bar_index, l.ce, x0, cm, mfcesty, mwm)
    if mflbl
        float y = mflvert == "High" ? l.hi : mflvert == "Low" ? l.lo : l.ce
        // thin boxes: the timeframe on one line, the arrow on the next
        // TF with an "i" for an IFVG: "1Hi"
        string tfn = mflshowtf ? mfNames.get(l.tfi - 3) + (l.inv ? "i" : "") + "\n" : ""
        string txt = tfn + (l.dir == 1 ? "↑" : "↓")
        if na(l.celb)
            l.celb := label.new(int((x0 + x1) / 2), y, txt, xloc = xloc.bar_index,
                 style = label.style_label_center, color = transparent, textcolor = cm, size = mflsz,
                 text_formatting = lblbold ? text.format_bold : text.format_none)
        else
            l.celb.set_xy(int((x0 + x1) / 2), y)
            l.celb.set_text(txt)
            l.celb.set_textcolor(cm)
    true

// FVG by timeframe: outline box (optionally filled), CE box and label
method drawM(Lvl l) =>
    int x = tright(xoff)
    int rk = l.rank()
    color c = attL(mfcol(l.tfi - 3, l.dir), rk)
    int mw = attW(mfwid, rk)
    color bg = mffill == "always" ? color.new(mfcol(l.tfi - 3, l.dir), attT(mftr, rk)) : transparent
    if na(l.bx)
        l.bx := box.new(l.t, l.hi, x, l.lo, xloc = xloc.bar_time, border_color = c,
             border_style = mfsty, border_width = mw, bgcolor = bg)
    else
        l.bx.set_lefttop(l.t, l.hi)
        l.bx.set_rightbottom(x, l.lo)
        l.bx.set_border_color(c)
        l.bx.set_bgcolor(bg)
    if mfshowce
        if na(l.cebx)
            l.cebx := box.new(l.t, l.ce, x, l.ce, xloc = xloc.bar_time, border_color = c,
                 border_style = mfcesty, border_width = mw, bgcolor = transparent)
        else
            l.cebx.set_lefttop(l.t, l.ce)
            l.cebx.set_rightbottom(x, l.ce)
            l.cebx.set_border_color(c)
    if mfshowform and not na(l.tfm) and l.tfm >= l.t
        if na(l.vln)
            l.vln := line.new(l.tfm, l.hi, l.tfm, l.lo, xloc = xloc.bar_time, color = c,
                 style = line.style_solid, width = mw > 1 ? mw - 1 : mw)
        else
            l.vln.set_xy1(l.tfm, l.hi)
            l.vln.set_xy2(l.tfm, l.lo)
            l.vln.set_color(c)
    if mflbl and (mflshowtf or mflshowtype or mflshowdir)
        float y = mflvert == "High" ? l.hi : mflvert == "Low" ? l.lo : l.ce
        int lx = mflpos == "Inside left" or mflpos == "Left of box" ? l.t : mflpos == "Center"
             ? int((l.t + x) / 2) : x
        string st = mflpos == "Right margin" or mflpos == "Inside left" ? label.style_label_left
             : mflpos == "Center" ? label.style_label_center : label.style_label_right
        string sg = mflshowdir ? (l.dir == 1 ? "+" : "-") : ""
        string ty = (mflshowtype ? (l.inv ? txtifvg : txtfvg) : "") + sg
        string tfn = mflshowtf ? mfNames.get(l.tfi - 3) : ""
        string txt = tfn + (tfn != "" and ty != "" ? " " : "") + ty
        if na(l.celb)
            l.celb := label.new(lx, y, txt, xloc = xloc.bar_time, style = st, color = transparent,
                 textcolor = c, size = mflsz, text_formatting = lblbold ? text.format_bold
                 : text.format_none)
        else
            l.celb.set_xy(lx, y)
            l.celb.set_style(st)
            l.celb.set_text(txt)
            l.celb.set_textcolor(c)
    true

// ============================================================ levels
// overlapping FVGs of the same timeframe and direction: returns false if the new one is discarded
fvgOk(int kind, int tfi, int dir, float lo, float hi) =>
    bool ok = true
    if fvgovl != "Keep all"
        if fvgovl == "Enclosing"
            for o in lvls
                if o.kind == kind and o.tfi == tfi and o.dir == dir and not o.inv and not o.dead
                    if lo <= o.hi and hi >= o.lo and o.lo <= lo and o.hi >= hi
                        ok := false
        if ok
            int i = lvls.size() - 1
            while i >= 0
                Lvl o = lvls.get(i)
                if o.kind == kind and o.tfi == tfi and o.dir == dir and not o.inv and not o.dead
                    if lo <= o.hi and hi >= o.lo
                        o.kill()
                        lvls.remove(i)
                i -= 1
    ok

// returns na when an FVG is discarded
addLvl(int kind, int tfi, int dir, float lo, float hi, int t) =>
    Lvl l = na
    if (kind != 0 and kind != 3) or fvgOk(kind, tfi, dir, lo, hi)
        l := Lvl.new(kind, tfi, dir, hi, lo, (hi + lo) / 2, math.max(t, tfirst))
        lvls.push(l)
        int keep = kind == 0 ? nfvg : kind == 1 ? nvi : kind == 3 ? nmf : nwk
        int cnt = 0
        for v in lvls
            if v.kind == kind and v.tfi == tfi and not v.dead
                cnt += 1
        if cnt > keep
            for i = 0 to lvls.size() - 1
                Lvl o = lvls.get(i)
                if o.kind == kind and o.tfi == tfi and not o.dead
                    o.kill()
                    lvls.remove(i)
                    break
    l

// one step of the wick group state machine (side 0 = upper wick, 1 = lower wick)
wickStep(int tfi, int side, bool q, float sz, float lo, float hi, int t) =>
    int idx = tfi * 2 + side
    Lvl cur = wcur.get(idx)
    if not q
        Lvl none = na
        wcur.set(idx, none)
    else if na(cur) or (cur.dead and sz > cur.wsize)
        Lvl n = addLvl(2, tfi, side == 0 ? -1 : 1, lo, hi, t)
        n.wsize := sz
        wcur.set(idx, n)
    else if sz > cur.wsize
        cur.lo := lo
        cur.hi := hi
        cur.ce := (lo + hi) / 2
        cur.t := math.max(t, tfirst)
        cur.wsize := sz
    true

// FVGs of one completed candle triple (1 newest, 3 oldest), with the VI extension: any gap between the
// body of candle 2 and the body of candle 1 or 3 widens the range, also when the wicks do not overlap
// (a gap), as in FVGS+IFVG+OTE
addFvgs(int kind, int tfi, float minsz, float o1, float h1, float l1, float c1,
     float o2, float h2, float l2, float c2, float h3, float l3, float o3, float c3,
     int t2, int tform) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float top3 = math.max(o3, c3)
    float bot3 = math.min(o3, c3)
    if l1 > h3 and l1 - h3 >= minsz
        float flo = h3
        float fhi = l1
        if fvgvi
            if botP > top3 and botP - top3 >= vimin
                flo := math.min(flo, top3)
                fhi := math.max(fhi, botP)
            if botC > topP and botC - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, botC)
        Lvl nb = addLvl(kind, tfi, 1, flo, fhi, t2)
        if not na(nb)
            nb.tfm := tform
    if h1 < l3 and l3 - h1 >= minsz
        float flo = h1
        float fhi = l3
        if fvgvi
            if topP < bot3 and bot3 - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, bot3)
            if topC < botP and botP - topC >= vimin
                flo := math.min(flo, topC)
                fhi := math.max(fhi, botP)
        Lvl ns = addLvl(kind, tfi, -1, flo, fhi, t2)
        if not na(ns)
            ns.tfm := tform
    true

// wick groups of one completed candle
addWicks(int tfi, float o1, float h1, float l1, float c1, int t1) =>
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float rng = h1 - l1
    float uw = h1 - topC
    float lw = botC - l1
    bool uq = rng > 0 and uw >= wkmin / 100.0 * rng
    bool lq = rng > 0 and lw >= wkmin / 100.0 * rng
    wickStep(tfi, 0, uq, uw, topC, h1, t1)
    wickStep(tfi, 1, lq, lw, l1, botC, t1)
    // ULW CE: both wicks of a D / W / M candle are long, level halfway between their CEs
    if doulw and tfi < 3 and uq and lq
        float p = ((topC + h1) / 2 + (l1 + botC) / 2) / 2
        addLvl(4, tfi, c1 >= p ? 1 : -1, p, p, t1)
    true

// called once per completed higher timeframe candle. 1 is the candle just completed,
// 2 the one before and 3 the one before that
processTf(int tfi, float o1, float h1, float l1, float c1, int t1,
     float o2, float h2, float l2, float c2, int t2, float h3, float l3, float o3, float c3) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    // no FVG / IFVG lines for D, W or M when the same timeframe is drawn as boxes (redundant)
    if dofvg and not mfEnabled(tfi + 3)
        addFvgs(0, tfi, fvgmin, o1, h1, l1, c1, o2, h2, l2, c2, h3, l3, o3, c3, t2, time)
    if dovi
        if botC > topP and l1 <= h2 and botC - topP >= vimin
            addLvl(1, tfi, 1, topP, botC, t1)
        if topC < botP and h1 >= l2 and botP - topC >= vimin
            addLvl(1, tfi, -1, topC, botP, t1)
    if dowk
        addWicks(tfi, o1, h1, l1, c1, t1)
    true

// checked every chart bar: FVG -> IFVG inversion (on a close of the FVG's own timeframe) and mitigation.
// Only surviving levels get drawn.
runMit() =>
    int i = lvls.size() - 1
    while i >= 0
        Lvl l = lvls.get(i)
        if l.dead
            lvls.remove(i)
        else
            bool fvgLive = (l.kind == 0 or l.kind == 3) and doifvg and not l.inv
            bool flipped = false
            if fvgLive and tfNew.get(l.tfi)
                float c = tfClose.get(l.tfi)
                if (l.dir == 1 and c < l.lo) or (l.dir == -1 and c > l.hi)
                    l.inv := true
                    l.dir := -l.dir
                    flipped := true
            if mmode > 0 and not fvgLive and not flipped
                bool m = false
                // close rule: on a close of the level's own timeframe, never on a chart close
                bool tn = tfNew.get(l.tfi)
                float tc = tfClose.get(l.tfi)
                if l.dir == 1
                    m := mmode == 1 ? low <= l.ce : mmode == 2 ? low <= l.lo : tn and tc < l.lo
                else
                    m := mmode == 1 ? high >= l.ce : mmode == 2 ? high >= l.hi : tn and tc > l.hi
                if m
                    l.kill()
                    lvls.remove(i)
        i -= 1
    true

// ============================================================ higher timeframe data
// previous completed candles only (lookahead_on with [n]), so nothing repaints
htfT(string tid, string tf) =>
    request.security(tid, tf,
         [time, open[1], high[1], low[1], close[1], time[1],
         open[2], high[2], low[2], close[2], time[2], high[3], low[3], open[3], close[3]],
         lookahead = barmerge.lookahead_on)

htf(string tf) =>
    htfT(syminfo.tickerid, tf)

setScore(float o1, float h1, float l1, float c1, float o2, float h2, float l2, float c2) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float vi = 0.0
    if botC > topP and l1 <= h2
        vi := botC - topP
    if topC < botP and h1 >= l2
        vi := botP - topC
    math.max(vi, h1 - topC, botC - l1)

// ============================================================ turtle soup / continuation
// per timeframe (15m, 1H, 4H, D): the previous completed candle and the forming one, built from chart bars
type TSt
    float po = na
    float ph = na
    float pl = na
    float pc = na
    float o = na
    float h = na
    float l = na
    bool armed = false   // a continuation was seen in the forming candle
    bool inv = false     // and then price went back beyond the previous close: no continuation for this candle

var array<TSt> tst = array.from(TSt.new(), TSt.new(), TSt.new(), TSt.new())
var array<string> tsNames = array.from("15m", "1H", "4H", "D")
array<bool> tsNew = array.from(timeframe.change("15"), timeframe.change("60"), timeframe.change("240"),
     timeframe.change("D"))

// continuation of the previous candle's direction now: price traded against it from the forming open and is
// back beyond the previous close
method tsCont(TSt s) =>
    s.pc < s.po ? s.h > s.o and close < s.pc : s.pc > s.po ? s.l < s.o and close > s.pc : false

method tsStep(TSt s, bool nw) =>
    if nw or na(s.o)
        if not na(s.o)
            s.po := s.o
            s.ph := s.h
            s.pl := s.l
            s.pc := close[1]
        s.o := open
        s.h := high
        s.l := low
        s.armed := false
        s.inv := false
    else
        s.h := math.max(s.h, high)
        s.l := math.min(s.l, low)
    if not na(s.pc)
        if s.armed and (s.pc < s.po ? close > s.pc : close < s.pc)
            s.inv := true
        if s.tsCont()
            s.armed := true
    true

// 1 turtle soup up, -1 turtle soup down, 2 continuation up, -2 continuation down, 0 nothing
method tsSig(TSt s) =>
    int r = 0
    if not na(s.pc)
        if s.pc < s.po and s.l < s.pl and close > s.pc
            r := 1
        else if s.pc > s.po and s.h > s.ph and close < s.pc
            r := -1
        else if not s.inv and s.tsCont()
            r := s.pc < s.po ? -2 : 2
    r
