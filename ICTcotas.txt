// Each FVG / volume imbalance is drawn with high, low and CE (midpoint). Wicks are grouped
// (consecutive candles with a wick on the same side) and the CE of the longest wick of the
// group (or of a single wick) is drawn. FVGs closed through become IFVGs. Mitigated levels are
// removed.

const color transparent = #ffffff00

getstyle(string s) =>
    switch s
        "dashed" => line.style_dashed
        "dotted" => line.style_dotted
        => line.style_solid

getsize(string s) =>
    switch s
        "tiny" => size.tiny
        "small" => size.small
        "large" => size.large
        "huge" => size.huge
        => size.normal

// ============================================================ setup
int tfsecs = timeframe.in_seconds()
int tfms = tfsecs * 1000
bool dayOk = dodaily and tfsecs <= 86400
bool wkOk = doweekly and tfsecs <= 604800
bool monOk = domonthly and tfsecs <= 2592000
int mmode = mitig == "CE touched" ? 1 : mitig == "Fully filled (wick)" ? 2
     : mitig == "Fully filled (close)" ? 3 : 0
string lblsz = getsize(lblsize)
string mflsz = getsize(mflsize)
var int tfirst = time

type Lvl
    int kind    // 0 FVG / IFVG, 1 volume imbalance, 2 wick
    int tfi     // 0 daily, 1 weekly, 2 monthly
    int dir     // 1 bullish / support, -1 bearish / resistance
    float hi
    float lo
    float ce
    int t
    float wsize = na
    bool inv = false    // FVG inverted into an IFVG
    bool dead = false
    line hiln = na
    line loln = na
    line celn = na
    label hilb = na
    label lolb = na
    label celb = na
    box bx = na
    box cebx = na
    int tfm = na      // formation time (close of the last candle of the FVG)
    line vln = na

var array<Lvl> lvls = array.new<Lvl>()
var array<Lvl> wcur = array.new<Lvl>(12, na)   // current wick group level: tf * 2 + side

// timeframe list for the FVGs by timeframe (kind 3 levels use tfi = 3 + index)
var array<int> mfSecs = array.from(900, 3600, 14400, 86400, 604800)
var array<string> mfTfs = array.from("15", "60", "240", "D", "W")
var array<string> mfNames = array.from("15m", "1H", "4H", "D", "W")
var array<bool> mfSel = array.from(mf15, mf1h, mf4h, mfd, mfw)
var array<int> mfLastT = array.new<int>(5, na)

mfEnabled(int i) =>
    int s = mfSecs.get(i)
    bool on = domf and s >= tfsecs and mfSel.get(i)
    if domf and s > tfsecs and mfnext > 0
        int rank = 0
        for j = 0 to i - 1
            if mfSecs.get(j) > tfsecs
                rank += 1
        if rank < mfnext
            on := true
    on

wkEnabled(int i) =>
    int s = mfSecs.get(i)
    dowk and s >= tfsecs and ((i == 1 and wk1h) or (i == 2 and wk4h))

mfcol(int i, int d) =>
    i == 0 ? (d == 1 ? mf15b : mf15s) : i == 1 ? (d == 1 ? mf1hb : mf1hs) : i == 2
         ? (d == 1 ? mf4hb : mf4hs) : i == 3 ? (d == 1 ? mfdb : mfds) : (d == 1 ? mfwb : mfws)

// ============================================================ helpers
rmLine(line l) =>
    if not na(l)
        l.delete()
    true

rmLabel(label l) =>
    if not na(l)
        l.delete()
    true

rmBox(box b) =>
    if not na(b)
        b.delete()
    true

tright(int n) =>
    time + n * tfms

method kill(Lvl l) =>
    l.dead := true
    rmLine(l.hiln)
    rmLine(l.loln)
    rmLine(l.celn)
    rmLabel(l.hilb)
    rmLabel(l.lolb)
    rmLabel(l.celb)
    rmBox(l.bx)
    rmBox(l.cebx)
    rmLine(l.vln)
    true

tcol(int tfi, int d) =>
    tfi >= 3 ? mfcol(tfi - 3, d) : tfi == 0 ? (d == 1 ? dbull : dbear) : tfi == 1
         ? (d == 1 ? wbull : wbear) : (d == 1 ? mbull : mbear)

ksty(int k) =>
    k == 0 ? fvgsty : k == 1 ? visty : wksty

kcesty(int k) =>
    k == 0 ? fvgcesty : k == 1 ? vicesty : wksty

kwid(int k, int tfi) =>
    (k == 0 ? fvgwid : k == 1 ? viwid : wkwid) + (tfi < 3 ? tfi * wextra : 0)

// part: 0 high, 1 low, 2 CE. Wick levels only show the CE
kshow(int k, int part) =>
    k == 0 ? (part == 0 ? fvgshi : part == 1 ? fvgslo : fvgsce) : k == 1
         ? (part == 0 ? vishi : part == 1 ? vislo : visce) : part == 2

kfill(int k) =>
    k == 0 ? not fvgnofill : k == 1 ? not vinofill : false

lbltxt(Lvl l, string part) =>
    string nm = l.kind == 0 ? (l.inv ? txtifvg : txtfvg) : l.kind == 1 ? txtvi
         : (l.dir == -1 ? txtwku : txtwkl)
    string sg = lbldir and l.kind < 2 ? (l.dir == 1 ? "+" : "-") : ""
    string tf = l.tfi >= 3 ? mfNames.get(l.tfi - 3) : l.tfi == 0 ? txtd : l.tfi == 1 ? txtw : txtm
    tf + " " + nm + sg + " " + part

mkLine(line cur, int x1, float p, int x2, color c, string sty, int w) =>
    line r = cur
    if na(r)
        r := line.new(x1, p, x2, p, xloc = xloc.bar_time, color = c, style = sty, width = w)
    else
        r.set_xy1(x1, p)
        r.set_xy2(x2, p)
        r.set_color(c)
    r

mkLbl(label cur, int x, float p, string txt, color c) =>
    label r = cur
    if na(r)
        r := label.new(x, p, txt, xloc = xloc.bar_time, style = label.style_label_left,
             color = transparent, textcolor = c, size = lblsz, text_formatting = lblbold
             ? text.format_bold : text.format_none)
    else
        r.set_xy(x, p)
        r.set_text(txt)
        r.set_textcolor(c)
    r

// create or update the objects of a level, right edge at the last bar
method draw(Lvl l) =>
    int x = tright(xoff)
    color c = tcol(l.tfi, l.dir)
    int w = kwid(l.kind, l.tfi)
    string s1 = ksty(l.kind)
    string s2 = kcesty(l.kind)
    if kfill(l.kind)
        color bg = color.new(c, l.kind == 0 ? fvgtr : vitr)
        if na(l.bx)
            l.bx := box.new(l.t, l.hi, x, l.lo, xloc = xloc.bar_time, border_color = transparent,
                 bgcolor = bg)
        else
            l.bx.set_lefttop(l.t, l.hi)
            l.bx.set_rightbottom(x, l.lo)
            l.bx.set_bgcolor(bg)
    if kshow(l.kind, 0)
        l.hiln := mkLine(l.hiln, l.t, l.hi, x, c, s1, w)
        if dolbl
            l.hilb := mkLbl(l.hilb, x, l.hi, lbltxt(l, ptxthi), c)
    if kshow(l.kind, 1)
        l.loln := mkLine(l.loln, l.t, l.lo, x, c, s1, w)
        if dolbl
            l.lolb := mkLbl(l.lolb, x, l.lo, lbltxt(l, ptxtlo), c)
    if kshow(l.kind, 2)
        l.celn := mkLine(l.celn, l.t, l.ce, x, c, s2, w)
        if dolbl
            l.celb := mkLbl(l.celb, x, l.ce, lbltxt(l, ptxtce), c)
    true

// FVG by timeframe: outline box (optionally filled), CE box and label
method drawM(Lvl l) =>
    int x = tright(xoff)
    color c = mfcol(l.tfi - 3, l.dir)
    color bg = mfnofill ? transparent : color.new(c, mftr)
    if na(l.bx)
        l.bx := box.new(l.t, l.hi, x, l.lo, xloc = xloc.bar_time, border_color = c,
             border_style = mfsty, border_width = mfwid, bgcolor = bg)
    else
        l.bx.set_lefttop(l.t, l.hi)
        l.bx.set_rightbottom(x, l.lo)
        l.bx.set_border_color(c)
        l.bx.set_bgcolor(bg)
    if mfshowce
        if na(l.cebx)
            l.cebx := box.new(l.t, l.ce, x, l.ce, xloc = xloc.bar_time, border_color = c,
                 border_style = mfcesty, border_width = mfwid, bgcolor = transparent)
        else
            l.cebx.set_lefttop(l.t, l.ce)
            l.cebx.set_rightbottom(x, l.ce)
            l.cebx.set_border_color(c)
    if mfshowform and not na(l.tfm) and l.tfm >= l.t
        if na(l.vln)
            l.vln := line.new(l.tfm, l.hi, l.tfm, l.lo, xloc = xloc.bar_time, color = c,
                 style = line.style_solid, width = mfwid>1?mfwid-1:mfwid)
        else
            l.vln.set_xy1(l.tfm, l.hi)
            l.vln.set_xy2(l.tfm, l.lo)
            l.vln.set_color(c)
    if mflbl and (mflshowtf or mflshowtype or mflshowdir)
        float y = mflvert == "High" ? l.hi : mflvert == "Low" ? l.lo : l.ce
        int lx = mflpos == "Inside left" or mflpos == "Left of box" ? l.t : mflpos == "Center"
             ? int((l.t + x) / 2) : x
        string st = mflpos == "Right margin" or mflpos == "Inside left" ? label.style_label_left
             : mflpos == "Center" ? label.style_label_center : label.style_label_right
        string sg = mflshowdir ? (l.dir == 1 ? "+" : "-") : ""
        string ty = (mflshowtype ? (l.inv ? txtifvg : txtfvg) : "") + sg
        string tfn = mflshowtf ? mfNames.get(l.tfi - 3) : ""
        string txt = tfn + (tfn != "" and ty != "" ? " " : "") + ty
        if na(l.celb)
            l.celb := label.new(lx, y, txt, xloc = xloc.bar_time, style = st, color = transparent,
                 textcolor = c, size = mflsz, text_formatting = lblbold ? text.format_bold
                 : text.format_none)
        else
            l.celb.set_xy(lx, y)
            l.celb.set_style(st)
            l.celb.set_text(txt)
            l.celb.set_textcolor(c)
    true

// ============================================================ levels
// overlapping FVGs of the same timeframe and direction: returns false if the new one is discarded
fvgOk(int kind, int tfi, int dir, float lo, float hi) =>
    bool ok = true
    if fvgovl != "Keep all"
        if fvgovl == "Enclosing"
            for o in lvls
                if o.kind == kind and o.tfi == tfi and o.dir == dir and not o.inv and not o.dead
                    if lo <= o.hi and hi >= o.lo and o.lo <= lo and o.hi >= hi
                        ok := false
        if ok
            int i = lvls.size() - 1
            while i >= 0
                Lvl o = lvls.get(i)
                if o.kind == kind and o.tfi == tfi and o.dir == dir and not o.inv and not o.dead
                    if lo <= o.hi and hi >= o.lo
                        o.kill()
                        lvls.remove(i)
                i -= 1
    ok

// returns na when an FVG is discarded
addLvl(int kind, int tfi, int dir, float lo, float hi, int t) =>
    Lvl l = na
    if (kind != 0 and kind != 3) or fvgOk(kind, tfi, dir, lo, hi)
        l := Lvl.new(kind, tfi, dir, hi, lo, (hi + lo) / 2, math.max(t, tfirst))
        lvls.push(l)
        int keep = kind == 0 ? nfvg : kind == 1 ? nvi : kind == 3 ? nmf : nwk
        int cnt = 0
        for v in lvls
            if v.kind == kind and v.tfi == tfi and not v.dead
                cnt += 1
        if cnt > keep
            for i = 0 to lvls.size() - 1
                Lvl o = lvls.get(i)
                if o.kind == kind and o.tfi == tfi and not o.dead
                    o.kill()
                    lvls.remove(i)
                    break
    l

// one step of the wick group state machine (side 0 = upper wick, 1 = lower wick)
wickStep(int tfi, int side, bool q, float sz, float lo, float hi, int t) =>
    int idx = tfi * 2 + side
    Lvl cur = wcur.get(idx)
    if not q
        Lvl none = na
        wcur.set(idx, none)
    else if na(cur) or (cur.dead and sz > cur.wsize)
        Lvl n = addLvl(2, tfi, side == 0 ? -1 : 1, lo, hi, t)
        n.wsize := sz
        wcur.set(idx, n)
    else if sz > cur.wsize
        cur.lo := lo
        cur.hi := hi
        cur.ce := (lo + hi) / 2
        cur.t := math.max(t, tfirst)
        cur.wsize := sz
    true

// FVGs of one completed candle triple (1 newest, 3 oldest), with the VI extension: any gap between the
// body of candle 2 and the body of candle 1 or 3 widens the range, also when the wicks do not overlap
// (a gap), as in FVGS+IFVG+OTE
addFvgs(int kind, int tfi, float minsz, float o1, float h1, float l1, float c1,
     float o2, float h2, float l2, float c2, float h3, float l3, float o3, float c3,
     int t2, int tform) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float top3 = math.max(o3, c3)
    float bot3 = math.min(o3, c3)
    if l1 > h3 and l1 - h3 >= minsz
        float flo = h3
        float fhi = l1
        if fvgvi
            if botP > top3 and botP - top3 >= vimin
                flo := math.min(flo, top3)
                fhi := math.max(fhi, botP)
            if botC > topP and botC - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, botC)
        Lvl nb = addLvl(kind, tfi, 1, flo, fhi, t2)
        if not na(nb)
            nb.tfm := tform
    if h1 < l3 and l3 - h1 >= minsz
        float flo = h1
        float fhi = l3
        if fvgvi
            if topP < bot3 and bot3 - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, bot3)
            if topC < botP and botP - topC >= vimin
                flo := math.min(flo, topC)
                fhi := math.max(fhi, botP)
        Lvl ns = addLvl(kind, tfi, -1, flo, fhi, t2)
        if not na(ns)
            ns.tfm := tform
    true

// wick groups of one completed candle
addWicks(int tfi, float o1, float h1, float l1, float c1, int t1) =>
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float rng = h1 - l1
    float uw = h1 - topC
    float lw = botC - l1
    bool uq = rng > 0 and uw >= wkmin / 100.0 * rng
    bool lq = rng > 0 and lw >= wkmin / 100.0 * rng
    wickStep(tfi, 0, uq, uw, topC, h1, t1)
    wickStep(tfi, 1, lq, lw, l1, botC, t1)
    true

// called once per completed higher timeframe candle. 1 is the candle just completed,
// 2 the one before and 3 the one before that
processTf(int tfi, float o1, float h1, float l1, float c1, int t1,
     float o2, float h2, float l2, float c2, int t2, float h3, float l3, float o3, float c3) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    if dofvg
        addFvgs(0, tfi, fvgmin, o1, h1, l1, c1, o2, h2, l2, c2, h3, l3, o3, c3, t2, time)
    if dovi
        if botC > topP and l1 <= h2 and botC - topP >= vimin
            addLvl(1, tfi, 1, topP, botC, t1)
        if topC < botP and h1 >= l2 and botP - topC >= vimin
            addLvl(1, tfi, -1, topC, botP, t1)
    if dowk
        addWicks(tfi, o1, h1, l1, c1, t1)
    true

// checked every chart bar: FVG -> IFVG inversion and mitigation.
// Only surviving levels get drawn.
runMit() =>
    int i = lvls.size() - 1
    while i >= 0
        Lvl l = lvls.get(i)
        if l.dead
            lvls.remove(i)
        else
            bool fvgLive = (l.kind == 0 or l.kind == 3) and doifvg and not l.inv
            bool flipped = false
            if fvgLive
                if (l.dir == 1 and close < l.lo) or (l.dir == -1 and close > l.hi)
                    l.inv := true
                    l.dir := -l.dir
                    flipped := true
            if mmode > 0 and not fvgLive and not flipped
                bool m = false
                if l.dir == 1
                    m := mmode == 1 ? low <= l.ce : mmode == 2 ? low <= l.lo : close < l.lo
                else
                    m := mmode == 1 ? high >= l.ce : mmode == 2 ? high >= l.hi : close > l.hi
                if m
                    l.kill()
                    lvls.remove(i)
        i -= 1
    true

// ============================================================ higher timeframe data
// previous completed candles only (lookahead_on with [n]), so nothing repaints
htf(string tf) =>
    request.security(syminfo.tickerid, tf,
         [time, open[1], high[1], low[1], close[1], time[1],
         open[2], high[2], low[2], close[2], time[2], high[3], low[3], open[3], close[3]],
         lookahead = barmerge.lookahead_on)

[dT, dO1, dH1, dL1, dC1, dT1, dO2, dH2, dL2, dC2, dT2, dH3, dL3, dO3, dC3] = htf(dayOk ? "D"
     : timeframe.period)
[wT, wO1, wH1, wL1, wC1, wT1, wO2, wH2, wL2, wC2, wT2, wH3, wL3, wO3, wC3] = htf(wkOk ? "W"
     : timeframe.period)
[mT, mO1, mH1, mL1, mC1, mT1, mO2, mH2, mL2, mC2, mT2, mH3, mL3, mO3, mC3] = htf(monOk ? "M"
     : timeframe.period)

bool newD = dayOk and dT != nz(dT[1], dT)
bool newW = wkOk and wT != nz(wT[1], wT)
bool newM = monOk and mT != nz(mT[1], mT)

if newD
    processTf(0, dO1, dH1, dL1, dC1, dT1, dO2, dH2, dL2, dC2, dT2, dH3, dL3, dO3, dC3)
if newW
    processTf(1, wO1, wH1, wL1, wC1, wT1, wO2, wH2, wL2, wC2, wT2, wH3, wL3, wO3, wC3)
if newM
    processTf(2, mO1, mH1, mL1, mC1, mT1, mO2, mH2, mL2, mC2, mT2, mH3, mL3, mO3, mC3)
// FVGs by timeframe: one request per enabled timeframe, processed on each new candle
for i = 0 to 4
    bool fvgOn = mfEnabled(i)
    bool wkOn = wkEnabled(i)
    if fvgOn or wkOn
        [t0, p1, q1, r1, u1, t1, p2, q2, r2, u2, t2, h3, l3, p3, u3] = htf(mfTfs.get(i))
        if not na(t0)
            int last = mfLastT.get(i)
            if not na(last) and t0 != last
                if fvgOn
                    addFvgs(3, 3 + i, mfmin, p1, q1, r1, u1, p2, q2, r2, u2, h3, l3, p3, u3, t2, t0)
                if wkOn
                    addWicks(3 + i, p1, q1, r1, u1, t1)
            mfLastT.set(i, t0)
runMit()

if barstate.islast
    for l in lvls

            l.drawM()
        else
            l.draw()
