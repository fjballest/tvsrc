// Wick CEs.
// Only relevant ones are kept: swing wicks, wicks next to displacement, the furthest wick
// in a run of same side wicks, and those later respected (pierced by wicks, not by bodies).
// The last few candles always show theirs.
// Kept CEs not yet crossed by a body extend to the current bar while price is near.
type WCE
    int b
    float ce
    float ext
    bool up
    line ln = na
    bool swing = false
    bool disp = false
    bool clus = false
    bool shadow = false
    bool tested = false
    bool dead = false

method kept(WCE w) =>
    not w.shadow and (w.swing or w.disp or w.clus or w.tested)

method hide(WCE w) =>
    if not na(w.ln)
        w.ln.delete()
        w.ln := na

type WRun
    WCE last = na
    WCE best = na
    int n = 0

method addw(WRun r, WCE w, int nclus) =>
    if not na(r.last) and r.last.b == w.b - 1
        r.n := r.n + 1
        if (w.up and w.ext > r.best.ext) or (not w.up and w.ext < r.best.ext)
            r.best.shadow := true
            r.best.clus := false
            r.best := w
        else
            w.shadow := true
    else
        r.n := 1
        r.best := w
    r.last := w
    if nclus > 0 and r.n >= nclus
        r.best.clus := true

var wces = array.new<WCE>()
var wru = WRun.new()
var wrl = WRun.new()

wfind(int b, bool up) =>
    WCE r = na
    n = wces.size()
    if n > 0
        for k = n - 1 to 0
            w = wces.get(k)
            if w.b < b
                break
            if w.b == b and w.up == up
                r := w
                break
    r

dbody = math.abs(dC - dO)
dbavg = ta.sma(dbody, 20)
dtr = math.max(dH - dL, math.abs(dH - nz(dC[1], dO)), math.abs(dL - nz(dC[1], dO)))
datr = ta.rma(dtr, 14)
dwph = ta.pivothigh(dH, oteswingleft, oteswingright)
dwpl = ta.pivotlow(dL, oteswingleft, oteswingright)

if dowick and compute and barstate.isnew and bar_index > 2
    i = bar_index - 1
    bodyh = math.max(dO[1], dC[1])
    bodyl = math.min(dO[1], dC[1])

    // the closed candle tests or crosses older CEs
    for w in wces
        if not w.dead and w.b < i
            if bodyl < w.ce and bodyh > w.ce
                w.dead := true
            else if wresp and i >= w.b + 2 and dH[1] >= w.ce and dL[1] <= w.ce
                w.tested := true

    bigcur = wdisp > 0 and dbody[1] >= wdisp * dbavg[1]
    bigprev = wdisp > 0 and dbody[2] >= wdisp * dbavg[2]
    if bigcur
        wp = wfind(i - 1, true)
        if not na(wp)
            wp.disp := true
        wp := wfind(i - 1, false)
        if not na(wp)
            wp.disp := true
    if wswing and not na(dwph[1])
        wp = wfind(i - oteswingright, true)
        if not na(wp)
            wp.swing := true
    if wswing and not na(dwpl[1])
        wp = wfind(i - oteswingright, false)
        if not na(wp)
            wp.swing := true

    // new CEs for the closed candle
    if dH[1] - bodyh > minwick
        w = WCE.new(i, math.avg(dH[1], bodyh), dH[1], true)
        w.disp := bigcur or bigprev
        wru.addw(w, wclus)
        wces.push(w)
    if bodyl - dL[1] > minwick
        w = WCE.new(i, math.avg(dL[1], bodyl), dL[1], false)
        w.disp := bigcur or bigprev
        wrl.addw(w, wclus)
        wces.push(w)

    // draw, extend, hide and forget
    // only recent CEs and, on each side, the outermost one price is still going to, extend.
    hold = math.max(wrecent, oteswingright + 1)
    n = wces.size()
    reach = wreach * datr[1]
    WCE outu = na
    WCE outl = na
    int outub = na
    int outlb = na
    if n > 0 and wreach > 0
        for w in wces
            if not w.dead and not w.shadow and (w.kept() or w.b > i - wrecent) and math.abs(w.ce - dC[1]) <= reach
                if w.up and w.ce > dC[1] and (na(outu) or w.ext > outu.ext)
                    outu := w
                if not w.up and w.ce < dC[1] and (na(outl) or w.ext < outl.ext)
                    outl := w
    if not na(outu)
        outub := outu.b
    if not na(outl)
        outlb := outl.b
    if n > 0
        for k = n - 1 to 0
            w = wces.get(k)
            keep = w.kept()
            recent = w.b > i - wrecent
            if not keep and not recent
                w.hide()
                if w.b <= i - hold and (w.dead or w.shadow or not wresp)
                    wces.remove(k)
            else
                near = wreach > 0 and not w.dead and math.abs(w.ce - dC[1]) <= reach
                out = w.up ? w.b == outub : w.b == outlb
                live = (near and recent) or out
                if na(w.ln)
                    w.ln := line.new(w.b, w.ce, w.b + wdx(), w.ce)
                if live
                    w.ln.set_x2(bar_index)
                    w.ln.set_width(out ? 2 : 1)
                else
                    w.ln.set_x2(w.b + wdx())
                    w.ln.set_width(1)
    while wces.size() > 400
        w = wces.shift()
        w.hide()

// higher high, lowest low, and nb of bars ago for each one 
method update(Fib f, float hh, float ll, int hi, int li) =>
    ival = hh-ll
    lvl = hh - ival * f.level
    if hi < li
        lvl := hh - ival * (1.0-f.level)
    if na(f.seg)
        f.seg := line.new(last_bar_index+fiboff, lvl, last_bar_index+fiboff+fibwid, lvl, width = 2, color = fibcol)
    else
        f.seg.set_y1(lvl)
        f.seg.set_y2(lvl)
        if barstate.isnew
            f.seg.set_x1(last_bar_index+fiboff)
            f.seg.set_x2(last_bar_index+fiboff+fibwid)
    if na(f.lbl)
        f.lbl := label.new(last_bar_index+fiboff+fibwid, lvl, 
             f.txt(), style=label.style_label_left, yloc = yloc.price,
             color = fibcol, textcolor = textcol, textalign = text.align_right)
    else
        f.lbl.set_y(lvl)
        if barstate.isnew
            f.lbl.set_x(last_bar_index+fiboff+fibwid)
    true

type OTE
    string tf // timeframe
    box otebox = na
    box oterect = na
    array<box> otes = na    // old ones
    int start = na // HTF start time
    bool isshort = false
    float hi = na
    float lo = na


method starts(OTE ote) =>
    timeframe.change(ote.tf)

method isok(OTE ote) =>
    timeframe.in_seconds(ote.tf) > timeframe.in_seconds()

method started(OTE ote) =>
    if not na(ote.otebox)
        ote.otes.push(ote.otebox)
        ote.otebox := na
    ote.start := time
    [o, c, h, l] = request.security(dataSym, ote.tf,
        [open[1], close[1], high[1], low[1]], lookahead = barmerge.lookahead_on)
    oteh = 0.0
    otel = 0.0
    ote.isshort := c < o
    if c > o
        oteh := h - (h-l) * 0.79
        otel := h - (h-l) * 0.62
    else
        oteh := h - (h-l) * (1-0.79)
        otel := h - (h-l) * (1-0.62)
    ote.hi := oteh
    ote.lo := otel
    if ote.otes.size() > 1 and ote.otes.size() >= notes
        b = ote.otes.shift()
        b.delete()

method updatebox(OTE ote) =>
    if  not na(ote.start) and na(ote.otebox)
        ote.otebox := box.new(ote.start, ote.hi, time, ote.lo, text = "OTE "+ote.tf,
            bgcolor = hotecolu, xloc=xloc.bar_time)
        if ote.isshort
            ote.otebox.set_bgcolor(hotecold)
    else if not na(ote.otebox)
        ote.otebox.set_right(time)
 
method updatemargin(OTE ote) =>
    if not na(ote.start) and na(ote.oterect) 
        ote.oterect := box.new(last_bar_index+moteoff, ote.hi, last_bar_index+moteoff+motewid, ote.lo,
            text = "OTE\n"+ote.tf, bgcolor = hotecolu, border_color = na)
        if ote.isshort
            ote.oterect.set_bgcolor(hotecold)
    else if timeframe.change(ote.tf)
        ote.oterect.set_top(ote.hi)
        ote.oterect.set_bottom(ote.lo)
        if ote.isshort
            ote.oterect.set_bgcolor(hotecold)
        else
            ote.oterect.set_bgcolor(hotecolu)
    else if barstate.isnew and not na(ote.oterect)
        ote.oterect.set_left(last_bar_index+moteoff)
        ote.oterect.set_right(last_bar_index+moteoff+motewid)
method updateote(OTE ote) =>
    if na(ote.otes)
        ote.otes := array.new<box>()
    if ote.starts()
        ote.started()
    if notes > 0
        ote.updatebox()
    if domarginotes
        ote.updatemargin()
