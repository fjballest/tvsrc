// Ask for two points for a range, and then
// draw quadrants and proyections on a given range to determine ICT relevant levels.

//@version=6
indicator("ICToctants", overlay=true, 
    max_lines_count    = 500,
    max_labels_count = 500,
    max_bars_back=5000
 
    )

getstyle(s) =>
    switch s
        "solid" => line.style_solid
        "dashed" => line.style_dashed
        "dotted" => line.style_dotted


// Line/Box style
type Sty
    string sty  // solid/dotted/dashed
    int wid     // width
    color col   // color
    color col2 = na   // background or secondary color if any


var G_MAIN = "Main settings"
var t0 = input.time(timestamp("2026-01-01"), title="Select First Candle (Start)", group=G_MAIN, confirm=true)
var t1 = input.time(timestamp("2026-02-01"), title="Select Second Candle (End)", group = G_MAIN, confirm=true)

var nq = input.int(4, "Number of levels (2, 4, 8)", options=[2,4,8], group = G_MAIN, 
     tooltip = "just C.E. or quadrants or octants")
var auto = input.bool(true, "Proyect the range as price moves outside (auto)", group = G_MAIN)
var po3up = input.int(1, "Number of proyections above",  minval=0, maxval=50, 
     group = G_MAIN, tooltip = "number of std.dev. lines to be drawn above the range unless auto")
var po3down = input.int(1, "Number of proyections below", group = G_MAIN, minval=0, maxval=50,
     tooltip = "number of std.dev. lines to be drawn below the range unless auto")

var dorect = input.bool(false, "Draw rectangle", group = G_MAIN, inline = "rect")
var rcol = input.color(color.new(color.blue,90), "", group = G_MAIN, inline = "rect")
var dovline = input.bool(true, "Draw time lines", group=G_MAIN, inline = "vln")
var vxlnsty = getstyle(input.string("solid", "line style", 
    options=["solid", "dashed", "dotted"], group = G_MAIN, inline="vln"))
var vlnwid = input.int(1, "", options = [1, 2, 3], group = G_MAIN, inline = "vln")
var vlncol = input.color(color.black, "", group = G_MAIN, inline = "vln")
var vlnsty = Sty.new(vxlnsty, vlnwid, vlncol)
var xlnsty = getstyle(input.string("solid", "line style", 
    options=["solid", "dashed", "dotted"], group = G_MAIN, inline="dayln"))
var lnwid = input.int(2, "", options = [1, 2, 3], group = G_MAIN, inline = "dayln")
var lncol = input.color(color.blue, "", group = G_MAIN, inline = "dayln")
var lnsty = Sty.new(xlnsty, lnwid, lncol, rcol)

var doalerts = input.bool(false, "alert of candle bounces on levels", group = G_MAIN)
var tfok = timeframe.isminutes and timeframe.multiplier <=60


// Definition of a range
type  RangeInfo
    int t0
    int t1
    Sty sty = na // style


// Instance of a range
type Range
    int startidx = na //  barindex when starts
    int endidx = na // end done
    float hi 
    float lo
    int hiidx = na
    int loidx = na
    int extendidx = na // end for lines that extend another zone
    box rect = na // rectangle drawn if any
    array<line> qs =  na // array of cuadrants and PO3s
    bool drawrect = false
    RangeInfo ri = na // for  range instances
    Sty sty = na// line style
    float minprj = na   // price for min stddev
    float maxprj = na   // price for max stddev
    float wid = na // sep between lines
    line vline0 = na
    line vline1 = na
    

mkRangeInfo(int t0, int t1, Sty sty = na) =>
    RangeInfo zi = na
    zi := RangeInfo.new(math.min(t0,t1), math.max(t0,t1))
    zi.sty := sty
    zi

method starts(RangeInfo zn) =>
    time == zn.t0
 
method ends(RangeInfo zn) =>
    time >= zn.t1


mkRange(RangeInfo zi = na) =>
    zz = Range.new(bar_index, na, high, low, bar_index, bar_index)
    zz.ri := zi
    zz

method mkline(Range zn, float p, bool isx = false) =>
    i = (isx?zn.endidx:zn.startidx)
    ln = line.new(i, p, last_bar_index, p, color = zn.ri.sty.col,
        extend = extend.right,
        width = zn.ri.sty.wid, style=zn.ri.sty.sty)
    zn.qs.push(ln)
   
method mklines(Range zn) =>
    zn.mkline(zn.hi)
    zn.mkline(zn.lo)
    rg = math.abs(zn.hi-zn.lo)
    zn.wid := rg/nq
    p = zn.lo
    zn.minprj := zn.lo
    zn.maxprj := zn.hi
    for i = 0 to nq-1
        p += zn.wid 
        zn.mkline(p)
    p := zn.hi
    for i = 0 to po3up-1
        if po3up <= 0
            break
        p += zn.wid 
        zn.mkline(p, isx=true)
        zn.maxprj := p
    for i = 1 to po3down-1
        if po3down <= 0
            break
        p -= zn.wid 
        zn.mkline(p, isx = true)
        zn.minprj := p



    
method update(Range zn, bool islast = false) =>
    if na(zn.endidx)
        if high >= zn.hi 
            zn.hiidx := bar_index
            zn.hi := high
        if low <= zn.lo
            zn.loidx := bar_index
            zn.lo := low
    if not na(zn.extendidx) and not na(zn.qs)
        for q in zn.qs
            q.set_extend(extend.none)
            q.set_x2(zn.extendidx)
    if not na(zn.endidx) and na(zn.extendidx) and auto
        if low < zn.minprj
            zn.minprj := zn.minprj - zn.wid
            zn.mkline(zn.minprj, isx = true)
        if high > zn.maxprj
            zn.maxprj := zn.maxprj + zn.wid
            zn.mkline(zn.maxprj, isx = true)
    true
method ended(Range zn) =>
    if na(zn.endidx)
        zn.endidx := bar_index-1
    if dovline
        if na(zn.vline0)
            zn.vline0 := line.new(zn.startidx, high, zn.startidx, low, color = vlnsty.col,
                extend = extend.both,
                width = vlnsty.wid, style=vlnsty.sty)
        if na(zn.vline1)
            zn.vline1 := line.new(zn.endidx, high, zn.endidx, low, color = vlnsty.col,
                extend = extend.both,
                width = vlnsty.wid, style=vlnsty.sty)
    if last_bar_index - zn.startidx < 5000
        if na(zn.rect) and dorect
            zn.rect := box.new(zn.startidx, zn.lo, zn.endidx, zn.hi,  border_color = zn.ri.sty.col, bgcolor = zn.ri.sty.col2)
        if na(zn.qs)
            zn.qs := array.new<line>()
        if zn.qs.size() == 0
            zn.mklines()
 
method limit(Range zn, int limit) => 
    if na(zn.extendidx)
        zn.extendidx := limit
        if not na(zn.qs)
            for q in zn.qs
                q.set_extend(extend.none)
                q.set_x2(zn.extendidx)




method kill(Range zn) =>
    if not na(zn.rect)
        zn.rect.delete()
        zn.rect := na
    if not na(zn.qs)
        while zn.qs.size() > 0
            line.delete(zn.qs.shift())
        zn.qs.clear()
        zn.qs := na
    if not na(zn.vline0)
        zn.vline0.delete()
        zn.vline0 := na
    if  not na(zn.vline1)
        zn.vline1.delete()
        zn.vline1 := na


method alerts(Range zn) =>
    if not na(zn) and not na(zn.qs)
        for ln in zn.qs
            y = ln.get_y1()
            if math.abs(y-low) < 2 and close > low+10
                alert("bounce up at octant", alert.freq_once_per_bar)
            if math.abs(y-high) < 2 and close < high-10
                alert("bounce down at octant", alert.freq_once_per_bar)
limitolds(array<Range> ranges) =>
    Range rg = ranges.size() > 1 ? ranges.get(ranges.size()-1) : na
    if not na(rg) and not na(rg.endidx)
        for i = 0 to ranges.size() - 2
            x = ranges.get(i)
            x.limit(bar_index-1)

rd = mkRangeInfo(t0, t1, lnsty)
 
var Range rg = na

if tfok and rd.starts()
    if not na(rg)
        rg.kill()
    rg := mkRange(rd)
 
if tfok and not na(rg) and rd.ends()
    rg.ended()
if tfok and not na(rg)
    rg.update()

if tfok and doalerts
    rg.alerts()
