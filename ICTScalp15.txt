// ICTScalp15: ICTScalp with one more required condition, for charts below 15m:
//  - Long: a candle rejected a bullish 15m FVG or IFVG (its low reached the FVG high or CE and it closed
//    above that level) and no later candle reached that level again; then an IFVG+ is confirmed on the
//    chart timeframe. Short: the same in the other direction (bearish 15m FVG / IFVG, its low or CE).
//  - The 15m FVGs / IFVGs are the 15m FVGs by timeframe of ICTcotas (always on in this script).
// Generated from ICTScalp.txt (mk15.py): everything else is ICTScalp, and changes to ICTScalp apply here.
//
// ICTScalp: enter-long / enter-short signals only, to experiment with scalping conditions.
// It copies, unchanged, the logic of the other indicators (no drawing):
//  - IFVGs and OTE IFVGs: close mode of FVGs+IFVG+OTE (last long FVG -> IFVG- candidate,
//    last short FVG -> IFVG+ candidate, confirmed by a close through it, cancelled by a close back)
//  - Higher timeframe levels: ICTcotas (D / W / M FVGs, IFVGs, VIs and wick CEs, plus the FVGs of
//    15m / 1H / 4H / D / W)
//  - Bias table: IFVGCISDbias (IFVG bias and delivery rows), with every timeframe selected
//
// Short when an IFVG- is confirmed on the chart timeframe, long when an IFVG+ is confirmed.
// Optional conditions (each one can be turned off):
//  - Entry: when price leaves the IFVG (the candle that confirms it) or when price returns to it
//    (a later candle wicks back into it without passing its far edge, once per confirmation)
//  - Only OTE IFVGs (flagged by the OTE logic of FVGs+IFVG+OTE) or any IFVG
//  - Rejection of an ICTcotas level (hi, lo or CE; CE only for wick levels) within the last bars:
//    wick through the level with the whole body on the trade side of it
//  - Agreement of the rightmost columns of the bias table (chart timeframe first)
//  - Wait until the delivery of the chart timeframe matches the direction of the trade
//  - Breakers (the breakers of FVGs+IFVG+OTE): the IFVG must overlap a breaker of the trade direction,
//    and / or the entry is on a chart FVG of the trade direction right above (long) / below (short) an
//    IFVG with a breaker, or on any FVG with a breaker; the entry candle closes at or beyond the CE of
//    the FVG / IFVG used
//  - Took liquidity: a short-term high (short) or low (long) was taken shortly before the entry
//    (short-term pivots and sweep rule of the FVG filter of FVGs+IFVG+OTE)
//  - Time: only inside a time interval, and / or only inside the ICT killzones (the Asia, London,
//    NY AM and NY PM sessions of ICTlevels, same time zone rule)
// Everything is evaluated on closed candles only.
//
// Show trades: entry at the close of the signal candle, stop a few points past the far edge of the
// IFVG, past candle 2 of its FVG, or past the short-term high / low that formed the IFVG, take profit
// at 1:N or at the next short-term high / low that gives at least 1:N. Simulate trades follows each
// trade until the stop or the take profit is hit and flags it (a candle that reaches both counts as
// the stop), optionally moving the stop to breakeven once the trade is positive and trailing it to the
// low / high of each new candle once the trade is a number of points up. Trades are drawn as
// red / green boxes (risk / reward) with the entry, stop and take profit segments, a label above each
// trade with the points to the stop / take profit and the result, and a table with the totals.
//
// Keep in sync: the sections below are copies. When FVGs+IFVG+OTE (FVG detection, pivots, OTE,
// close-mode IFVG), ICTcotas (levels, mitigation) or IFVGCISDbias (ifvgCode, deliveryCode, secPair)
// change, the same change has to be made here.
// ICTScalpStrategy.txt is this script as a strategy: same code, a strategy() declaration and the
// orders at the "strategy orders" mark. A change to either file has to be made to both.

longFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l1
    lo = h3
    blow1 = math.min(o1,c1)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi3 = math.max(o3,c3)
    // include suspension blocks
    fhi = hi
    flo = lo
    if bhi3 < blow2
        lo := bhi3
    if blow1 > bhi2
        hi := blow1
    [hi,lo, fhi, flo]

shortFVGHL(o1,c1,h1,l1,o2,c2,h2,l2,o3,c3,h3,l3) =>
    hi = l3
    lo = h1
    blow3 = math.min(o3,c3)
    bhi2 = math.max(o2,c2)
    blow2 = math.min(o2,c2)
    bhi1 = math.max(o1,c1)
    // include suspension blocks
    fhi = hi
    flo = lo
    if blow3 > bhi2
        hi := blow3
    if bhi1 < blow2
        lo := bhi1
    [hi, lo, fhi, flo]

// only what the IFVG logic needs of an FVG
type FVG
    float hi
    float lo
    float ce
    bool islong
    int start
    int sbar
    float c2hi  // high of candle 2 (middle candle of the FVG)
    float c2lo  // low of candle 2
    bool used = false   // breaker FVG entry taken

FVG lastlongfvg = na
FVG lastshortfvg = na
[gLhi, gLlo, gLfhi, gLflo] = longFVGHL(open[1],close[1],high[1],low[1],
     open[2],close[2],high[2],low[2],
     open[3],close[3],high[3],low[3])
if gLfhi - gLflo > minfvg and gLfhi - gLflo < maxfvg
    lastlongfvg := FVG.new(gLhi, gLlo, math.avg(gLhi, gLlo), true, time[3], bar_index - 3, high[2], low[2])
[gShi, gSlo, gSfhi, gSflo] = shortFVGHL(open[1],close[1],high[1],low[1],
     open[2],close[2],high[2],low[2],
     open[3],close[3],high[3],low[3])
if gSfhi - gSflo > minfvg and gSfhi - gSflo < maxfvg
    lastshortfvg := FVG.new(gShi, gSlo, math.avg(gShi, gSlo), false, time[3], bar_index - 3, high[2], low[2])

tfName(string s) =>
    float m = str.tonumber(s)
    string r = s
    if not na(m)
        r := m >= 60 and m % 60 == 0 ? str.tostring(m / 60, "#") + "h" : str.tostring(m, "#") + "m"
    else if s == "1D"
        r := "D"
    r

type PivSet
    array<int> hb       // pivot-high chart bar index
    array<float> hp     // pivot-high price
    array<int> lb       // pivot-low chart bar index
    array<float> lp     // pivot-low price
    string tag = ""     // timeframe label
    bool htf = false    // higher timeframe set?

method addHigh(PivSet ps, int b, float p) =>
    ps.hb.push(b)
    ps.hp.push(p)
    if ps.hb.size() > 60
        ps.hb.shift()
        ps.hp.shift()

method addLow(PivSet ps, int b, float p) =>
    ps.lb.push(b)
    ps.lp.push(p)
    if ps.lb.size() > 60
        ps.lb.shift()
        ps.lp.shift()

method lastHigh(PivSet ps) =>
    ps.hp.size() > 0 ? ps.hp.last() : na

method lastLow(PivSet ps) =>
    ps.lp.size() > 0 ? ps.lp.last() : na

// chart bar index of the chart candle that opens at (or right before) time t

// Returns [legHigh, legLow] of the leg that precedes the FVG starting at bar sbar.
priorLeg(int dir, int sbar, PivSet ps) =>
    float h = na
    float l = na
    if dir == -1
        int n = ps.hb.size()
        if n > 0
            for i = n - 1 to 0
                int hb = ps.hb.get(i)
                if hb < sbar and bar_index - hb < 4000
                    float lo = low[bar_index - hb]
                    for k = hb to sbar
                        lo := math.min(lo, low[bar_index - k])
                    h := ps.hp.get(i)
                    l := lo
                    break
    else
        int n = ps.lb.size()
        if n > 0
            for i = n - 1 to 0
                int lb = ps.lb.get(i)
                if lb < sbar and bar_index - lb < 4000
                    float hi = high[bar_index - lb]
                    for k = lb to sbar
                        hi := math.max(hi, high[bar_index - k])
                    l := ps.lp.get(i)
                    h := hi
                    break
    [h, l]

// OTE tracker (FVGs+IFVG+OTE without the bands)
type OteTrack
    int dir = 0
    float zlo = na      // IFVG low
    float zhi = na      // IFVG high
    float legHi = na
    float legLo = na
    float rext = na     // running retrace extreme since the FVG (high for dir -1, low for dir +1)
    bool armed = false  // IFVG confirmed by a close through it
    bool fired = false  // OTE signal already fired for this confirmation
    int ctime = na      // confirmation time
    int cbar = na       // confirmation bar index
    float pHi = na      // prior leg high (leg before the FVG)
    float pLo = na      // prior leg low
    int sbar = na       // bar index where the FVG starts
    bool rejHit = false // confirming candle rejected the OTE of the prior leg
    string tag = ""     // timeframe label

method oteStart(OteTrack t, int dir, float zlo, float zhi, float origin, int sbar, PivSet ps) =>
    t.rejHit := false
    t.sbar := sbar
    t.tag := ps.tag
    [pgh, pgl] = priorLeg(dir, sbar, ps)
    t.pHi := pgh
    t.pLo := pgl
    float o = origin
    if ps.htf
        float adj = dir == -1 ? ps.lastHigh() : ps.lastLow()
        if not na(adj)
            o := dir == -1 ? math.max(origin, adj) : math.min(origin, adj)
    t.dir := dir
    t.zlo := zlo
    t.zhi := zhi
    t.armed := false
    t.fired := false
    t.ctime := na
    t.rext := origin
    t.legHi := dir == -1 ? o : na
    t.legLo := dir == 1 ? o : na

method oteConfirm(OteTrack t) =>
    if t.dir != 0 and not t.armed
        // rejection of the OTE of the PRIOR leg by the confirming candle
        if useOTE and oteRej and not na(t.pHi) and not na(t.pLo) and t.pHi > t.pLo
            float pr = t.pHi - t.pLo
            float ext = na
            float rLo = na
            float rHi = na
            bool hit = false
            if t.dir == -1
                ext := oteRejAny ? math.max(t.rext, high) : high
                rLo := t.pLo + oteMin * pr
                rHi := t.pLo + oteMax * pr
                hit := ext >= rLo and ext <= (oteRejCap ? rHi : t.pHi) and close <= rHi
            else
                ext := oteRejAny ? math.min(t.rext, low) : low
                rLo := t.pHi - oteMax * pr
                rHi := t.pHi - oteMin * pr
                hit := ext <= rHi and ext >= (oteRejCap ? rLo : t.pLo) and close >= rLo
            if hit
                t.rejHit := true
        t.armed := true
        t.fired := t.rejHit // one OTE signal per IFVG
        t.ctime := time
        t.cbar := bar_index
        if t.dir == -1
            t.legHi := math.max(t.legHi, high)
            t.legLo := low
        else
            t.legLo := math.min(t.legLo, low)
            t.legHi := high

method oteCancel(OteTrack t) =>
    t.rejHit := false
    t.armed := false
    if t.dir == -1
        t.legLo := na
    else
        t.legHi := na

// call once per bar. Returns the direction (-1 / 1) on the bar the OTE is matched, else 0
method oteStep(OteTrack t) =>
    int s = 0
    if useOTE and t.dir != 0
        if not t.armed
            if t.dir == -1
                t.legHi := math.max(t.legHi, high)
                t.rext := math.max(t.rext, high)
            else
                t.legLo := math.min(t.legLo, low)
                t.rext := math.min(t.rext, low)
        else
            if t.rejHit
                s := t.dir
                t.rejHit := false
            float r = t.legHi - t.legLo
            if r > 0
                float bLo = na
                float bHi = na
                if t.dir == -1
                    bLo := t.legLo + oteMin * r
                    bHi := t.legLo + oteMax * r
                else
                    bLo := t.legHi - oteMax * r
                    bHi := t.legHi - oteMin * r
                float mid = math.avg(t.zlo, t.zhi)
                bool m = false
                if oteMode == "Overlap"
                    m := t.zhi >= bLo and t.zlo <= bHi
                else if oteMode == "Fully inside"
                    m := t.zlo >= bLo and t.zhi <= bHi
                else
                    m := mid >= bLo and mid <= bHi
                bool tap = t.dir == -1 ? high >= t.zlo : low <= t.zhi
                if m and tap and not t.fired and time != t.ctime and barstate.isconfirmed
                    t.fired := true
                    s := t.dir
            if t.dir == -1
                t.legLo := math.min(t.legLo, low)
            else
                t.legHi := math.max(t.legHi, high)
    s

type OteGroup
    OteTrack t0
    OteTrack t1
    OteTrack t2
    bool flagged = false
    string tags = ""

method gStart(OteGroup g, int dir, float zlo, float zhi, float origin, int sbar) =>
    g.flagged := false
    g.tags := ""
    g.t0.oteStart(dir, zlo, zhi, origin, sbar, pv0)
    if hasH1
        g.t1.oteStart(dir, zlo, zhi, origin, sbar, pv1)
    if hasH2
        g.t2.oteStart(dir, zlo, zhi, origin, sbar, pv2)

method gConfirm(OteGroup g) =>
    g.t0.oteConfirm()
    g.t1.oteConfirm()
    g.t2.oteConfirm()

method gCancel(OteGroup g) =>
    g.flagged := false
    g.tags := ""
    g.t0.oteCancel()
    g.t1.oteCancel()
    g.t2.oteCancel()

method gStep(OteGroup g) =>
    int s0 = g.t0.oteStep()
    int s1 = g.t1.oteStep()
    int s2 = g.t2.oteStep()
    int s = 0
    if not g.flagged and (s0 != 0 or s1 != 0 or s2 != 0)
        g.flagged := true
        s := s0 != 0 ? s0 : (s1 != 0 ? s1 : s2)
        string tg = ""
        if s0 != 0
            tg += g.t0.tag
        if s1 != 0
            tg += (tg == "" ? "" : "+") + g.t1.tag
        if s2 != 0
            tg += (tg == "" ? "" : "+") + g.t2.tag
        g.tags := tg
    s

type Lvl
    int kind    // 0 FVG / IFVG, 1 volume imbalance, 2 wick, 3 FVG by timeframe
    int tfi     // 0 daily, 1 weekly, 2 monthly, 3.. 15m / 1H / 4H / D / W
    int dir     // 1 bullish / support, -1 bearish / resistance
    float hi
    float lo
    float ce
    int t
    float wsize = na
    bool inv = false    // FVG inverted into an IFVG
    bool dead = false
    int tfm = na      // formation time (close of the last candle of the FVG)

var array<Lvl> lvls = array.new<Lvl>()
var array<Lvl> wcur = array.new<Lvl>(12, na)   // current wick group level: tf * 2 + side

var array<int> mfSecs = array.from(900, 3600, 14400, 86400, 604800, 2628003)
var array<string> mfTfs = array.from("15", "60", "240", "D", "W", "M")
var array<bool> mfSel = array.from(mf15, mf1h, mf4h, mfd, mfw, mfm)
var array<int> mfLastT = array.new<int>(6, na)
// per level timeframe (tfi 0..8): a candle of it completed on this chart bar, and its close.
// An FVG inverts into an IFVG only on a close of its own timeframe, never on a chart close (as in ICTcotas).
array<bool> tfNew = array.new<bool>(9, false)
var array<float> tfClose = array.new<float>(9, na)

// only timeframes above the chart are used, never the chart's own
mfOk(int s) =>
    s > tfsecs

mfEnabled(int i) =>
    int s = mfSecs.get(i)
    bool on = (domf and mfOk(s) and mfSel.get(i)) or (i == 0 and mfOk(s))
    if domf and s > tfsecs and mfnext > 0
        int rank = 0
        for j = 0 to i - 1
            if mfSecs.get(j) > tfsecs
                rank += 1
        if rank < mfnext
            on := true
    on

wkEnabled(int i) =>
    int s = mfSecs.get(i)
    dowk and mfOk(s) and ((i == 1 and wk1h) or (i == 2 and wk4h))

method kill(Lvl l) =>
    l.dead := true
    true

fvgOk(int kind, int tfi, int dir, float lo, float hi) =>
    bool ok = true
    if fvgovl != "Keep all"
        if fvgovl == "Enclosing"
            for o in lvls
                if o.kind == kind and o.tfi == tfi and o.dir == dir and not o.inv and not o.dead
                    if lo <= o.hi and hi >= o.lo and o.lo <= lo and o.hi >= hi
                        ok := false
        if ok
            int i = lvls.size() - 1
            while i >= 0
                Lvl o = lvls.get(i)
                if o.kind == kind and o.tfi == tfi and o.dir == dir and not o.inv and not o.dead
                    if lo <= o.hi and hi >= o.lo
                        o.kill()
                        lvls.remove(i)
                i -= 1
    ok

addLvl(int kind, int tfi, int dir, float lo, float hi, int t) =>
    Lvl l = na
    if (kind != 0 and kind != 3) or fvgOk(kind, tfi, dir, lo, hi)
        l := Lvl.new(kind, tfi, dir, hi, lo, (hi + lo) / 2, math.max(t, tfirst))
        lvls.push(l)
        int keep = kind == 0 ? nfvg : kind == 1 ? nvi : kind == 3 ? nmf : nwk
        int cnt = 0
        for v in lvls
            if v.kind == kind and v.tfi == tfi and not v.dead
                cnt += 1
        if cnt > keep
            for i = 0 to lvls.size() - 1
                Lvl o = lvls.get(i)
                if o.kind == kind and o.tfi == tfi and not o.dead
                    o.kill()
                    lvls.remove(i)
                    break
    l

wickStep(int tfi, int side, bool q, float sz, float lo, float hi, int t) =>
    int idx = tfi * 2 + side
    Lvl cur = wcur.get(idx)
    if not q
        Lvl none = na
        wcur.set(idx, none)
    else if na(cur) or (cur.dead and sz > cur.wsize)
        Lvl n = addLvl(2, tfi, side == 0 ? -1 : 1, lo, hi, t)
        n.wsize := sz
        wcur.set(idx, n)
    else if sz > cur.wsize
        cur.lo := lo
        cur.hi := hi
        cur.ce := (lo + hi) / 2
        cur.t := math.max(t, tfirst)
        cur.wsize := sz
    true

addFvgs(int kind, int tfi, float minsz, float o1, float h1, float l1, float c1,
     float o2, float h2, float l2, float c2, float h3, float l3, float o3, float c3,
     int t2, int tform) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float top3 = math.max(o3, c3)
    float bot3 = math.min(o3, c3)
    if l1 > h3 and l1 - h3 >= minsz
        float flo = h3
        float fhi = l1
        if fvgvi
            if botP > top3 and botP - top3 >= vimin
                flo := math.min(flo, top3)
                fhi := math.max(fhi, botP)
            if botC > topP and botC - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, botC)
        Lvl nb = addLvl(kind, tfi, 1, flo, fhi, t2)
        if not na(nb)
            nb.tfm := tform
    if h1 < l3 and l3 - h1 >= minsz
        float flo = h1
        float fhi = l3
        if fvgvi
            if topP < bot3 and bot3 - topP >= vimin
                flo := math.min(flo, topP)
                fhi := math.max(fhi, bot3)
            if topC < botP and botP - topC >= vimin
                flo := math.min(flo, topC)
                fhi := math.max(fhi, botP)
        Lvl ns = addLvl(kind, tfi, -1, flo, fhi, t2)
        if not na(ns)
            ns.tfm := tform
    true

addWicks(int tfi, float o1, float h1, float l1, float c1, int t1) =>
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    float rng = h1 - l1
    float uw = h1 - topC
    float lw = botC - l1
    bool uq = rng > 0 and uw >= wkmin / 100.0 * rng
    bool lq = rng > 0 and lw >= wkmin / 100.0 * rng
    wickStep(tfi, 0, uq, uw, topC, h1, t1)
    wickStep(tfi, 1, lq, lw, l1, botC, t1)
    true

processTf(int tfi, float o1, float h1, float l1, float c1, int t1,
     float o2, float h2, float l2, float c2, int t2, float h3, float l3, float o3, float c3) =>
    float topP = math.max(o2, c2)
    float botP = math.min(o2, c2)
    float topC = math.max(o1, c1)
    float botC = math.min(o1, c1)
    if dofvg
        addFvgs(0, tfi, fvgmin, o1, h1, l1, c1, o2, h2, l2, c2, h3, l3, o3, c3, t2, time)
    if dovi
        if botC > topP and l1 <= h2 and botC - topP >= vimin
            addLvl(1, tfi, 1, topP, botC, t1)
        if topC < botP and h1 >= l2 and botP - topC >= vimin
            addLvl(1, tfi, -1, topC, botP, t1)
    if dowk
        addWicks(tfi, o1, h1, l1, c1, t1)
    true

runMit() =>
    int i = lvls.size() - 1
    while i >= 0
        Lvl l = lvls.get(i)
        if l.dead
            lvls.remove(i)
        else
            bool fvgLive = (l.kind == 0 or l.kind == 3) and doifvg and not l.inv
            bool flipped = false
            if fvgLive and tfNew.get(l.tfi)
                float c = tfClose.get(l.tfi)
                if (l.dir == 1 and c < l.lo) or (l.dir == -1 and c > l.hi)
                    l.inv := true
                    l.dir := -l.dir
                    flipped := true
            if mmode > 0 and not fvgLive and not flipped
                bool m = false
                // close rule: on a close of the level's own timeframe, never on a chart close
                bool tn = tfNew.get(l.tfi)
                float tc = tfClose.get(l.tfi)
                if l.dir == 1
                    m := mmode == 1 ? low <= l.ce : mmode == 2 ? low <= l.lo : tn and tc < l.lo
                else
                    m := mmode == 1 ? high >= l.ce : mmode == 2 ? high >= l.hi : tn and tc > l.hi
                if m
                    l.kill()
                    lvls.remove(i)
        i -= 1
    true

         open[2], high[2], low[2], close[2], time[2], high[3], low[3], open[3], close[3]],
         lookahead = barmerge.lookahead_on)

     : timeframe.period)

// rejection of a level by this candle, tested against the levels alive before the mitigation of this bar
// (a wick that fills a level also rejects it). Short: wick up to the level, whole body below it.
// Long: wick down to the level, whole body above it. Wick levels count by their CE only.
rejCheck() =>
    bool bear = false
    bool bull = false
    float bTop = math.max(open, close)
    float bBot = math.min(open, close)
    for l in lvls
        if not l.dead
            for k = 0 to 2
                if l.kind != 2 or k == 2
                    float p = k == 0 ? l.hi : k == 1 ? l.lo : l.ce
                    if high >= p and bTop < p
                        bear := true
                    if low <= p and bBot > p
                        bull := true
    [bear, bull]

// 15m FVG / IFVG rejection, tested like rejCheck against the levels alive before the mitigation of this bar.
// Long: a bullish 15m FVG / IFVG whose CE (or else its high) is reached by the low of the candle, with the
// close above that level. Short: a bearish one whose CE (or else its low) is reached by the high, with the
// close below it. When several levels are rejected, the one closest to the close is kept.
rej15() =>

    float pS = na
    for l in lvls

                float p = high >= l.ce ? l.ce : high >= l.lo ? l.lo : na
                if not na(p) and close < p
                    pS := na(pS) ? p : math.min(pS, p)
    [pL, pS]

// Returns 0 = none confirmed, 1 = IFVG+ only, 2 = IFVG- only, 3 = both.
ifvgCode() =>
    var float bearHi = na
    var float bearLo = na
    var bool bearConf = false
    var float bullHi = na
    var float bullLo = na
    var bool bullConf = false

    [lhi, llo, lfhi, lflo] = longFVGHL(open[1],close[1],high[1],low[1],
         open[2],close[2],high[2],low[2],
         open[3],close[3],high[3],low[3])
    if lfhi - lflo > bMinfvg and lfhi - lflo < bMaxfvg
        bearHi := lhi
        bearLo := llo
        bearConf := false

    [shi, slo, sfhi, sflo] = shortFVGHL(open[1],close[1],high[1],low[1],
         open[2],close[2],high[2],low[2],
         open[3],close[3],high[3],low[3])
    if sfhi - sflo > bMinfvg and sfhi - sflo < bMaxfvg
        bullHi := shi
        bullLo := slo
        bullConf := false

    if bullConf and close < bullLo
        bullConf := false
    if not na(bullHi) and not bullConf and close > bullHi
        bullConf := true

    if bearConf and close > bearHi
        bearConf := false
    if not na(bearLo) and not bearConf and close < bearLo
        bearConf := true

    (bullConf ? 1 : 0) + (bearConf ? 2 : 0)

// Returns (st + 1) + (inRng ? 3 : 0): 0 bearish, 1 undefined, 2 bullish, +3 range-locked.
deliveryCode() =>
    var int st = 0
    var float ref = na
    var float refLo = na
    var float refHi = na
    var bool conf = true
    var int flips = 0
    var bool inRng = false
    var float rTop = na
    var float rBot = na
    var float sTop = na
    var float sBot = na
    var int pend = 0
    var float pOpen = na
    var float pLvl = na
    var float pNewOpen = na
    var float pNewHi = na
    var float pNewLo = na
    var float lastUpLo = na
    var float lastUpHi = na
    var float lastDnLo = na
    var float lastDnHi = na

    bool up = close > open
    bool dn = close < open

    // inner candle detection (all ranges are those of the raw previous candles)
    bool useHL = innerRange == "High-Low"
    float xLo = math.min(open, close)
    float xHi = math.max(open, close)
    bool inner = false
    if exclInner and bar_index > 0
        float pLo = useHL ? low[1] : math.min(open[1], close[1])
        float pHi = useHL ? high[1] : math.max(open[1], close[1])
        inner := xLo >= pLo and xHi <= pHi
        if not inner and dn and not na(lastUpLo)
            inner := xLo >= lastUpLo and xHi <= lastUpHi
        if not inner and up and not na(lastDnLo)
            inner := xLo >= lastDnLo and xHi <= lastDnHi

    if inner
        // excluded candle: no reference, no continuation, no change of state
        bool skipped = true
    else if st == 0
        if up or dn
            st := up ? 1 : -1
            ref := open
            refLo := low
            refHi := high
            conf := true
            flips := 0
    else if inRng
        // range locked: only a close beyond the top / bottom open changes the delivery
        if up and close > rTop
            st := 1
            ref := open
            refLo := low
            refHi := high
            inRng := false
            conf := true
            flips := 0
            sTop := na
            sBot := na
            pend := 0
        else if dn and close < rBot
            st := -1
            ref := open
            refLo := low
            refHi := high
            inRng := false
            conf := true
            flips := 0
            sTop := na
            sBot := na
            pend := 0
        else
            rTop := math.max(rTop, open)
            rBot := math.min(rBot, open)
    else
        bool flip = false
        float oRef = ref        // level broken by the change of state
        float nRef = open       // new reference (open of the candle of the new direction)
        float nHi = high
        float nLo = low
        if st == 1
            if pend == -1
                if dn
                    pNewOpen := open
                    pNewHi := high
                    pNewLo := low
                if close < pOpen and low < pLvl
                    flip := true
                    oRef := pOpen
                    nRef := pNewOpen
                    nHi := pNewHi
                    nLo := pNewLo
                else if close > pOpen or up
                    pend := 0
            if not flip
                if up
                    if open > ref
                        // continuation: up-close candle offering a higher open
                        conf := true
                        flips := 0
                        sTop := na
                        sBot := na
                    ref := open
                    refLo := low
                    refHi := high
                else if dn and close < ref
                    if not deferOn or low < refLo
                        flip := true
                    else
                        pend := -1
                        pOpen := ref
                        pLvl := refLo
                        pNewOpen := open
                        pNewHi := high
                        pNewLo := low
        else
            if pend == 1
                if up
                    pNewOpen := open
                    pNewHi := high
                    pNewLo := low
                if close > pOpen and high > pLvl
                    flip := true
                    oRef := pOpen
                    nRef := pNewOpen
                    nHi := pNewHi
                    nLo := pNewLo
                else if close < pOpen or dn
                    pend := 0
            if not flip
                if dn
                    if open < ref
                        // continuation: down-close candle offering a lower open
                        conf := true
                        flips := 0
                        sTop := na
                        sBot := na
                    ref := open
                    refLo := low
                    refHi := high
                else if up and close > ref
                    if not deferOn or high > refHi
                        flip := true
                    else
                        pend := 1
                        pOpen := ref
                        pLvl := refHi
                        pNewOpen := open
                        pNewHi := high
                        pNewLo := low
        if flip
            if conf
                flips := 1
                sTop := math.max(oRef, nRef)
                sBot := math.min(oRef, nRef)
            else
                flips += 1
                sTop := math.max(nz(sTop, oRef), math.max(oRef, nRef))
                sBot := math.min(nz(sBot, oRef), math.min(oRef, nRef))
            conf := false
            st := -st
            ref := nRef
            refLo := nLo
            refHi := nHi
            pend := 0
            if flips >= lockN
                inRng := true
                rTop := sTop
                rBot := sBot

    // remember the range of this candle for the next ones
    if up
        lastUpLo := useHL ? low : xLo
        lastUpHi := useHL ? high : xHi
    else if dn
        lastDnLo := useHL ? low : xLo
        lastDnHi := useHL ? high : xHi

    (st + 1) + (inRng ? 3 : 0)

type Breaker
    int dir         // 1 bullish (long), -1 bearish (short)
    float hi        // zone
    float lo
    bool broken = false
    int bkBar = na  // bar of the close beyond the zone
    bool swung = false  // a swing beyond the zone formed after the break: entries allowed

// zone from candle k bars back: the up-close (dir 1) / down-close (dir -1) candle there or the nearest one
// before it (not before bar 'from'), plus the consecutive ones before it in "Whole sequence" mode
brkZone(int dir, int k0, int from) =>
    float zh = na
    float zl = na
    bool found = false
    int kmax = math.min(bar_index - from, 300)
    int k = k0
    while k <= kmax and not found
        if (close[k] - open[k]) * dir > 0
            zh := high[k]
            zl := low[k]
            found := true
        k += 1
    if found and brkMode == "Whole sequence"
        while k <= kmax and (close[k] - open[k]) * dir > 0
            zh := math.max(zh, high[k])
            zl := math.min(zl, low[k])
            k += 1
    [zh, zl, found]

brkAdd(int dir, float zh, float zl) =>
    brks.push(Breaker.new(dir, zh, zl))
    int cnt = 0
    for b in brks
        if b.dir == dir
            cnt += 1
    if cnt > brkKeep
        for i = 0 to brks.size() - 1
            if brks.get(i).dir == dir
                brks.remove(i)
                break
    true

if brkUse and barstate.isconfirmed
    if not na(bPL)
        bPrevL := bLastL
        bPrevLb := bLastLb
        bLastL := bPL
        bLastLb := bar_index - brkLen
        // bullish: low, high, lower low (the high between the two lows)
        if not na(bPrevL) and bLastL < bPrevL and not na(bLastHb) and bLastHb > bPrevLb and bLastHb < bLastLb
            [zh, zl, ok] = brkZone(1, bar_index - bLastHb, bPrevLb)
            if ok
                brkAdd(1, zh, zl)
    if not na(bPH)
        bPrevH := bLastH
        bPrevHb := bLastHb
        bLastH := bPH
        bLastHb := bar_index - brkLen
        // bearish: high, low, higher high (the low between the two highs)
        if not na(bPrevH) and bLastH > bPrevH and not na(bLastLb) and bLastLb > bPrevHb and bLastLb < bLastHb
            [zh, zl, ok] = brkZone(-1, bar_index - bLastLb, bPrevHb)
            if ok
                brkAdd(-1, zh, zl)
    if brks.size() > 0
        for i = brks.size() - 1 to 0
            Breaker b = brks.get(i)
            if not b.broken
                if (b.dir == 1 and close > b.hi) or (b.dir == -1 and close < b.lo)
                    b.broken := true
                    b.bkBar := bar_index
            else if (b.dir == 1 and close < b.lo) or (b.dir == -1 and close > b.hi)
                brks.remove(i)  // closed through the other side: invalidated
            else if not b.swung
                // after the break: a swing high above the zone (bullish) / low below it (bearish)
                if b.dir == 1 and not na(bPH) and bar_index - brkLen >= b.bkBar and bPH > b.hi
                    b.swung := true
                else if b.dir == -1 and not na(bPL) and bar_index - brkLen >= b.bkBar and bPL < b.lo
                    b.swung := true

// a breaker of direction dir (broken, swing made after the break) overlaps f
brkHas(int dir, FVG f) =>

    if not na(f)
        for b in brks
            if b.dir == dir and b.swung and f.hi >= b.lo and f.lo <= b.hi
                r := true
    r

                     ? f.hi > iv.hi and f.lo >= iv.lo and f.lo - iv.hi <= ih
                     : f.lo < iv.lo and f.hi <= iv.hi and iv.lo - f.hi <= ih)

    else if dlvCur == 1
        goLong := true
// time filters (session definitions and time zone rule of ICTlevels)
string tkr = str.upper(syminfo.ticker)
string dsc = str.upper(syminfo.description)

     or str.contains(tkr, "FDX")

string asiastr = useUS ? "2000-0000" : "2300-0600"

     or (kzLondon and not na(time(timeframe.period, londonstr, tz)))
     or (kzNyam and not na(time(timeframe.period, nyamstr, tz)))
     or (kzNypm and not na(time(timeframe.period, nypmstr, tz)))

// ============================================================ trades
type Trade
    int dir         // 1 long, -1 short
    float entry
    float stop      // current stop (moved by breakeven / trailing)
    float tp
    int bar         // entry bar
    line eln
    line sln
    line tln
    bool done = false
    box rbx = na    // risk box (entry to initial stop)
    box gbx = na    // reward box (entry to take profit)
    float best = na // best price reached after the entry
    bool trail = false
    label lbl = na  // points label above the trade
    string ltxt = ""

// totals of the simulated trades
var int nWin = 0
var int nLoss = 0
var int nBe = 0
var float pWin = 0.0
var float pLoss = 0.0

var array<Trade> trades = array.new<Trade>()

// stop of an open trade after a closed candle: breakeven once a candle closes in profit by more than
// beMin, trailing to the low (long) / high (short) of each candle once price went trPts in favour (best
// is the best price reached so far). The stop never moves backwards. Shared with the strategy orders.
manageStop(int dir, float entry, float stp, bool trail, float best) =>
    float st = stp
    bool tr = trail
    if beOn and (close - entry) * dir > beMin
        st := dir == 1 ? math.max(st, entry) : math.min(st, entry)
    if trOn and (best - entry) * dir >= trPts
        tr := true
    if tr
        st := dir == 1 ? math.max(st, low) : math.min(st, high)
    [st, tr]

// follow the open trades (from the candle after the entry): the stop and take profit are tested with
// the levels set at the previous close, then the stop is managed with this candle
if showTrades and simulate and barstate.isconfirmed
    for t in trades
        if not t.done and bar_index > t.bar
            bool hitS = t.dir == -1 ? high >= t.stop : low <= t.stop
            bool hitT = not t.trail and (t.dir == -1 ? low <= t.tp : high >= t.tp)
            t.eln.set_x2(bar_index)
            t.sln.set_x2(bar_index)
            t.tln.set_x2(bar_index)
            if not na(t.rbx)
                t.rbx.set_right(bar_index)
                t.gbx.set_right(bar_index)
            if hitS or hitT
                t.done := true
                float ex = hitS ? t.stop : t.tp
                float pts = (ex - t.entry) * t.dir
                string res = pts > 0 ? "WIN" : pts < 0 ? "LOSS" : "BE"
                if not na(t.gbx) and (ex - t.tp) * t.dir > 0
                    if t.dir == 1
                        t.gbx.set_top(ex)
                    else
                        t.gbx.set_bottom(ex)
                if pts > 0
                    nWin += 1
                    pWin += pts
                else if pts < 0
                    nLoss += 1
                    pLoss += pts
                else
                    nBe += 1
                color rc = pts > 0 ? tpCol : pts < 0 ? stopCol : entryCol
                if not na(t.lbl)
                    t.lbl.set_text(t.ltxt + "\n" + res + " " + (pts > 0 ? "+" : "") + str.tostring(pts, "#.##"))
                    t.lbl.set_color(rc)
                else
                    label.new(bar_index, ex, res + " " + str.tostring(pts, "#.##"),
                         style = label.style_label_left, color = color.new(color.white, 100),
                         textcolor = rc, size = size.small)
            else if beOn or trOn
                t.best := na(t.best) ? (t.dir == 1 ? high : low)
                     : (t.dir == 1 ? math.max(t.best, high) : math.min(t.best, low))
                [st, tr] = manageStop(t.dir, t.entry, t.stop, t.trail, t.best)
                t.stop := st
                t.trail := tr
                t.sln.set_y1(st)
                t.sln.set_y2(st)

// short-term high (dir -1) / low (dir 1) formed from bar b0 to this bar
swingFrom(int dir, int b0) =>
    float r = na
    array<int> bs = dir == -1 ? sthB : stlB
    array<float> ps = dir == -1 ? sthP : stlP
    int n = bs.size()
    if n > 0
        for i = n - 1 to 0
            int b = bs.get(i)
            if b < b0
                break
            float p = ps.get(i)
            r := na(r) ? p : (dir == -1 ? math.max(r, p) : math.min(r, p))
    if na(r)
        int k0 = math.min(bar_index - b0, 4900)
        for k = 0 to k0
            r := na(r) ? (dir == -1 ? high[k] : low[k]) : (dir == -1 ? math.max(r, high[k]) : math.min(r, low[k]))
    r

// stop and take profit of a trade entered at the close of this candle; ok = false when the stop is
// not on the losing side of the entry
// safest of candle 2 and the wicks against the trade (dir -1: highs, dir 1: lows) that reached the IFVG,
// from candle 2 to this bar
c2Wicks(int dir, FVG f) =>
    float r = dir == -1 ? f.c2hi : f.c2lo
    int k0 = math.min(bar_index - f.sbar - 1, 4900)
    if k0 >= 0
        for k = 0 to k0
            if dir == -1 and high[k] >= f.lo
                r := math.max(r, high[k])
            else if dir == 1 and low[k] <= f.hi
                r := math.min(r, low[k])
    r

tradeLevels(int dir, FVG f) =>
    float lvl = stopMode == "Candle 2" ? (dir == -1 ? f.c2hi : f.c2lo)
         : stopMode == "Candle 2 and IFVG wicks" ? c2Wicks(dir, f)
         : stopMode == "Short-term high / low" ? swingFrom(dir, f.sbar)
         : (dir == -1 ? f.hi : f.lo)
    float stp = lvl - dir * stopDelta
    float risk = (close - stp) * dir
    float tp = close + dir * rrN * risk
    // nearest short-term high (long) / low (short) not taken yet, at least 1:N away
    if tpMode == "Next short-term high / low"
        float nx = na
        array<float> qs = dir == 1 ? uhP : ulP
        for q in qs
            if (q - tp) * dir >= 0
                nx := na(nx) ? q : (dir == 1 ? math.min(nx, q) : math.max(nx, q))
        if not na(nx)
            tp := nx
    [stp, tp, risk > 0]

newTrade(int dir, FVG f) =>
    [stp, tp, ok] = tradeLevels(dir, f)
    if ok
        int x2 = simulate ? bar_index : bar_index + segLen
        line e = line.new(bar_index, close, x2, close, color = entryCol, width = 2)
        line sl = line.new(bar_index, stp, x2, stp, color = stopCol, width = 2)
        line tl = line.new(bar_index, tp, x2, tp, color = tpCol, width = 2)
        box rb = na
        box gb = na
        if boxOn
            rb := box.new(bar_index, math.max(close, stp), x2, math.min(close, stp), border_width = 0,
                 bgcolor = color.new(stopCol, boxTr))
            gb := box.new(bar_index, math.max(close, tp), x2, math.min(close, tp), border_width = 0,
                 bgcolor = color.new(tpCol, boxTr))
        label lb = na
        string lt = "SL " + str.tostring(math.abs(close - stp), "#.##") + "   TP "
             + str.tostring(math.abs(tp - close), "#.##")
        if lblOn
            lb := label.new(bar_index, math.max(close, stp, tp), lt, style = label.style_label_down,
                 color = color.new(entryCol, 10), textcolor = color.white, size = size.normal)
        trades.push(Trade.new(dir, close, stp, tp, bar_index, e, sl, tl, not simulate, rb, gb, na, false, lb, lt))
        // keep the list short; old lines are removed by the 500 line limit
        if trades.size() > 160
            trades.shift()
    true

if showTrades and goShort
    newTrade(-1, entS)
if showTrades and goLong
    newTrade(1, entL)
